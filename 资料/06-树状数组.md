## 1. 基础版：单点修改 + 前缀和 / 区间和

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

struct BIT {
    int n;
    vector<ll> t;
    explicit BIT(int n_) : n(n_), t(n_ + 1, 0) {}

    void add(int i, ll v) {                 // 下标 i 从 1 开始
        for (; i <= n; i += i & -i) t[i] += v;
    }

    ll sum(int i) const {                   // [1, i]
        ll s = 0;
        for (; i > 0; i -= i & -i) s += t[i];
        return s;
    }

    ll range(int l, int r) const { return sum(r) - sum(l - 1); }

    // 建立：O(n)，比 n 次 add 快
    void build(const vector<ll> &a) {       // a 下标 1..n
        for (int i = 1; i <= n; ++i) {
            t[i] += a[i];
            int j = i + (i & -i);
            if (j <= n) t[j] += t[i];
        }
    }
};
```

**注意**：`build` 要求 `t` 初始为 0，且 `a` 是下标 1..n 的数组。

## 2. 区间修改 + 单点查询（差分 BIT）

维护差分数组 `d[i] = a[i] - a[i-1]`：区间加 `[l,r] += v` ⇒ `d[l] += v, d[r+1] -= v`；单点查询 = 前缀和。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

struct BITD {
    int n;
    vector<ll> t;
    explicit BITD(int n_) : n(n_), t(n_ + 2, 0) {}
    void add(int i, ll v) { for (; i <= n; i += i & -i) t[i] += v; }
    ll sum(int i) const { ll s = 0; for (; i > 0; i -= i & -i) s += t[i]; return s; }
    void range_add(int l, int r, ll v) { add(l, v); if (r + 1 <= n) add(r + 1, -v); }
    ll point_query(int i) const { return sum(i); }
};
```

## 3. 区间修改 + 区间查询 ★

维护两个 BIT：`B1` 存 `d[i]`，`B2` 存 `d[i] * i`。

```
前缀和 S(x) = Σ_{i=1}^{x} a[i] = (x+1) * Σ d[i] - Σ d[i]*i
```

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

struct BITRangeAddRangeSum {
    int n;
    vector<ll> b1, b2;
    explicit BITRangeAddRangeSum(int n_) : n(n_), b1(n_ + 2, 0), b2(n_ + 2, 0) {}

    void _add(vector<ll> &b, int i, ll v) {
        for (; i <= n; i += i & -i) b[i] += v;
    }
    ll _sum(const vector<ll> &b, int i) const {
        ll s = 0; for (; i > 0; i -= i & -i) s += b[i]; return s;
    }

    void range_add(int l, int r, ll v) {
        _add(b1, l, v); _add(b2, l, v * (l - 1));
        if (r + 1 <= n) { _add(b1, r + 1, -v); _add(b2, r + 1, -v * r); }
    }
    ll prefix_sum(int x) const {          // [1, x]
        return _sum(b1, x) * x - _sum(b2, x);
    }
    ll range_sum(int l, int r) const { return prefix_sum(r) - prefix_sum(l - 1); }
};
```

## 4. 二维树状数组（单点改 + 子矩阵和）

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

struct BIT2D {
    int n, m;
    vector<vector<ll>> t;
    BIT2D(int n_, int m_) : n(n_), m(m_), t(n_ + 1, vector<ll>(m_ + 1, 0)) {}

    void add(int x, int y, ll v) {
        for (int i = x; i <= n; i += i & -i)
            for (int j = y; j <= m; j += j & -j)
                t[i][j] += v;
    }
    ll sum(int x, int y) const {          // [1..x][1..y]
        ll s = 0;
        for (int i = x; i > 0; i -= i & -i)
            for (int j = y; j > 0; j -= j & -j)
                s += t[i][j];
        return s;
    }
    ll rect(int x1, int y1, int x2, int y2) const {
        return sum(x2, y2) - sum(x1 - 1, y2) - sum(x2, y1 - 1) + sum(x1 - 1, y1 - 1);
    }
};
// 内存 n*m*8 字节；2e3 x 2e3 约 32MB，够用
```

**二维区间改区间查**：需要 4 个 BIT，代码长且常数大，一般改用**二维线段树 / 树套树 / CDQ**，或者题目数据小直接差分。比赛里 90% 的二维题是"单点改 + 子矩阵和"。

## 5. 树状数组维护前缀最值（只能单调更新）

**限制**：只能"变大"（求 max）或"变小"（求 min），**不支持任意修改**。

```cpp
#include <bits/stdc++.h>
using namespace std;

struct BITMax {
    int n;
    vector<int> t;
    explicit BITMax(int n_) : n(n_), t(n_ + 1, INT_MIN) {}

    void update(int i, int v) {           // 把位置 i 的值更新为 max(t[i], v)，不支持变小
        for (; i <= n; i += i & -i) t[i] = max(t[i], v);
    }
    int query(int i) const {              // [1, i] 最大值
        int r = INT_MIN;
        for (; i > 0; i -= i & -i) r = max(r, t[i]);
        return r;
    }
};
```

> 需要支持任意修改 + 区间最值时用**线段树**。树状数组最值只能做"只增不减"的离线问题（如 LIS 的 `O(n log n)` 优化）。

## 6. 树状数组上二分（求第 k 小 / kth）

**前提**：BIT 存的是频次（非负），且需要"前缀和单调"。

```cpp
#include <bits/stdc++.h>
using namespace std;

struct BITKth {
    int n, LOG;
    vector<int> t;
    explicit BITKth(int n_) : n(n_), LOG(0), t(n_ + 1, 0) {
        while ((1 << LOG) <= n) ++LOG;
    }
    void add(int i, int v) { for (; i <= n; i += i & -i) t[i] += v; }

    // 返回最小的 pos 使得前缀和 >= k（k >= 1）；不存在返回 n+1
    int kth(int k) const {
        int pos = 0, acc = 0;
        for (int pw = 1 << (LOG - 1); pw; pw >>= 1) {
            if (pos + pw <= n && acc + t[pos + pw] < k) {
                pos += pw;
                acc += t[pos];
            }
        }
        return pos + 1;
    }
};
// 动态第 k 小：值域离散化后，加点 add(rank,1)，删点 add(rank,-1)，查询 kth(k)
```

**经典应用**：动态中位数（对顶堆/BIT 二分）、约瑟夫问题、区间第 k 小（配合主席树）。

## 7. 逆序对（离散化 + BIT）

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

ll count_inversions(vector<int> a) {      // 返回逆序对数
    int n = (int)a.size();
    vector<int> v = a;
    sort(v.begin(), v.end());
    v.erase(unique(v.begin(), v.end()), v.end());
    int m = (int)v.size();
    vector<int> t(m + 1, 0);
    auto add = [&](int i, int d) { for (; i <= m; i += i & -i) t[i] += d; };
    auto sum = [&](int i) { ll s = 0; for (; i > 0; i -= i & -i) s += t[i]; return s; };
    ll ans = 0;
    for (int i = n - 1; i >= 0; --i) {
        int r = int(lower_bound(v.begin(), v.end(), a[i]) - v.begin()) + 1;
        ans += sum(r - 1);               // 统计右侧比 a[i] 小的个数（严格逆序）
        add(r, 1);
    }
    return ans;
}
```

**变体**：
- 统计 `a[i] > a[j] (i<j)`：从右往左，查 `<= a[i]-1` 的数量（上面就是）。
- 统计 `a[i] >= a[j] (i<j)`：从左往右，查 `>= a[i]` 的数量。
- **三维偏序**：BIT + CDQ（见 14 偏序文件）。

## 8. 树状数组 vs 线段树 选择建议

| 需求 | 推荐 |
|---|---|
| 单点改 + 前缀和/区间和 | **BIT**（常数小、好写） |
| 区间加 + 单点查 | **BIT 差分** |
| 区间加 + 区间和 | BIT 双数组（上面第 3 节） |
| 区间最值、区间赋值、区间历史最值 | 线段树 |
| 动态第 k 小、逆序对 | BIT + 二分 |
| 二维单点改 + 子矩阵和 | 二维 BIT |
| 需要"合并"任意两个区间（非可减） | 线段树 / ST 表 |
| 需要懒标记 | 线段树 |

**记忆法**：BIT 只能做**可减**的运算（和、异或、计数）；遇到 max/min/赋值/懒标记，直接上线段树。

## 9. 树状数组套权值树（简单二维偏序 / 动态第 k 小）

当值域大且带修改时，用 BIT 的每个节点挂一棵**权值线段树**（或 `vector` + `sort` 后离线）。完整代码见 [22-高级数据结构](22-高级数据结构-主席树与树套树.md) 的"树套树"部分。

```cpp
#include <bits/stdc++.h>
using namespace std;
// 离线二维数点：BIT + 排序
// 问题：m 个询问 (x1,x2,y1,y2)，求矩形内点数。
// 做法：把询问拆成 4 个前缀查询 (x, y)，按 x 排序，扫描加点（按 y 插入 BIT），BIT 查前缀和。
struct Query { int x, y, id, sign; };
vector<long long> offline_2d_points(vector<pair<int,int>> pts, vector<array<int,4>> rects) {
    vector<Query> qs;
    for (int i = 0; i < (int)rects.size(); ++i) {
        auto [x1, y1, x2, y2] = rects[i];      // 闭区间
        qs.push_back({x2, y2, i, 1});
        qs.push_back({x1 - 1, y2, i, -1});
        qs.push_back({x2, y1 - 1, i, -1});
        qs.push_back({x1 - 1, y1 - 1, i, 1});
    }
    vector<int> ys;
    for (auto &p : pts) ys.push_back(p.second);
    for (auto &q : qs) ys.push_back(q.y);
    sort(ys.begin(), ys.end());
    ys.erase(unique(ys.begin(), ys.end()), ys.end());
    auto yid = [&](int y) {
        return int(lower_bound(ys.begin(), ys.end(), y) - ys.begin()) + 1;
    };

    sort(pts.begin(), pts.end());
    sort(qs.begin(), qs.end(), [](const Query &a, const Query &b) { return a.x < b.x; });
    vector<int> bit(ys.size() + 2, 0);
    auto add = [&](int i) { for (; i <= (int)ys.size(); i += i & -i) ++bit[i]; };
    auto sum = [&](int i) { int s = 0; for (; i > 0; i -= i & -i) s += bit[i]; return s; };

    vector<long long> ans(rects.size(), 0);
    size_t pi = 0;
    for (auto &q : qs) {
        while (pi < pts.size() && pts[pi].first <= q.x) { add(yid(pts[pi].second)); ++pi; }
        ans[q.id] += 1LL * q.sign * sum(yid(q.y));
    }
    return ans;
}
```

## 10. 小结：树状数组的 5 个必背写法

1. `add(i,v)` / `sum(i)`：`i += i & -i` / `i -= i & -i`
2. 差分：区间加 → `add(l,v), add(r+1,-v)`，单点查 → `sum(i)`
3. 区间和双 BIT：`S(x) = (x+1)*ΣB1 - ΣB2`
4. `kth`：从 `1 << LOG` 往下试，`acc + t[pos+pw] < k` 就跳
5. 离散化 + 逆序对：从右往左扫

## 11. 思路与例题索引

| 算法 | 思路一句话 | 用途 | 例题 |
|---|---|---|---|
| 单点改 + 前缀和 | `i += i&-i` 上跳维护，下跳查询 | 区间和 | P3374【模板】树状数组 1 |
| 差分 BIT | 区间加 → 差分数组单点改 | 区间加、单点查 | P3368【模板】树状数组 2 |
| 双 BIT 区间和 | `S(x) = (x+1)Σd - Σ(d·i)` | 区间加 + 区间和 | P4514 上帝造题的七分钟 |
| 二维 BIT | 两重 `i += i&-i, j += j&-j` | 子矩阵和 | P4514、P4054 |
| BIT 最值 | 只能"只增不减" | LIS 的 O(n log n) 优化 | P1020 导弹拦截 |
| BIT 上二分 `kth` | 从 `1<<LOG` 往下试，前缀和 < k 就跳 | 动态第 k 小、中位数 | P1168 中位数 |
| 离散化 + 逆序对 | 从右往左扫，查比它小的个数 | 逆序对 | P1908 逆序对 |

**选择**：能"减"的（和/异或/计数）用 BIT；max/min/赋值用线段树。
