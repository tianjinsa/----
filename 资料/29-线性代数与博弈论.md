# 第一部分：线性代数

## 1. 高斯消元（实数，解线性方程组）

**用途**：解 `n` 元一次方程组、求秩、求逆矩阵、解期望 DP 的环。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ld = long double;

// 解 A x = b，返回解的个数：0 无解，1 唯一解，INF 无穷多解
// 唯一解时 ans 存结果
constexpr int INF_SOL = -1;

int gauss(vector<vector<ld>> A, vector<ld> b, vector<ld> &ans) {
    int n = (int)A.size(), m = (int)A[0].size();
    vector<vector<ld>> mat(n, vector<ld>(m + 1));
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) mat[i][j] = A[i][j];
        mat[i][m] = b[i];
    }
    vector<int> where(m, -1);
    int row = 0;
    for (int col = 0; col < m && row < n; ++col) {
        int sel = row;
        for (int i = row; i < n; ++i)
            if (fabsl(mat[i][col]) > fabsl(mat[sel][col])) sel = i;
        if (fabsl(mat[sel][col]) < 1e-12L) continue;       // 该列全零
        swap(mat[sel], mat[row]);
        where[col] = row;
        for (int i = 0; i < n; ++i) if (i != row) {        // 消成单位阵
            ld f = mat[i][col] / mat[row][col];
            for (int j = col; j <= m; ++j) mat[i][j] -= f * mat[row][j];
        }
        ++row;
    }
    ans.assign(m, 0);
    for (int i = 0; i < m; ++i)
        if (where[i] != -1) ans[i] = mat[where[i]][m] / mat[where[i]][i];
    for (int i = 0; i < n; ++i) {                          // 检查矛盾
        ld s = 0;
        for (int j = 0; j < m; ++j) s += ans[j] * mat[i][j];
        if (fabsl(s - mat[i][m]) > 1e-8L) return 0;
    }
    for (int i = 0; i < m; ++i) if (where[i] == -1) return INF_SOL;
    return 1;
}
```

## 2. 高斯消元（异或方程组 / 模意义）

```cpp
#include <bits/stdc++.h>
using namespace std;

// 异或方程组：n 个方程 m 个变量，a[i][j] ∈ {0,1}，b[i] ∈ {0,1}
// 返回秩；无解返回 -1；ans 存一组解
int gauss_xor(vector<vector<int>> a, vector<int> b, vector<int> &ans) {
    int n = (int)a.size(), m = (int)a[0].size();
    vector<int> where(m, -1);
    int row = 0;
    for (int col = 0; col < m && row < n; ++col) {
        int sel = -1;
        for (int i = row; i < n; ++i) if (a[i][col]) { sel = i; break; }
        if (sel == -1) continue;
        swap(a[sel], a[row]); swap(b[sel], b[row]);
        where[col] = row;
        for (int i = 0; i < n; ++i) if (i != row && a[i][col]) {
            for (int j = col; j < m; ++j) a[i][j] ^= a[row][j];
            b[i] ^= b[row];
        }
        ++row;
    }
    ans.assign(m, 0);
    for (int i = 0; i < m; ++i) if (where[i] != -1) ans[i] = b[where[i]];
    for (int i = 0; i < n; ++i) {
        int s = 0;
        for (int j = 0; j < m; ++j) s ^= ans[j] & a[i][j];
        if (s != b[i]) return -1;
    }
    return row;                          // 秩（自由变量个数 = m - row）
}
```

**用途**：
- 开关灯问题（每个灯是一个异或方程）。
- 期望 DP 的环（列方程 + 高斯消元，模意义下）。
- 求线性空间维数。

## 3. 线性基 ★

**用途**：异或线性空间。
- 最大异或和、第 k 大异或和。
- 判断一个数能否被异或出来。
- **本场 A 题**的核心数据结构（`Basis::insert` 返回秩是否增加）。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

struct LinearBasis {
    static constexpr int BITS = 62;           // 值域 < 2^62
    ll b[BITS + 1];
    bool hasZero = true;

    LinearBasis() { memset(b, 0, sizeof b); }

    // 插入 x，返回 true 表示秩增加
    bool insert(ll x) {
        for (int i = BITS; i >= 0; --i) {
            if (!((x >> i) & 1)) continue;
            if (!b[i]) { b[i] = x; return true; }
            x ^= b[i];
        }
        return false;
    }
    // 能否被异或表示
    bool canRepresent(ll x) const {
        for (int i = BITS; i >= 0; --i) if ((x >> i) & 1) {
            if (!b[i]) return false;
            x ^= b[i];
        }
        return true;
    }
    // 最大异或和
    ll maxXor() const {
        ll res = 0;
        for (int i = BITS; i >= 0; --i) res = max(res, res ^ b[i]);
        return res;
    }
    // 最小异或和（非零）
    ll minXor() const {
        for (int i = 0; i <= BITS; ++i) if (b[i]) return b[i];
        return 0;
    }
    // 第 k 小异或和（1-based，含 0）
    ll kthXor(ll k) const {
        // 先化成"每个基只有一个最高位"的规范形式
        ll t[BITS + 1];
        memcpy(t, b, sizeof b);
        for (int i = 0; i <= BITS; ++i)
            for (int j = i - 1; j >= 0; --j)
                if ((t[i] >> j) & 1) t[i] ^= t[j];
        ll res = 0, cnt = 0;
        for (int i = 0; i <= BITS; ++i) if (t[i]) {
            if ((k >> cnt) & 1) res ^= t[i];
            ++cnt;
        }
        return res;
    }
    int rank() const { int r = 0; for (int i = 0; i <= BITS; ++i) if (b[i]) ++r; return r; }
};
```

**实例（A 题）**：`insert` 返回 `true` 时就说明"公共空间的秩增加"，把该向量加入 `events` 传播给各局部集合。

## 4. 矩阵树定理（生成树计数）

**用途**：**无向图生成树个数** = 拉普拉斯矩阵任意主子式的行列式。
**有向图**：内向树/外向树用不同的拉普拉斯矩阵。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

// 模 p 逆元（p 为质数）
ll mod_inv(ll a, ll p) {
    ll r = 1, b = a % p, e = p - 2;
    while (e) { if (e & 1) r = r * b % p; b = b * b % p; e >>= 1; }
    return r;
}

// 行列式 mod p（p 为质数），O(n^3)
ll determinant_mod(vector<vector<ll>> a, ll p) {
    int n = (int)a.size();
    ll res = 1;
    for (int i = 0; i < n; ++i) {
        int sel = -1;
        for (int j = i; j < n; ++j) if (a[j][i] % p) { sel = j; break; }
        if (sel == -1) return 0;
        if (sel != i) { swap(a[sel], a[i]); res = (p - res) % p; }
        for (int j = i + 1; j < n; ++j) {
            if (!a[j][i]) continue;
            ll f = (ll)a[j][i] * mod_inv(a[i][i], p) % p;
            for (int k = i; k < n; ++k) a[j][k] = ((a[j][k] - f * a[i][k]) % p + p) % p;
        }
        res = res * a[i][i] % p;
    }
    return (res % p + p) % p;
}

// 无向图生成树计数：n 个点，m 条边，返回生成树个数 mod p
ll count_spanning_trees(int n, const vector<pair<int,int>> &edges, ll p) {
    vector<vector<ll>> lap(n, vector<ll>(n, 0));
    for (auto [u, v] : edges) {
        if (u == v) continue;
        --u; --v;
        lap[u][u] = (lap[u][u] + 1) % p;
        lap[v][v] = (lap[v][v] + 1) % p;
        lap[u][v] = (lap[u][v] - 1 + p) % p;
        lap[v][u] = (lap[v][u] - 1 + p) % p;
    }
    // 去掉最后一行一列
    vector<vector<ll>> minor(n - 1, vector<ll>(n - 1));
    for (int i = 0; i < n - 1; ++i)
        for (int j = 0; j < n - 1; ++j) minor[i][j] = lap[i][j];
    return determinant_mod(minor, p);
}
```

**实例**：P4111 [HEOI2015] 小 Z 的房间、P4455 [CQOI2018] 社交网络（有向）。

# 第二部分：博弈论

## 5. Nim 游戏

**结论**：`n` 堆石子，每次从一堆取任意个（至少 1 个），取完者胜。
**先手必胜 ⟺ 异或和 `≠ 0`**。

```cpp
#include <bits/stdc++.h>
using namespace std;

// 返回 true 表示先手必胜
bool nim_win(const vector<int> &piles) {
    int x = 0;
    for (int v : piles) x ^= v;
    return x != 0;
}
```

**反 Nim（Anti-Nim）**：取完者**输**。

```
若所有堆都是 1：先手必胜 ⟺ 堆数为偶数
否则：先手必胜 ⟺ 异或和 != 0
```

```cpp
#include <bits/stdc++.h>
using namespace std;

bool anti_nim_win(const vector<int> &piles) {
    int x = 0, cnt1 = 0;
    for (int v : piles) { x ^= v; if (v == 1) ++cnt1; }
    if (cnt1 == (int)piles.size()) return (cnt1 % 2 == 0);
    return x != 0;
}
```

## 6. SG 函数（有向图游戏）

**用途**：把"一个游戏"拆成"若干子游戏"，整体 SG = 各子游戏 SG 的异或。
**定理**：`SG(u) = mex{ SG(v) : u -> v }`；先手必胜 ⟺ `SG(起点) != 0`。

```cpp
#include <bits/stdc++.h>
using namespace std;

// DAG 上的 SG 函数（暴力 mex）
struct SG {
    int n;
    vector<vector<int>> g;
    vector<int> sg;
    explicit SG(int n_) : n(n_), g(n_ + 1), sg(n_ + 1, -1) {}
    void addEdge(int u, int v) { g[u].push_back(v); }

    int dfs(int u) {
        if (sg[u] != -1) return sg[u];
        vector<bool> seen(g[u].size() + 2, false);
        for (int v : g[u]) {
            int s = dfs(v);
            if (s < (int)seen.size()) seen[s] = true;
        }
        int mex = 0;
        while (mex < (int)seen.size() && seen[mex]) ++mex;
        return sg[u] = mex;
    }
    int run(int start) { return dfs(start); }
};
```

**常见子游戏的 SG 公式**：

| 游戏 | SG |
|---|---|
| 一堆石子可取 `1..k` 个 | `n % (k+1)` |
| 一堆石子可取 `1..n`（任意） | `n` |
| 阶梯 Nim（第 i 级取，往下落） | 奇数级异或和 |
| 分割游戏（切成两半） | 见具体题 |
| 取石子（取完后不能取者输） | 见 anti-Nim |

**实例（P2197 取石子游戏）**：Nim 模板。
**实例（P2148 [SDOI2009] E&D）**：SG 打表找规律。
**实例（P2575 高手过招）**：一行棋子的 SG。

## 8. 威佐夫博弈（Wythoff）

**结论**：两堆 `(a, b)`，可从一堆取任意个或从两堆同时取相同个。**先手必败 ⟺ `a = ⌊k·φ⌋`，`b = ⌊k·φ²⌋ = a + k`**。

```cpp
#include <bits/stdc++.h>
using namespace std;

// 返回 true 表示先手必败（奇异局势）
bool wythoff_losing(int a, int b) {
    if (a > b) swap(a, b);
    double phi = (1 + sqrt(5.0)) / 2;
    int k = (int)((b - a) * phi);
    return a == k;
}
```

## 9. 二分图博弈

**结论**：在二分图上，棋子从起点出发，两人轮流沿边移动且不能重复经过点，不能移动者输。
**先手必胜 ⟺ 起点属于所有最大匹配**（即存在一个最大匹配覆盖起点）。

## 12. 博弈题解题流程

```
1. 是"多个独立子游戏"吗？
   是 -> 求每个子游戏的 SG，异或
   否 -> 直接分析奇异局势

2. 子游戏的 SG 能算吗？
   - 规则简单 -> 打表找规律
   - 是 Nim 变形 -> 套结论（阶梯/anti/巴什）
   - 是图上游戏 -> DFS 求 mex

3. 打表找规律的方法：
   - 写暴力 SG，输出前 30 项
   - 观察周期性
   - 猜公式，用暴力对拍验证
```

**打表 SG 的模板**：

```cpp
#include <bits/stdc++.h>
using namespace std;

// 例：一堆石子，每次可取 1,3,4 个，打表求 SG
int main() {
    const int N = 50;
    vector<int> sg(N + 1, 0);
    vector<bool> seen(N + 2);
    int moves[3] = {1, 3, 4};
    for (int i = 1; i <= N; ++i) {
        fill(seen.begin(), seen.end(), false);
        for (int m : moves) if (i >= m) seen[sg[i - m]] = true;
        int mex = 0;
        while (seen[mex]) ++mex;
        sg[i] = mex;
        printf("sg[%d]=%d\n", i, sg[i]);
    }
    return 0;
}
```

        scanf("%lld", &x);
        lb.insert(x);
    }
    ll mx = lb.maxXor();
    // ========================================================

    // ===================== 【输出】改这里 =====================
    printf("%lld\n", mx);
    // ========================================================
