# 17 搜索与剪枝 / 启发式算法

> **用途**：当正解是"指数级"时（`n <= 20`、状态数可控、答案是构造性），用搜索 + 剪枝。
> **判据**：`n <= 20` 想状压/爆搜；`n <= 40` 想 meet-in-the-middle；答案要求最优但状态爆炸 → 启发式（A*/IDA*）、模拟退火。

---

## 1. DFS + 剪枝（最常用）

**实例 1：P1040 加分二叉树 / P1025 数的划分** —— 组合搜索 + 可行性/最优性剪枝。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

// 实例：把 n 划分成 k 个正整数之和（不计顺序），求方案数
// 关键剪枝：下一项 >= 上一项（避免重复），且剩余可分配
ll n, k;
ll ans = 0;
void dfs(ll last, ll rem, ll parts) {     // 上一项、剩余和、剩余个数
    if (parts == 0) { if (rem == 0) ++ans; return; }
    if (rem < last * parts) return;                   // 剪枝：最小也放不下
    if (rem > (rem / parts + 1) * k) { /* 可选：更强的界 */ }
    for (ll x = last; x * parts <= rem; ++x) dfs(x, rem - x, parts - 1);
}
```

**实例 2：P1120 小木棍** —— 经典的"排序 + 多重剪枝"题。

```cpp
// 剪枝 4 板斧（这题必背）：
// 1. 从大到小排序，优先放大的（搜索树更小）
// 2. 若当前木棍放不进去，相同长度的木棍直接跳过
// 3. 若当前是某根新木棍的第一段且失败了，直接 return（换更小的也没用）
// 4. 若当前木棍恰好填满一根且失败了，直接 return
```

```cpp
#include <bits/stdc++.h>
using namespace std;

int n, a[70], target, cnt;
int used[70];

bool dfs(int idx, int cur, int done) {     // 当前下标、当前长度、已完成根数
    if (done == cnt) return true;
    if (cur == target) return dfs(0, 0, done + 1);
    int last = -1;
    for (int i = idx; i < n; ++i) {
        if (used[i] || a[i] == last) continue;          // 剪枝 2：同长度跳过
        if (cur + a[i] > target) continue;
        used[i] = 1;
        if (dfs(i + 1, cur + a[i], done)) return true;
        used[i] = 0;
        last = a[i];
        if (cur == 0 || cur + a[i] == target) break;    // 剪枝 3、4
    }
    return false;
}
```

**实例 3：P1379 八数码** —— 状态搜索 + 哈希去重（或双向 BFS / IDA*）。

## 2. 双向 BFS

**用途**：状态空间大但**起点和终点都明确**，且每步可逆。复杂度从 `b^d` 降到 `b^(d/2)`。

```cpp
#include <bits/stdc++.h>
using namespace std;

// 双向 BFS 求最短步数（一般形式：字符串/编码状态）
// dist1: 从起点搜；dist2: 从终点搜；每次扩展层数小的一侧
struct BiBFS {
    // 由使用者提供：起始状态、终止状态、邻居生成函数、状态哈希
    // 这里给出骨架，配合 unordered_map<string,int> 使用
    static int bfs(const string &s, const string &t,
                   const function<vector<string>(const string&)> &nxt) {
        if (s == t) return 0;
        unordered_map<string,int> d1, d2;
        queue<string> q1, q2;
        d1[s] = 0; q1.push(s);
        d2[t] = 0; q2.push(t);
        while (!q1.empty() && !q2.empty()) {
            // 扩展较小的一侧
            bool fromStart = q1.size() <= q2.size();
            auto &q = fromStart ? q1 : q2;
            auto &dcur = fromStart ? d1 : d2;
            auto &dother = fromStart ? d2 : d1;
            int layer = (int)q.size();
            for (int rep = 0; rep < layer; ++rep) {
                string cur = q.front(); q.pop();
                int dc = dcur[cur];
                for (const string &v : nxt(cur)) {
                    if (dcur.count(v)) continue;
                    if (dother.count(v)) return dc + 1 + dother[v];
                    dcur[v] = dc + 1;
                    q.push(v);
                }
            }
        }
        return -1;
    }
};
```

**实例：P1379 八数码** —— 用 `string` 表示 3x3，BFS 邻居是空格上下左右交换。双向 BFS 可过 `9!` 状态。

## 3. IDA*（迭代加深 + 估价函数）

**用途**：深搜层数不确定、但答案深度不大；内存受限（比 A* 省内存）。

```cpp
#include <bits/stdc++.h>
using namespace std;

// IDA* 骨架：f = g + h，h 必须是"不超过真实剩余步数"的估计（可采纳）
// 若 f > 当前限制，剪枝；若到达目标返回
struct IDAStar {
    int limit;                              // 当前深度限制
    bool found;

    // 由使用者提供：状态的 h 函数、目标判定、邻居扩展
    // 下面以"八数码"的曼哈顿距离估价为例
    static int manhattan(const string &s) {
        int h = 0;
        for (int i = 0; i < 9; ++i) {
            if (s[i] == '0') continue;
            int v = s[i] - '1';
            h += abs(i / 3 - v / 3) + abs(i % 3 - v % 3);
        }
        return h;
    }
};
```

**实例：八数码的 IDA***

```cpp
#include <bits/stdc++.h>
using namespace std;

string tar = "123456780";
int dx[4] = {-1, 1, 0, 0}, dy[4] = {0, 0, -1, 1};
bool ok = false;

int h(const string &s) {
    int r = 0;
    for (int i = 0; i < 9; ++i) {
        if (s[i] == '0') continue;
        int v = s[i] - '1';
        r += abs(i / 3 - v / 3) + abs(i % 3 - v % 3);
    }
    return r;
}

bool dfs(string s, int g, int limit, int pre) {
    int f = g + h(s);
    if (f > limit) return false;
    if (s == tar) return true;
    int p = (int)s.find('0');
    int x = p / 3, y = p % 3;
    for (int d = 0; d < 4; ++d) {
        int nx = x + dx[d], ny = y + dy[d];
        if (nx < 0 || nx > 2 || ny < 0 || ny > 2) continue;
        if (pre != -1 && d == (pre ^ 1)) continue;          // 不走回头路
        int np = nx * 3 + ny;
        swap(s[p], s[np]);
        if (dfs(s, g + 1, limit, d)) return true;
        swap(s[p], s[np]);
    }
    return false;
}
```

**A* vs IDA***

| | A* | IDA* |
|---|---|---|
| 内存 | 大（要存所有访问状态） | 小（只有递归栈） |
| 适用 | 状态可达、要最短 | 内存受限、深度浅 |
| 实现 | 优先队列 + `dist` | 迭代加深 + `h` 剪枝 |

## 4. Meet in the Middle

**用途**：`n <= 40` 的子集和/计数问题。把 `n` 分成两半，各枚举 `2^(n/2)`，再合并。

**实例：P4799 世界冰球锦标赛** —— 求花费 `<= m` 的方案数。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
    int n; ll m;
    scanf("%d %lld", &n, &m);
    vector<ll> a(n);
    for (auto &x : a) scanf("%lld", &x);
    int n1 = n / 2, n2 = n - n1;
    vector<ll> s1, s2;
    auto gen = [&](int l, int r, vector<ll> &out) {
        out.push_back(0);
        for (int i = l; i < r; ++i) {
            int sz = (int)out.size();
            for (int j = 0; j < sz; ++j)
                if (out[j] + a[i] <= m) out.push_back(out[j] + a[i]);
        }
    };
    gen(0, n1, s1);
    gen(n1, n, s2);
    sort(s2.begin(), s2.end());
    ll ans = 0;
    for (ll x : s1) ans += upper_bound(s2.begin(), s2.end(), m - x) - s2.begin();
    printf("%lld\n", ans);
    return 0;
}
```

## 5. 模拟退火 / 爬山（随机化）

**用途**：求函数极值、TSP、最小圆覆盖等**连续或离散**优化问题；当正解太难时骗分。

```cpp
#include <bits/stdc++.h>
using namespace std;

// 爬山（单峰/局部搜索）：每次尝试邻居，变好就接受
template <class State, class F, class Neighbor>
State hill_climb(State s, F value, Neighbor neighbor, int iterations = 100000) {
    double best = value(s);
    for (int it = 0; it < iterations; ++it) {
        State t = neighbor(s, it);
        double v = value(t);
        if (v > best) { best = v; s = t; }
    }
    return s;
}

// 模拟退火：以概率 exp((new-old)/T) 接受变差的解
template <class State, class F, class Neighbor>
State simulated_annealing(State s, F value, Neighbor neighbor,
                          double T0 = 1e4, double T_end = 1e-6, double cool = 0.999) {
    mt19937 rng(chrono::steady_clock::now().time_since_epoch().count());
    uniform_real_distribution<double> U(0.0, 1.0);
    double cur = value(s), best = cur;
    State bs = s;
    for (double T = T0; T > T_end; T *= cool) {
        State t = neighbor(s);
        double v = value(t);
        if (v > cur || U(rng) < exp((v - cur) / T)) {
            s = t; cur = v;
            if (v > best) { best = v; bs = t; }
        }
    }
    return bs;
}
```

**实例：TSP（旅行商）**

```cpp
#include <bits/stdc++.h>
using namespace std;

// TSP 模拟退火：state 是访问顺序（permutation），value 是总路程，邻接用"交换两个点"
// 注意：用 double 存坐标，dist 用 hypot
// 关键参数：T0 = 1e5, cool = 0.9999, 迭代 1e6
```

**实例：P1337 吊打XXX（平衡点）** —— 模拟退火求势能最小点。


## 6. 剪枝技巧清单

| 剪枝 | 说明 |
|---|---|
| **可行性剪枝** | 当前状态已经违反约束（超额、越界）就返回 |
| **最优性剪枝** | 当前代价 + 下界估计 `>=` 已有最优解就返回 |
| **记忆化** | 状态哈希后存结果（注意状态空间大小） |
| **排序** | 把大的、分支少的放前面，尽早失败 |
| **对称性剪枝** | 相同元素只试第一个 |
| **边界剪枝** | 剩余元素和不够/太多就返回 |
| **IDA\* 估价** | `f = g + h`，`h` 可采纳 |
| **双向搜索** | 从头尾同时搜 |
| **随机打乱** | 避免被构造数据卡（比赛常见） |

**实例：P1034 矩形覆盖 / P2324 骑士精神（IDA*）**

## 7. Dancing Links（DLX，精确覆盖）

**用途**：数独、N 皇后、精确覆盖/重复覆盖问题。代码较长，**只有题目明确是精确覆盖才用**。

```cpp
#include <bits/stdc++.h>
using namespace std;

// DLX 精确覆盖：n 行 m 列，选若干行使每列恰好被覆盖一次
struct DLX {
    static const int MAXN = 10005;
    int n, m, sz;
    int U[MAXN], D[MAXN], L[MAXN], R[MAXN], row[MAXN], col[MAXN];
    vector<int> H;                       // 每行头
    vector<int> S;                       // 每列节点数
    vector<int> ans;

    void init(int rows, int cols) {
        n = rows; m = cols;
        for (int i = 0; i <= m; ++i) {
            S.push_back(0);
            U[i] = D[i] = i;
            L[i] = i - 1; R[i] = i + 1;
        }
        L[0] = m; R[m] = 0;
        H.assign(n + 1, -1);
        sz = m + 1;
    }
    void insert(int r, int c) {
        ++S[c];
        row[sz] = r; col[sz] = c;
        U[sz] = U[c]; D[sz] = c;
        D[U[c]] = sz; U[c] = sz;
        if (H[r] == -1) { H[r] = sz; L[sz] = R[sz] = sz; }
        else {
            L[sz] = L[H[r]]; R[sz] = H[r];
            R[L[H[r]]] = sz; L[H[r]] = sz;
        }
        ++sz;
    }
    void remove(int c) {
        R[L[c]] = R[c]; L[R[c]] = L[c];
        for (int i = D[c]; i != c; i = D[i])
            for (int j = R[i]; j != i; j = R[j]) {
                U[D[j]] = U[j]; D[U[j]] = D[j]; --S[col[j]];
            }
    }
    void restore(int c) {
        for (int i = U[c]; i != c; i = U[i])
            for (int j = L[i]; j != i; j = L[j]) {
                U[D[j]] = j; D[U[j]] = j; ++S[col[j]];
            }
        R[L[c]] = c; L[R[c]] = c;
    }
    bool dance(int dep) {
        if (R[0] == 0) return true;
        int c = R[0];
        for (int i = R[0]; i != 0; i = R[i]) if (S[i] < S[c]) c = i;
        remove(c);
        for (int i = D[c]; i != c; i = D[i]) {
            ans.push_back(row[i]);
            for (int j = R[i]; j != i; j = R[j]) remove(col[j]);
            if (dance(dep + 1)) return true;
            for (int j = L[i]; j != i; j = L[j]) restore(col[j]);
            ans.pop_back();
        }
        restore(c);
        return false;
    }
};
```

> **建议**：DLX 只在"精确覆盖"模板题里用（P4929）。普通搜索题用 DFS + 剪枝更快写。

## 8. 搜索题的判断流程

```
1. n 多大？
   n <= 20  → 状压 DP / DFS+剪枝 / 枚举子集
   n <= 40  → meet in the middle
   n 更大   → 一定有贪心/DP 正解，别搜索

2. 要最短步数？
   → BFS（状态可哈希）；状态爆炸 → 双向 BFS / IDA*

3. 要最优解？
   → A* / IDA*（有可采纳的 h）/ 模拟退火（放弃精确）

4. 要方案数？
   → 记忆化搜索（状态可压缩）

5. 有没有特殊结构？
   → 精确覆盖 → DLX；数独 → DLX/DFS+位运算
```


### 怎么用（照抄这段 `main`，只改【输入】【输出】两处）

上面是模板本体。实际写题时：**把上面的 struct 整段粘过来 → 再粘这段 `main` → 只改标了【输入】【输出】的行**。

```cpp skip
#include <bits/stdc++.h>
using namespace std;

// ===== 粘贴点：把上面的模拟退火 / DLX 粘到这里 =====

int main() {
    // ===================== 【输入】改这里 =====================
    int n;
    scanf("%d", &n);                      // 点数（以 TSP 为例）
    vector<double> xs(n), ys(n);
    for (int i = 0; i < n; ++i) scanf("%lf %lf", &xs[i], &ys[i]);
    // ========================================================

    // ===================== 【核心】算法流程 =====================
    // 模拟退火（求近似最优）：
    //   1) 随机给一个解 s，算它的 value(s)
    //   2) 每次随机扰动得到邻居 t
    //   3) 如果更好直接接受；更差则以 exp((v_new - v_old) / T) 的概率接受
    //   4) 温度 T 每轮乘 0.999 下降，T 很小时基本不再接受变差解
    //   5) 全程记录见过的最好解
    vector<int> perm(n);
    iota(perm.begin(), perm.end(), 0);
    auto value = [&](const vector<int> &p) {
        double d = 0;
        for (int i = 0; i + 1 < n; ++i) {
            double dx = xs[p[i]] - xs[p[i + 1]];
            double dy = ys[p[i]] - ys[p[i + 1]];
            d += sqrt(dx * dx + dy * dy);
        }
        return d;
    };
    auto neighbor = [&](const vector<int> &p) {
        vector<int> q = p;
        int i = rand() % n, j = rand() % n;
        swap(q[i], q[j]);
        return q;
    };
    auto best = simulated_annealing(perm, value, neighbor);
    // ========================================================

    // ===================== 【输出】改这里 =====================
    printf("%.6f\n", value(best));
    // ========================================================
    return 0;
}
```

## 9. 思路与例题索引

| 算法 | 思路一句话 | 用途 | 例题 |
|---|---|---|---|
| DFS + 剪枝 | 可行性/最优性/对称性剪枝 | 组合搜索 | P1120 小木棍、P1025 数的划分 |
| 双向 BFS | 从起终点同时搜，相遇即最短路 | 状态空间大、有明确终态 | P1379 八数码难题 |
| IDA* | 迭代加深 + 可采纳启发函数 | 内存受限的最优搜索 | P2324 骑士精神、P1379 |
| A* | 优先队列 + `f = g + h` | 最短路/搜索 | k 短路（见 31） |
| Meet in the Middle | 拆两半各枚举再合并 | `n <= 40` 子集和 | P4799 世界冰球锦标赛 |
| 模拟退火 | 概率接受变差解，温度下降 | 求极值、TSP | P1337 吊打XXX、P2503 均分数据 |
| 爬山 | 只接受变好的解 | 多峰但局部可改进 | P3878 分金币 |
| DLX | 十字链表精确覆盖 | 数独、N 皇后 | P4929【模板】舞蹈链 |
| 随机化贪心 | 多次随机打乱求最优 | 构造、近似 | P2503、CF 系列构造题 |

**判据**：`n <= 20` 状压/爆搜；`n <= 40` meet-in-middle；有明确终态且要最短 → 双向 BFS/IDA*。
