## 2. 多项式求逆

**用途**：给定 `A(x)`，求 `B(x)` 使 `A*B ≡ 1 (mod x^n)`。
**思路**：牛顿迭代 `B = B * (2 - A*B) (mod x^n)`，每步长度翻倍。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

namespace Poly {
    constexpr int MOD = 998244353, G = 3;
    ll qpow(ll a, ll b) {
        ll r = 1; a %= MOD;
        while (b) { if (b & 1) r = r * a % MOD; a = a * a % MOD; b >>= 1; }
        return r;
    }

    // ---- 自包含 NTT ----
    void ntt(vector<int> &a, bool invert) {
        int n = (int)a.size();
        for (int i = 1, j = 0; i < n; ++i) {
            int bit = n >> 1;
            for (; j & bit; bit >>= 1) j ^= bit;
            j ^= bit;
            if (i < j) swap(a[i], a[j]);
        }
        for (int len = 2; len <= n; len <<= 1) {
            ll w = qpow(G, (MOD - 1) / len);
            if (invert) w = qpow(w, MOD - 2);
            int half = len >> 1;
            for (int i = 0; i < n; i += len) {
                ll wn = 1;
                for (int j = 0; j < half; ++j) {
                    int u = a[i + j];
                    int v = (int)(wn * a[i + j + half] % MOD);
                    int x = u + v; if (x >= MOD) x -= MOD;
                    int y = u - v; if (y < 0) y += MOD;
                    a[i + j] = x; a[i + j + half] = y;
                    wn = wn * w % MOD;
                }
            }
        }
        if (invert) {
            ll inv_n = qpow(n, MOD - 2);
            for (int &x : a) x = (int)(x * inv_n % MOD);
        }
    }

    vector<int> multiply(vector<int> a, vector<int> b) {
        if (a.empty() || b.empty()) return {};
        int need = (int)a.size() + (int)b.size() - 1;
        int n = 1; while (n < need) n <<= 1;
        a.resize(n); b.resize(n);
        ntt(a, false); ntt(b, false);
        for (int i = 0; i < n; ++i) a[i] = (ll)a[i] * b[i] % MOD;
        ntt(a, true);
        a.resize(need);
        return a;
    }

    // 多项式求逆，结果长度 n
    vector<int> inverse(const vector<int> &A, int n) {
        vector<int> B(1, (int)qpow(A[0], MOD - 2));
        int len = 1;
        while (len < n) {
            len <<= 1;
            vector<int> a(A.begin(), A.begin() + min((int)A.size(), len));
            vector<int> t = multiply(a, B);          // A*B
            for (int &x : t) x = (MOD - x) % MOD;
            t[0] = (t[0] + 2) % MOD;                 // 2 - A*B
            B = multiply(B, t);
            B.resize(len);
        }
        B.resize(n);
        return B;
    }

    // 求导 / 积分
    vector<int> derivate(const vector<int> &a) {
        int n = (int)a.size();
        vector<int> r(max(1, n - 1));
        for (int i = 1; i < n; ++i) r[i - 1] = (ll)a[i] * i % MOD;
        return r;
    }
    vector<int> integrate(const vector<int> &a) {
        int n = (int)a.size();
        vector<int> r(n + 1), inv(n + 2, 1);
        for (int i = 1; i <= n + 1; ++i) inv[i] = (ll)(MOD - MOD / i) * inv[MOD % i] % MOD;
        for (int i = 0; i < n; ++i) r[i + 1] = (ll)a[i] * inv[i + 1] % MOD;
        return r;
    }

    // 多项式 ln：要求 A[0] = 1，结果长度 n
    vector<int> ln(const vector<int> &A, int n) {
        vector<int> d = derivate(A);                 // A'
        vector<int> invA = inverse(A, n);            // 1/A
        vector<int> t = multiply(d, invA);            // A'/A
        t.resize(n - 1);
        vector<int> res = integrate(t);              // ∫ A'/A
        res.resize(n);
        return res;
    }
}
```

> 上面已经把 `ntt` / `multiply` **写在 `Poly` 命名空间内部**，所以 2~6 节的所有函数都在 `namespace Poly` 里，直接调用 `Poly::inverse(...)` 即可。
> 若你更想用 04 号文件里更快的 `PolyNTT`，把 `Poly::ntt` / `Poly::multiply` 换成它的实例方法即可（接口一致）。

## 3. 多项式 exp

**用途**：`B = exp(A)`，即 `B' = A' * B`，要求 `A[0] = 0`。

```cpp skip
// 片段：写在 namespace Poly 内部（依赖上面的 ln / multiply），单独编译不通过
// 多项式 exp（牛顿迭代），结果长度 n
vector<int> exp(const vector<int> &A, int n) {
    vector<int> B(1, 1);
    int len = 1;
    while (len < n) {
        len <<= 1;
        vector<int> lnB = ln(B, len);                    // ln(B)
        vector<int> diff(len);
        for (int i = 0; i < len; ++i) {
            int a = (i < (int)A.size()) ? A[i] : 0;
            diff[i] = ((a - lnB[i]) % MOD + MOD) % MOD;
        }
        diff[0] = (diff[0] + 1) % MOD;                    // 1 + A - ln(B)
        B = multiply(B, diff);
        B.resize(len);
    }
    B.resize(n);
    return B;
}
```

**用途**：`exp` 是"多项式的指数函数"，配合 `ln` 可以算 `pow`（见下）。

## 4. 多项式开根

```cpp skip
// 多项式开根（B*B = A），要求 A[0] 是二次剩余，结果长度 n
vector<int> sqrt_poly(const vector<int> &A, int n) {
    vector<int> B(1, 1);                                  // 若 A[0] 不是 1，改成 sqrt(A[0])
    int len = 1;
    while (len < n) {
        len <<= 1;
        vector<int> invB = inverse(B, len);
        vector<int> t(A.begin(), A.begin() + min((int)A.size(), len));
        t.resize(len);
        t = multiply(t, invB);                            // A/B
        B.resize(len);
        for (int i = 0; i < len; ++i) B[i] = (ll)(B[i] + t[i]) * ((MOD + 1) / 2) % MOD;
    }
    B.resize(n);
    return B;
}
```

## 5. 多项式除法与取模

**用途**：`A = Q*B + R`，`deg R < deg B`。
**技巧**：反转系数，把除法变成乘法（用求逆）。

```cpp skip
// A / B（B 非零），返回商 Q
vector<int> poly_div(const vector<int> &A, const vector<int> &B) {
    int n = (int)A.size(), m = (int)B.size();
    if (n < m) return {0};
    vector<int> ra(A.rbegin(), A.rend());
    vector<int> rb(B.rbegin(), B.rend());
    int k = n - m + 1;
    vector<int> invRb = inverse(rb, k);
    vector<int> q = multiply(ra, invRb);
    q.resize(k);
    reverse(q.begin(), q.end());
    return q;
}
```

## 7. FWT（快速沃尔什变换）

**用途**：**下标做位运算的卷积**：
`C[i] = Σ_{j op k = i} A[j] * B[k]`，`op ∈ {and, or, xor}`。`O(n log n)`。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
constexpr int MOD = 998244353;

// ---- 或卷积 ----
void fwt_or(vector<ll> &a, bool invert) {
    int n = (int)a.size();
    for (int len = 1; len < n; len <<= 1)
        for (int i = 0; i < n; i += len << 1)
            for (int j = 0; j < len; ++j) {
                ll u = a[i + j], v = a[i + j + len];
                if (!invert) a[i + j + len] = (u + v) % MOD;
                else a[i + j + len] = (v - u + MOD) % MOD;
            }
}

// ---- 与卷积 ----
void fwt_and(vector<ll> &a, bool invert) {
    int n = (int)a.size();
    for (int len = 1; len < n; len <<= 1)
        for (int i = 0; i < n; i += len << 1)
            for (int j = 0; j < len; ++j) {
                ll u = a[i + j], v = a[i + j + len];
                if (!invert) a[i + j] = (u + v) % MOD;
                else a[i + j] = (u - v + MOD) % MOD;
            }
}

// ---- 异或卷积 ----
void fwt_xor(vector<ll> &a, bool invert) {
    int n = (int)a.size();
    for (int len = 1; len < n; len <<= 1)
        for (int i = 0; i < n; i += len << 1)
            for (int j = 0; j < len; ++j) {
                ll u = a[i + j], v = a[i + j + len];
                a[i + j] = (u + v) % MOD;
                a[i + j + len] = (u - v + MOD) % MOD;
            }
    if (invert) {
        ll inv_n = 1;
        ll b = n, e = MOD - 2, base = n % MOD;
        while (e) { if (e & 1) inv_n = inv_n * base % MOD; base = base * base % MOD; e >>= 1; }
        (void)b;
        for (ll &x : a) x = x * inv_n % MOD;
    }
}

// 通用：异或卷积
vector<ll> xor_convolution(vector<ll> a, vector<ll> b) {
    int n = 1;
    while (n < (int)max(a.size(), b.size())) n <<= 1;
    a.resize(n); b.resize(n);
    fwt_xor(a, false); fwt_xor(b, false);
    for (int i = 0; i < n; ++i) a[i] = a[i] * b[i] % MOD;
    fwt_xor(a, true);
    return a;
}
```

**实例**：CF 662C Binary Table（异或卷积 + 状压 DP）、P4717【模板】快速沃尔什变换。

**注意**：异或卷积的逆变换要除以 `n`；`and/or` 不需要。

## 8. Berlekamp-Massey（线性递推）

### 8.1 用途

**给一个数列的前若干项，求最短线性递推式**，然后用它算第 `n` 项（配合多项式快速幂 / Kitamasa）。
**典型场景**：猜结论题、矩阵快速幂的替代、`n` 很大但答案是线性递推。

### 8.2 原理

维护当前递推式 `rec`，逐个元素检验，不满足就用"偏差"修正。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

// 求最短线性递推：s[n] = Σ rec[i] * s[n-1-i]
// 返回 rec（长度 = 阶数）
template <class T>
vector<T> berlekamp_massey(const vector<T> &s, T MOD) {
    vector<T> C(1, 1), B(1, 1);        // 当前/上一次的递推多项式
    int L = 0, m = 1;
    T b = 1;
    auto mod = [&](T x) { return ((x % MOD) + MOD) % MOD; };
    auto inv = [&](T x) {
        T r = 1, base = mod(x), e = MOD - 2;
        while (e) { if (e & 1) r = mod(r * base); base = mod(base * base); e >>= 1; }
        return r;
    };
    for (int n = 0; n < (int)s.size(); ++n) {
        T d = s[n];
        for (int i = 1; i <= L; ++i) d = mod(d + mod(C[i] * s[n - i]));
        if (d == 0) { ++m; continue; }
        vector<T> T_ = C;
        T coef = mod(d * inv(b));
        if (C.size() < B.size() + m) C.resize(B.size() + m, 0);
        for (size_t i = 0; i < B.size(); ++i) C[i + m] = mod(C[i + m] - coef * B[i]);
        if (2 * L <= n) {
            L = n + 1 - L;
            B = T_;
            b = d;
            m = 1;
        } else {
            ++m;
        }
    }
    C.resize(L + 1);
    vector<T> rec(L);
    for (int i = 0; i < L; ++i) rec[i] = mod(-C[i + 1]);
    return rec;
}
```

**用法**：

```cpp
// 1. 暴力算前 2k 项（k 是递推阶数上界）
// 2. rec = berlekamp_massey(seq, MOD);
// 3. 用 Kitamasa 算第 n 项（O(k^2 log n) 或 O(k log k log n)）
```

**注意**：要保证**前 2k 项**足够（`k` 为真实阶数）。序列越长越保险。

**Kitamasa（线性递推算第 n 项，O(k^2 log n)）**：

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

// rec: s[n] = rec[0]*s[n-1] + rec[1]*s[n-2] + ... + rec[k-1]*s[n-k]
// 返回第 n 项（0-based）
ll linear_recurrence(const vector<ll> &init, const vector<ll> &rec, ll n, ll MOD) {
    int k = (int)rec.size();
    if (n < (ll)init.size()) return init[n] % MOD;
    auto combine = [&](const vector<ll> &a, const vector<ll> &b) {
        vector<ll> res(2 * k, 0);
        for (int i = 0; i < k; ++i)
            for (int j = 0; j < k; ++j)
                res[i + j] = (res[i + j] + a[i] * b[j]) % MOD;
        for (int i = 2 * k - 2; i >= k; --i) {
            for (int j = 0; j < k; ++j)
                res[i - 1 - j] = (res[i - 1 - j] + res[i] * rec[j]) % MOD;
        }
        res.resize(k);
        return res;
    };
    vector<ll> pol(k, 0), e(k, 0);
    pol[0] = 1;
    if (k == 1) e[0] = rec[0];
    else e[1] = 1;
    while (n) {
        if (n & 1) pol = combine(pol, e);
        e = combine(e, e);
        n >>= 1;
    }
    ll ans = 0;
    for (int i = 0; i < k; ++i) ans = (ans + pol[i] * init[i]) % MOD;
    return ans;
}
```

## 9. 生成函数速查

| 数列 | 生成函数 |
|---|---|
| `1,1,1,...` | `1/(1-x)` |
| `1,2,3,4,...` | `1/(1-x)²` |
| `C(n,k)` | `(1+x)^n` |
| `C(n+k-1,k)`（隔板） | `1/(1-x)^n` |
| 斐波那契 | `x/(1-x-x²)` |
| 卡特兰数 | `(1-sqrt(1-4x))/(2x)` |
| 错排 | `e^(-x)/(1-x)` |

**常见技巧**：
- **卷积** = 生成函数相乘。
- **`1/(1-x)`** = 前缀和。
- **`x/(1-x)`** = 差分。
- **`exp`** = 集合划分；**`ln`** = 去掉空集。
- **求 `f` 的第 n 项** → 多项式求逆/exp 后取系数。

    vector<int> invA = Poly::inverse(A, n);   // 1/A
    vector<int> lnA  = Poly::ln(A, n);        // 要求 A[0] = 1
    // ========================================================

    // ===================== 【输出】改这里 =====================
    for (int i = 0; i < n; ++i) printf("%d%c", invA[i], " \n"[i + 1 == n]);
    // ========================================================
    return 0;
}

## 11. 思路与例题索引

| 算法 | 思路一句话 | 用途 | 例题 |
|---|---|---|---|
| 多项式求逆 | 牛顿迭代 `B = B(2-AB)` | 除法、ln、exp 的基础 | P4238【模板】多项式乘法逆 |
| 多项式 ln | 求导 + 逆 + 积分 | 生成函数变换 | P4725【模板】多项式对数函数 |
| 多项式 exp | 牛顿迭代 `B = B(1 + A - ln B)` | 计数问题 | P4726【模板】多项式指数函数 |
| 多项式开根 | 牛顿迭代 `B = (B + A/B)/2` | 生成函数 | P5205【模板】多项式开根 |
| 多项式除法 | 反转 + 求逆 | 多项式取模 | P4512【模板】多项式除法 |
| 多项式 pow | `exp(k ln A)` | 快速幂 | P5273【模板】多项式幂函数 |
| FWT | 按位分治变换 | 位运算卷积 | P4717【模板】快速沃尔什变换 |
| BM | 偏差修正递推式 | 猜线性递推 | P5487【模板】Berlekamp-Massey |
| Kitamasa | 多项式取模快速幂 | 线性递推第 n 项 | P5487、P1939 矩阵加速 |

**判断**：卷积 → NTT；`1/A` → 逆；`ln/exp` → 牛顿；位运算卷积 → FWT；猜递推 → BM。
