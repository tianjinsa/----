## 1. 前缀和 / 差分 / 二维前缀和

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
    // ---- 一维前缀和 ----
    vector<ll> a = {0, 1, 2, 3, 4};          // 1-based
    vector<ll> pre(a.size(), 0);
    for (size_t i = 1; i < a.size(); ++i) pre[i] = pre[i - 1] + a[i];
    ll s = pre[4] - pre[1];                  // a[2]+a[3]+a[4]

    // ---- 一维差分 ----
    vector<ll> d(a.size() + 1, 0);
    auto range_add = [&](int l, int r, ll v) { d[l] += v; d[r + 1] -= v; };
    range_add(1, 3, 5);
    // 还原：for (i) d[i] += d[i-1]

    // ---- 二维前缀和 ----
    int n = 3, m = 4;
    vector<vector<ll>> g(n + 1, vector<ll>(m + 1, 0)), p2(n + 1, vector<ll>(m + 1, 0));
    for (int i = 1; i <= n; ++i)
        for (int j = 1; j <= m; ++j)
            p2[i][j] = g[i][j] + p2[i - 1][j] + p2[i][j - 1] - p2[i - 1][j - 1];
    auto rect = [&](int x1, int y1, int x2, int y2) {
        return p2[x2][y2] - p2[x1 - 1][y2] - p2[x2][y1 - 1] + p2[x1 - 1][y1 - 1];
    };
    (void)rect; (void)s;
    return 0;
}
```

## 2. 双指针 / 滑动窗口

```cpp
#include <bits/stdc++.h>
using namespace std;

// 求和 >= S 的最短子数组长度
int min_len_sum_at_least(const vector<int> &a, int S) {
    int n = (int)a.size(), ans = n + 1, sum = 0, l = 0;
    for (int r = 0; r < n; ++r) {
        sum += a[r];
        while (sum - a[l] >= S) sum -= a[l++];
        if (sum >= S) ans = min(ans, r - l + 1);
    }
    return ans == n + 1 ? -1 : ans;
}

// 恰好含 k 种不同字符的最长子串
int longest_k_distinct(const string &s, int k) {
    unordered_map<char,int> cnt;
    int l = 0, ans = 0;
    for (int r = 0; r < (int)s.size(); ++r) {
        ++cnt[s[r]];
        while ((int)cnt.size() > k) {
            if (--cnt[s[l]] == 0) cnt.erase(s[l]);
            ++l;
        }
        ans = max(ans, r - l + 1);
    }
    return ans;
}
```

**用途**：最短/最长满足条件的连续区间、去重、滑动窗口最大值（配单调队列）。

## 3. 单调栈 / 单调队列

```cpp
#include <bits/stdc++.h>
using namespace std;

// 单调栈：求每个元素左边第一个更小的位置
vector<int> prev_smaller(const vector<int> &a) {          // 下标 0..n-1
    int n = (int)a.size();
    vector<int> res(n, -1), stk;
    for (int i = 0; i < n; ++i) {
        while (!stk.empty() && a[stk.back()] >= a[i]) stk.pop_back();
        res[i] = stk.empty() ? -1 : stk.back();
        stk.push_back(i);
    }
    return res;
}

// 单调队列：滑动窗口最大值
vector<int> sliding_max(const vector<int> &a, int k) {
    int n = (int)a.size();
    vector<int> res;
    deque<int> dq;
    for (int i = 0; i < n; ++i) {
        while (!dq.empty() && a[dq.back()] <= a[i]) dq.pop_back();
        dq.push_back(i);
        while (dq.front() <= i - k) dq.pop_front();
        if (i >= k - 1) res.push_back(a[dq.front()]);
    }
    return res;
}
```

**用途**：直方图最大矩形、接雨水、滑动窗口最值、DP 优化（见 30 号文件）。

## 4. 离散化

```cpp
#include <bits/stdc++.h>
using namespace std;

// 值 -> 排名（1-based）
template <class T>
vector<int> discretize(const vector<T> &a) {
    vector<T> v = a;
    sort(v.begin(), v.end());
    v.erase(unique(v.begin(), v.end()), v.end());
    vector<int> r(a.size());
    for (size_t i = 0; i < a.size(); ++i)
        r[i] = int(lower_bound(v.begin(), v.end(), a[i]) - v.begin()) + 1;
    return r;
}
```

## 5. 位运算技巧

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    int x = 0b101100;
    int low = x & -x;                       // 最低位的 1
    int clr = x & (x - 1);                  // 去掉最低位的 1
    bool one = (x & (x - 1)) == 0;          // 是否 2 的幂（x>0）
    int pc = __builtin_popcount(x);         // 1 的个数
    int lz = __builtin_clz(x);              // 前导零（x!=0）
    int tz = __builtin_ctz(x);              // 后缀零（x!=0）
    int hb = 31 - __builtin_clz(x);         // 最高位下标

    // 枚举子集（含空集）
    int mask = 0b1011;
    for (int s = mask;; s = (s - 1) & mask) {
        // ...
        if (s == 0) break;
    }

    // 枚举 mask 的超集
    for (int s = mask; s < (1 << 4); s = (s + 1) | mask) { }

    // 位提取/设置/翻转
    x |= (1 << 3); x &= ~(1 << 3); x ^= (1 << 3);
    bool bit = (x >> 3) & 1;
    (void)low; (void)clr; (void)one; (void)pc; (void)lz; (void)tz; (void)hb;
    return 0;
}
```

**`__builtin` 系列**（`int` / `long long` 各一套）：

| 函数 | 含义 |
|---|---|
| `__builtin_popcount(x)` / `popcountll` | 1 的个数 |
| `__builtin_clz(x)` / `clzll` | 前导零个数（`x != 0`） |
| `__builtin_ctz(x)` / `ctzll` | 末尾零个数（`x != 0`） |
| `__builtin_ffs(x)` | 最低位 1 的位置（1-based） |
| `__builtin_parity(x)` | 1 的个数的奇偶 |

## 6. 构造题技巧

| 技巧 | 说明 |
|---|---|
| **小数据找规律** | 打表 `n=1..10` 的答案，猜公式 |
| **对称构造** | 构造回文、镜像 |
| **分块构造** | 把答案拆成若干块，每块单独构造 |
| **逆推** | 从目标状态倒推操作序列 |
| **随机化 + 调整** | 随机生成后局部修正 |
| **按奇偶/模分类** | 分情况讨论 |
| **极限情况** | 先处理 `n=1,2` 等边界 |
| **输出任意解** | 能用简单规律就别硬套复杂结构 |

**实例（本场 A 题）：线性基传播构造**

> A 题要求"最少操作次数 + 构造"。
> **思路**：把每个集合看成 `F₂^m` 的线性子空间。维护公共线性基 `common`，反复把"局部空间 `U_i` 中缺失的元素"并入 `common`，直到不动点；答案是 `V \ ∩S_i`。
> **关键性质**：若 `|U_i| > 2|S_i|`，则缺失元素能张成整个 `U_i` → 可以直接"吸收"，不必显式枚举。
> 这样总枚举量被控制在 `O(mC + 2^m)`。

**实例（本场 H 题）：交互题按位还原**

> **思路**：对每个二进制位 `b`，查询"第 b 位为 1 的顶点集合"的割边数，再逐个翻转顶点得到 `Δ_v`。
> 利用 `Σ Δ_v ≡ 2(n-1) - c (mod 3)` 省掉最后一个顶点的查询。
> **技巧**：利用"模 3 求和"把 `n` 次查询压到 `n-1` 次；利用"内部点某位变化为 0"区分端点。

## 7. 对拍（比赛必备）

**思路**：写三个程序——暴力 `brute.cpp`、正解 `sol.cpp`、数据生成 `gen.cpp`；循环对比。

```cpp
// gen.cpp —— 随机数据生成
#include <bits/stdc++.h>
using namespace std;
mt19937_64 rng(chrono::steady_clock::now().time_since_epoch().count());
int main(int argc, char **argv) {
    int seed = argc > 1 ? atoi(argv[1]) : (int)rng();
    mt19937 rnd(seed);
    int n = rnd() % 10 + 1;
    printf("%d\n", n);
    for (int i = 0; i < n; ++i) printf("%d ", (int)(rnd() % 100));
    printf("\n");
    return 0;
}
```

```bash
# Windows (PowerShell) 对拍脚本
$i = 0
while ($true) {
    $i++
    .\gen.exe $i > in.txt
    .\brute.exe < in.txt > out1.txt
    .\sol.exe   < in.txt > out2.txt
    if (Compare-Object (Get-Content out1.txt) (Get-Content out2.txt)) {
        Write-Output "WA on seed $i"; break
    }
    Write-Output "seed $i OK"
}
```

```bash
# Linux 对拍
for ((i=1;;i++)); do
    ./gen $i > in.txt
    ./brute < in.txt > out1.txt
    ./sol   < in.txt > out2.txt
    if ! diff -q out1.txt out2.txt > /dev/null; then
        echo "WA on seed $i"; break
    fi
done
```

**对拍要点**：
1. 数据要**覆盖边界**（`n=1`、全相同、极值）。
2. 暴力要**绝对正确**（慢没关系）。
3. 输出比较可忽略行末空格（用 `diff -w`）。
4. 找到反例后**保存输入**，逐步缩小数据。

## 8. 调试清单（交题前必查）

| 类别 | 检查项 |
|---|---|
| 输入 | 多测是否清空？`n=1`？空输入？ |
| 溢出 | `int * int` 是否超 `2^31`？用 `ll` 了吗？ |
| 边界 | 数组越界？`0/1` 下标混用？ |
| 初始化 | 全局数组默认 0，局部数组未初始化？ |
| 取模 | 负数取模？`(x % MOD + MOD) % MOD` |
| 精度 | `double` 够吗？`eps` 合适吗？ |
| 递归 | 深度会不会爆栈？（开大栈或改迭代） |
| 输出 | 行末空格、换行、多测间空行 |
| 复杂度 | 最坏情况能过吗？常数大吗？ |
| 题意 | 输出格式、大小写、`YES/NO` |

## 9. 常见"读错题"陷阱

- "**区间是闭还是开**"：`[l, r]` vs `[l, r)`。
- "**下标从 0 还是 1**"。
- "**多组数据之间是否有空行**"。
- "**答案是否要取模**"。
- "**要输出方案还是只需计数**"。
- "**无解输出什么**"（`-1` / `NO` / `Impossible`）。
- "**图是否保证连通**"。
- "**是否可能有重边/自环**"。
