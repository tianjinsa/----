## 1. 基础版：单点改 + 区间查（无懒标记）

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

struct SegTree {
    int n;
    vector<ll> t;
    explicit SegTree(int n_) : n(n_), t(4 * n_, 0) {}

    void build(int p, int l, int r, const vector<ll> &a) {   // a 下标 1..n
        if (l == r) { t[p] = a[l]; return; }
        int m = (l + r) >> 1;
        build(p << 1, l, m, a);
        build(p << 1 | 1, m + 1, r, a);
        t[p] = t[p << 1] + t[p << 1 | 1];
    }

    void update(int p, int l, int r, int pos, ll v) {
        if (l == r) { t[p] = v; return; }
        int m = (l + r) >> 1;
        if (pos <= m) update(p << 1, l, m, pos, v);
        else update(p << 1 | 1, m + 1, r, pos, v);
        t[p] = t[p << 1] + t[p << 1 | 1];
    }

    ll query(int p, int l, int r, int ql, int qr) const {
        if (ql <= l && r <= qr) return t[p];
        int m = (l + r) >> 1;
        ll res = 0;
        if (ql <= m) res += query(p << 1, l, m, ql, qr);
        if (qr > m) res += query(p << 1 | 1, m + 1, r, ql, qr);
        return res;
    }
};
```

## 2. 懒标记：区间加 + 区间和 ★

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

// 区间加、区间和
struct SegAddSum {
    int n;
    vector<ll> sum, lazy;
    explicit SegAddSum(int n_) : n(n_), sum(4 * n_, 0), lazy(4 * n_, 0) {}

    void build(int p, int l, int r, const vector<ll> &a) {
        if (l == r) { sum[p] = a[l]; return; }
        int m = (l + r) >> 1;
        build(p << 1, l, m, a);
        build(p << 1 | 1, m + 1, r, a);
        sum[p] = sum[p << 1] + sum[p << 1 | 1];
    }

    void apply(int p, int l, int r, ll v) {
        sum[p] += v * (r - l + 1);
        lazy[p] += v;
    }
    void push(int p, int l, int r) {
        if (!lazy[p]) return;
        int m = (l + r) >> 1;
        apply(p << 1, l, m, lazy[p]);
        apply(p << 1 | 1, m + 1, r, lazy[p]);
        lazy[p] = 0;
    }

    void add(int p, int l, int r, int ql, int qr, ll v) {
        if (ql <= l && r <= qr) { apply(p, l, r, v); return; }
        push(p, l, r);
        int m = (l + r) >> 1;
        if (ql <= m) add(p << 1, l, m, ql, qr, v);
        if (qr > m) add(p << 1 | 1, m + 1, r, ql, qr, v);
        sum[p] = sum[p << 1] + sum[p << 1 | 1];
    }

    ll query(int p, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr) return sum[p];
        push(p, l, r);
        int m = (l + r) >> 1;
        ll res = 0;
        if (ql <= m) res += query(p << 1, l, m, ql, qr);
        if (qr > m) res += query(p << 1 | 1, m + 1, r, ql, qr);
        return res;
    }
};
```

## 3. 懒标记：区间加 + 区间乘 + 区间和（双标记）

**标记顺序**：先乘后加。`sum = sum * mul + add * len`。下传时 `mul` 和 `add` 都要按 `fa.mul` 缩放。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

struct SegMulAddSum {
    int n;
    ll MOD;
    vector<ll> sum, mul, add;

    SegMulAddSum(int n_, ll mod) : n(n_), MOD(mod),
        sum(4 * n_, 0), mul(4 * n_, 1), add(4 * n_, 0) {}

    void build(int p, int l, int r, const vector<ll> &a) {
        if (l == r) { sum[p] = a[l] % MOD; return; }
        int m = (l + r) >> 1;
        build(p << 1, l, m, a);
        build(p << 1 | 1, m + 1, r, a);
        sum[p] = (sum[p << 1] + sum[p << 1 | 1]) % MOD;
    }

    void apply(int p, int l, int r, ll m_, ll a_) {
        sum[p] = (sum[p] * m_ + a_ * (r - l + 1)) % MOD;
        mul[p] = mul[p] * m_ % MOD;
        add[p] = (add[p] * m_ + a_) % MOD;
    }
    void push(int p, int l, int r) {
        if (mul[p] == 1 && add[p] == 0) return;
        int m = (l + r) >> 1;
        apply(p << 1, l, m, mul[p], add[p]);
        apply(p << 1 | 1, m + 1, r, mul[p], add[p]);
        mul[p] = 1; add[p] = 0;
    }

    // op=1: 区间加 v；op=2: 区间乘 v
    void modify(int p, int l, int r, int ql, int qr, int op, ll v) {
        if (ql <= l && r <= qr) {
            if (op == 1) apply(p, l, r, 1, v);
            else apply(p, l, r, v, 0);
            return;
        }
        push(p, l, r);
        int m = (l + r) >> 1;
        if (ql <= m) modify(p << 1, l, m, ql, qr, op, v);
        if (qr > m) modify(p << 1 | 1, m + 1, r, ql, qr, op, v);
        sum[p] = (sum[p << 1] + sum[p << 1 | 1]) % MOD;
    }

    ll query(int p, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr) return sum[p];
        push(p, l, r);
        int m = (l + r) >> 1;
        ll res = 0;
        if (ql <= m) res += query(p << 1, l, m, ql, qr);
        if (qr > m) res += query(p << 1 | 1, m + 1, r, ql, qr);
        return res % MOD;
    }
};
```

> **推导**：`(x*mul + add) * m2 + a2 = x*(mul*m2) + (add*m2 + a2)`。所以 `mul' = mul*m2`，`add' = add*m2 + a2`。上面 `apply` 的写法与此一致。

## 4. 区间赋值（懒标记 = 赋值标记）

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

// 区间赋值 + 区间和（也可改成区间最值）
struct SegAssignSum {
    int n;
    vector<ll> sum;
    vector<ll> tag;                  // tag[p] != INF 表示整段被赋值
    static constexpr ll NONE = LLONG_MIN;

    explicit SegAssignSum(int n_) : n(n_), sum(4 * n_, 0), tag(4 * n_, NONE) {}

    void build(int p, int l, int r, const vector<ll> &a) {
        if (l == r) { sum[p] = a[l]; return; }
        int m = (l + r) >> 1;
        build(p << 1, l, m, a); build(p << 1 | 1, m + 1, r, a);
        sum[p] = sum[p << 1] + sum[p << 1 | 1];
    }
    void apply(int p, int l, int r, ll v) { sum[p] = v * (r - l + 1); tag[p] = v; }
    void push(int p, int l, int r) {
        if (tag[p] == NONE) return;
        int m = (l + r) >> 1;
        apply(p << 1, l, m, tag[p]);
        apply(p << 1 | 1, m + 1, r, tag[p]);
        tag[p] = NONE;
    }
    void assign(int p, int l, int r, int ql, int qr, ll v) {
        if (ql <= l && r <= qr) { apply(p, l, r, v); return; }
        push(p, l, r);
        int m = (l + r) >> 1;
        if (ql <= m) assign(p << 1, l, m, ql, qr, v);
        if (qr > m) assign(p << 1 | 1, m + 1, r, ql, qr, v);
        sum[p] = sum[p << 1] + sum[p << 1 | 1];
    }
    ll query(int p, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr) return sum[p];
        push(p, l, r);
        int m = (l + r) >> 1;
        ll res = 0;
        if (ql <= m) res += query(p << 1, l, m, ql, qr);
        if (qr > m) res += query(p << 1 | 1, m + 1, r, ql, qr);
        return res;
    }
};
```

## 5. 区间最大子段和（经典：SPOJ GSS3）

每个节点维护 `sum, pre, suf, best` 四元组，`merge` 时按标准公式。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

struct Node {
    ll sum, pre, suf, best;
    Node() : sum(0), pre(0), suf(0), best(0) {}
    Node(ll v) : sum(v), pre(v), suf(v), best(v) {}
    static Node merge(const Node &a, const Node &b) {
        Node c;
        c.sum = a.sum + b.sum;
        c.pre = max(a.pre, a.sum + b.pre);
        c.suf = max(b.suf, b.sum + a.suf);
        c.best = max({a.best, b.best, a.suf + b.pre});
        return c;
    }
};

struct SegGSS {
    int n;
    vector<Node> t;
    explicit SegGSS(int n_) : n(n_), t(4 * n_) {}

    void build(int p, int l, int r, const vector<ll> &a) {
        if (l == r) { t[p] = Node(a[l]); return; }
        int m = (l + r) >> 1;
        build(p << 1, l, m, a);
        build(p << 1 | 1, m + 1, r, a);
        t[p] = Node::merge(t[p << 1], t[p << 1 | 1]);
    }
    void update(int p, int l, int r, int pos, ll v) {
        if (l == r) { t[p] = Node(v); return; }
        int m = (l + r) >> 1;
        if (pos <= m) update(p << 1, l, m, pos, v);
        else update(p << 1 | 1, m + 1, r, pos, v);
        t[p] = Node::merge(t[p << 1], t[p << 1 | 1]);
    }
    Node query(int p, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr) return t[p];
        int m = (l + r) >> 1;
        if (qr <= m) return query(p << 1, l, m, ql, qr);
        if (ql > m) return query(p << 1 | 1, m + 1, r, ql, qr);
        return Node::merge(query(p << 1, l, m, ql, qr), query(p << 1 | 1, m + 1, r, ql, qr));
    }
};
```

> **`best` 初值**：若允许空段，`pre = suf = best = max((ll)0, v)`；若不允许空段，就是 `v`。上面是**不允许空段**的写法。

## 6. 区间最值 + 区间加（懒标记）

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

struct SegAddMax {
    int n;
    vector<ll> mx, lazy;
    explicit SegAddMax(int n_) : n(n_), mx(4 * n_, LLONG_MIN / 4), lazy(4 * n_, 0) {}

    void build(int p, int l, int r, const vector<ll> &a) {
        if (l == r) { mx[p] = a[l]; return; }
        int m = (l + r) >> 1;
        build(p << 1, l, m, a); build(p << 1 | 1, m + 1, r, a);
        mx[p] = max(mx[p << 1], mx[p << 1 | 1]);
    }
    void apply(int p, ll v) { mx[p] += v; lazy[p] += v; }
    void push(int p) {
        if (!lazy[p]) return;
        apply(p << 1, lazy[p]); apply(p << 1 | 1, lazy[p]);
        lazy[p] = 0;
    }
    void add(int p, int l, int r, int ql, int qr, ll v) {
        if (ql <= l && r <= qr) { apply(p, v); return; }
        push(p);
        int m = (l + r) >> 1;
        if (ql <= m) add(p << 1, l, m, ql, qr, v);
        if (qr > m) add(p << 1 | 1, m + 1, r, ql, qr, v);
        mx[p] = max(mx[p << 1], mx[p << 1 | 1]);
    }
    ll query(int p, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr) return mx[p];
        push(p);
        int m = (l + r) >> 1;
        ll res = LLONG_MIN / 4;
        if (ql <= m) res = max(res, query(p << 1, l, m, ql, qr));
        if (qr > m) res = max(res, query(p << 1 | 1, m + 1, r, ql, qr));
        return res;
    }
};
```

## 7. 动态开点线段树（值域大 / 可持久化基础）

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

// 动态开点权值线段树：单点加 + 区间和
struct DynSeg {
    struct Node { int ls = 0, rs = 0; ll sum = 0; };
    vector<Node> tr;
    ll L, R;                          // 值域 [L, R]

    DynSeg(ll l, ll r, int maxNodes = 4000000) : L(l), R(r) {
        tr.reserve(maxNodes);
        tr.push_back(Node{});          // 0 号为空节点
    }
    int newNode() { tr.push_back(Node{}); return (int)tr.size() - 1; }

    void add(int &p, ll l, ll r, ll pos, ll v) {
        if (!p) p = newNode();
        tr[p].sum += v;
        if (l == r) return;
        ll m = (l + r) >> 1;
        if (pos <= m) add(tr[p].ls, l, m, pos, v);
        else add(tr[p].rs, m + 1, r, pos, v);
    }
    ll query(int p, ll l, ll r, ll ql, ll qr) const {
        if (!p || ql > r || qr < l) return 0;
        if (ql <= l && r <= qr) return tr[p].sum;
        ll m = (l + r) >> 1;
        return query(tr[p].ls, l, m, ql, qr) + query(tr[p].rs, m + 1, r, ql, qr);
    }
};
```

## 8. 权值线段树求全局第 k 小 / 排名

```cpp
#include <bits/stdc++.h>
using namespace std;

struct WeightSeg {
    int n;                          // 值域 [1, n]
    vector<int> cnt;
    explicit WeightSeg(int n_) : n(n_), cnt(4 * n_, 0) {}

    void add(int p, int l, int r, int pos, int d) {
        cnt[p] += d;
        if (l == r) return;
        int m = (l + r) >> 1;
        if (pos <= m) add(p << 1, l, m, pos, d);
        else add(p << 1 | 1, m + 1, r, pos, d);
    }
    // 第 k 小（k >= 1）
    int kth(int p, int l, int r, int k) const {
        if (l == r) return l;
        int m = (l + r) >> 1;
        if (cnt[p << 1] >= k) return kth(p << 1, l, m, k);
        return kth(p << 1 | 1, m + 1, r, k - cnt[p << 1]);
    }
    // <= x 的个数
    int count_le(int p, int l, int r, int x) const {
        if (x >= r) return cnt[p];
        int m = (l + r) >> 1;
        if (x <= m) return count_le(p << 1, l, m, x);
        return cnt[p << 1] + count_le(p << 1 | 1, m + 1, r, x);
    }
};
```

**区间第 k 小**（静态）→ 主席树，见 [22-高级数据结构](22-高级数据结构-主席树与树套树.md)。

## 9. 扫描线（矩形面积并）

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

// 矩形面积并：n 个矩形，坐标可大（离散化 y）
struct Scanline {
    struct Event {
        int x, y1, y2, type;          // type: +1 进入, -1 离开
        bool operator<(const Event &o) const { return x < o.x; }
    };
    struct SegNode {
        int cnt, len;                  // 覆盖次数、被覆盖长度
    };

    vector<int> ys;
    vector<SegNode> tr;

    void push_up(int p, int l, int r) {
        if (tr[p].cnt > 0) tr[p].len = ys[r] - ys[l];
        else if (l + 1 == r) tr[p].len = 0;
        else tr[p].len = tr[p << 1].len + tr[p << 1 | 1].len;
    }
    void update(int p, int l, int r, int ql, int qr, int v) {
        if (ql <= l && r <= qr) { tr[p].cnt += v; push_up(p, l, r); return; }
        int m = (l + r) >> 1;
        if (ql < m) update(p << 1, l, m, ql, qr, v);
        if (qr > m) update(p << 1 | 1, m, r, ql, qr, v);
        push_up(p, l, r);
    }

    // rects: {x1, y1, x2, y2}（左下、右上，x1<x2, y1<y2）
    ll area_union(const vector<array<int, 4>> &rects) {
        vector<Event> ev;
        ys.clear();
        for (auto &r : rects) {
            ev.push_back({r[0], r[1], r[3], 1});
            ev.push_back({r[2], r[1], r[3], -1});
            ys.push_back(r[1]); ys.push_back(r[3]);
        }
        sort(ys.begin(), ys.end());
        ys.erase(unique(ys.begin(), ys.end()), ys.end());
        int m = (int)ys.size();
        if (m < 2) return 0;
        tr.assign(4 * m, {0, 0});
        sort(ev.begin(), ev.end());
        ll ans = 0;
        for (size_t i = 0; i < ev.size(); ++i) {
            if (i) ans += 1LL * tr[1].len * (ev[i].x - ev[i - 1].x);
            int l = int(lower_bound(ys.begin(), ys.end(), ev[i].y1) - ys.begin());
            int r = int(lower_bound(ys.begin(), ys.end(), ev[i].y2) - ys.begin());
            update(1, 0, m - 1, l, r, ev[i].type);
        }
        return ans;
    }
};
```

**注意**：`Scanline` 的线段树下标是**离散化的 y 区间**`[0, m-1]`，叶子表示 `[ys[l], ys[l+1])`，所以 `l+1==r` 是叶子。`update(1,0,m-1,l,r,...)` 中 `r` 是开区间端点，与 `push_up` 的 `ys[r]-ys[l]` 配套。

## 10. 线段树二分（找第一个满足条件的位置）

```cpp
#include <bits/stdc++.h>
using namespace std;
// 例：区间置 0 / 区间求和；找 [l, n] 中第一个 1 的位置
// 用线段树维护 sum，查询时优先走左子树
struct SegFindFirst {
    int n;
    vector<int> sum;
    explicit SegFindFirst(int n_) : n(n_), sum(4 * n_, 0) {}
    void build(int p, int l, int r, const vector<int> &a) {
        if (l == r) { sum[p] = a[l]; return; }
        int m = (l + r) >> 1;
        build(p << 1, l, m, a); build(p << 1 | 1, m + 1, r, a);
        sum[p] = sum[p << 1] + sum[p << 1 | 1];
    }
    void assign(int p, int l, int r, int ql, int qr, int v) {
        if (ql <= l && r <= qr) { sum[p] = v * (r - l + 1); return; }
        int m = (l + r) >> 1;
        if (ql <= m) assign(p << 1, l, m, ql, qr, v);
        if (qr > m) assign(p << 1 | 1, m + 1, r, ql, qr, v);
        sum[p] = sum[p << 1] + sum[p << 1 | 1];
    }
    // 在 [ql, qr] 找第一个 sum > 0 的下标，找不到返回 -1
    int find_first(int p, int l, int r, int ql, int qr) {
        if (r < ql || l > qr || sum[p] == 0) return -1;
        if (l == r) return l;
        int m = (l + r) >> 1;
        int res = find_first(p << 1, l, m, ql, qr);
        if (res != -1) return res;
        return find_first(p << 1 | 1, m + 1, r, ql, qr);
    }
};
```

## 11. 常见坑

| 坑 | 说明 |
|---|---|
| 下标 0-based / 1-based 混用 | 统一用 1-based，`build(1,1,n)` |
| 数组开 `4*n` | 开 `4*n+5`，动态开点要算节点数 |
| `push` 时没算对区间长度 | `apply(p,l,r,...)` 要传 `l,r` |
| 懒标记没下传就合并 | 先 `push` 再递归 |
| 多测忘清空 | `fill(sum.begin(), sum.end(), 0)` 或重建 |
| `int * int` 溢出 | 区间和、`lazy * len` 用 `ll` |
| 区间最值初值 | 用 `LLONG_MIN/4` 防加法溢出 |
| 递归深度 | 线段树递归深度 `O(log n)`，安全 |
| 修改/查询的 `ql > qr` | 特判返回 |

**通用 `merge` 线段树（可扩展版）**：把节点类型 + 合并函数 + 懒标记操作模板化，见 [22-高级数据结构](22-高级数据结构-主席树与树套树.md) 的"通用线段树"。
