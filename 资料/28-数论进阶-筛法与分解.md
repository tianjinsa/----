## 1. Miller-Rabin 素性测试 + PollardRho 分解

**用途**：`n <= 1e18` 判素数、分解质因数。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using i128 = __int128_t;
using ull = unsigned long long;

ll mul_mod(ll a, ll b, ll m) { return (ll)((i128)a * b % m); }

ll qpow_mod(ll a, ll b, ll m) {
    ll r = 1 % m; a %= m;
    while (b) { if (b & 1) r = mul_mod(r, a, m); a = mul_mod(a, a, m); b >>= 1; }
    return r;
}

// Miller-Rabin：确定性版本（前 12 个素数足够覆盖 1e18）
bool is_prime(ll n) {
    if (n < 2) return false;
    for (ll p : {2LL, 3LL, 5LL, 7LL, 11LL, 13LL, 17LL, 19LL, 23LL, 29LL, 31LL, 37LL}) {
        if (n % p == 0) return n == p;
    }
    ll d = n - 1, s = 0;
    while ((d & 1) == 0) { d >>= 1; ++s; }
    for (ll a : {2LL, 3LL, 5LL, 7LL, 11LL, 13LL, 17LL, 19LL, 23LL, 29LL, 31LL, 37LL}) {
        if (a >= n) continue;
        ll x = qpow_mod(a, d, n);
        if (x == 1 || x == n - 1) continue;
        bool comp = true;
        for (ll r = 1; r < s; ++r) {
            x = mul_mod(x, x, n);
            if (x == n - 1) { comp = false; break; }
        }
        if (comp) return false;
    }
    return true;
}

mt19937_64 rng_pollard(chrono::steady_clock::now().time_since_epoch().count());

ll pollard(ll n) {
    if (n % 2 == 0) return 2;
    ll x = (ll)(rng_pollard() % (n - 2)) + 2;
    ll y = x, c = (ll)(rng_pollard() % (n - 1)) + 1, d = 1;
    auto f = [&](ll v) { return (mul_mod(v, v, n) + c) % n; };
    while (d == 1) {
        x = f(x);
        y = f(f(y));
        d = __gcd(llabs(x - y), n);
    }
    return d == n ? pollard(n) : d;
}

void factor_rec(ll n, vector<ll> &out) {
    if (n == 1) return;
    if (is_prime(n)) { out.push_back(n); return; }
    ll d = pollard(n);
    factor_rec(d, out);
    factor_rec(n / d, out);
}

// 返回质因数（可重复）
vector<ll> factorize(ll n) {
    vector<ll> out;
    factor_rec(n, out);
    sort(out.begin(), out.end());
    return out;
}
```

## 2. BSGS（离散对数）

**用途**：解 `a^x ≡ b (mod p)`，`gcd(a,p) = 1`。`O(√p)`。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

// 解 a^x ≡ b (mod p)，要求 gcd(a,p)=1；返回最小非负 x，无解返回 -1
ll bsgs(ll a, ll b, ll p) {
    a %= p; b %= p;
    if (b == 1 % p) return 0;
    ll m = (ll)ceil(sqrt((double)p)) + 1;
    unordered_map<ll, ll> table;             // a^j -> j
    ll cur = b;
    for (ll j = 0; j < m; ++j) {
        if (!table.count(cur)) table[cur] = j;
        cur = cur * a % p;
    }
    // a^m
    ll am = 1;
    for (ll i = 0; i < m; ++i) am = am * a % p;
    cur = 1;
    for (ll i = 1; i <= m; ++i) {
        cur = cur * am % p;
        auto it = table.find(cur);
        if (it != table.end()) return i * m - it->second;
    }
    return -1;
}
```

**扩展 BSGS（`gcd(a,p) != 1`）**：不断提取 `g = gcd(a, p)`，直到互质，记录提取次数。

## 3. 原根与阶（本场 E 题相关）

### 3.1 概念

- `ord_n(a)`：最小的 `x > 0` 使 `a^x ≡ 1 (mod n)`；不存在则为 0。
- **原根** `g`：`ord_n(g) = φ(n)`，即 `g` 生成整个单位群。
- **原根存在**：`n = 1, 2, 4, p^k, 2p^k`（`p` 为奇素数）。
- **求原根**：枚举 `g`，检查对 `φ(n)` 的每个质因子 `q`，`g^(φ(n)/q) != 1`。

### 3.2 求原根

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using i128 = __int128_t;

ll phi(ll n) {
    ll r = n;
    for (ll p = 2; p * p <= n; ++p)
        if (n % p == 0) { r = r / p * (p - 1); while (n % p == 0) n /= p; }
    if (n > 1) r = r / n * (n - 1);
    return r;
}

ll qpow_mod(ll a, ll b, ll m) {
    ll r = 1 % m; a %= m;
    while (b) { if (b & 1) r = (i128)r * a % m; a = (i128)a * a % m; b >>= 1; }
    return r;
}

// 返回最小原根；不存在返回 -1
ll primitive_root(ll n) {
    if (n == 1) return 0;
    if (n == 2) return 1;
    ll ph = phi(n);
    // 分解 phi
    vector<ll> qs;
    ll t = ph;
    for (ll p = 2; p * p <= t; ++p)
        if (t % p == 0) { qs.push_back(p); while (t % p == 0) t /= p; }
    if (t > 1) qs.push_back(t);
    for (ll g = 2; g < n; ++g) {
        if (__gcd(g, n) != 1) continue;
        bool ok = true;
        for (ll q : qs) if (qpow_mod(g, ph / q, n) == 1) { ok = false; break; }
        if (ok) return g;
    }
    return -1;
}
```

### 3.3 阶的求法

`ord_n(a)` 一定是 `φ(n)` 的约数。枚举 `φ(n)` 的约数，找最小的满足 `a^x ≡ 1` 的。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using i128 = __int128_t;

ll qpow(ll a, ll b, ll m) {
    ll r = 1 % m; a %= m;
    while (b) { if (b & 1) r = (i128)r * a % m; a = (i128)a * a % m; b >>= 1; }
    return r;
}

// 返回 ord_n(a)；要求 gcd(a,n)=1
ll multiplicative_order(ll a, ll n) {
    ll ph = 1, t = n;
    for (ll p = 2; p * p <= t; ++p)
        if (t % p == 0) {
            ll pk = 1;
            while (t % p == 0) { t /= p; pk *= p; }
            ph *= pk / p * (p - 1);
        }
    if (t > 1) ph *= (t - 1);
    ll ans = ph;
    for (ll d = 2; d * d <= ans; ++d)
        while (ans % d == 0 && qpow(a, ans / d, n) == 1) ans /= d;
    if (ans > 1 && qpow(a, ans / ans, n) == 1) { /* 题目无关，保留 */ }
    return ans;
}
```

### 3.4 E 题：`Σ ord_n(a)` 的做法

**思路**（题解原话）：

1. `gcd(a,n) > 1` 的 `a` 阶为 0，忽略；只看单位群。
2. 用 CRT 把单位群拆成**循环群的直积**：
   - 奇素数幂 `p^e` → `C_{p^(e-1)(p-1)}`
   - `2^1` → 平凡；`2^2` → `C_2`；`2^e (e>=3)` → `C_2 × C_{2^(e-2)}`
3. 把各循环群的大小分解质因数，按质数 `q` 归类：`G_q = C_{q^{a_1}} × ... × C_{q^{a_t}}`。
4. `F(k)` = `G_q` 中阶整除 `q^k` 的元素数 = `q^{Σ min(a_j, k)}`。
5. 阶恰为 `q^k` 的元素数 = `F(k) - F(k-1)`；该质数部分阶之和 `W_q = 1 + Σ q^k (F(k)-F(k-1))`。
6. 答案 = `Π_q W_q`。

## 4. 杜教筛（亚线性求和）

**用途**：求 `Σ_{i=1}^{n} f(i)`，`n <= 1e10`，配合 Dirichlet 卷积能找到 `g` 使 `f*g = h` 前缀和好算。
**典型题**：P4213【模板】杜教筛（`Σμ`、`Σφ`）。

**公式**：`S(n) = Σ_{i=1}^{n} h(i) - Σ_{l=2}^{n} g(l) · S(⌊n/l⌋)`。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using i128 = __int128_t;

// 例：Σ_{i=1}^{n} mu(i) 和 Σ φ(i)
struct DuJiaoSieve {
    static constexpr int N = 5000000;
    vector<int> primes, mu;
    vector<ll> phi, musum, phisum;
    vector<bool> comp;
    unordered_map<ll, ll> memoMu, memoPhi;

    DuJiaoSieve() {
        mu.assign(N + 1, 0);
        phi.assign(N + 1, 0);
        musum.assign(N + 1, 0);
        phisum.assign(N + 1, 0);
        comp.assign(N + 1, false);
        mu[1] = 1; phi[1] = 1;
        for (int i = 2; i <= N; ++i) {
            if (!comp[i]) { primes.push_back(i); mu[i] = -1; phi[i] = i - 1; }
            for (int p : primes) {
                if ((ll)i * p > N) break;
                comp[i * p] = true;
                if (i % p == 0) { mu[i * p] = 0; phi[i * p] = phi[i] * p; break; }
                mu[i * p] = -mu[i];
                phi[i * p] = phi[i] * (p - 1);
            }
        }
        for (int i = 1; i <= N; ++i) {
            musum[i] = musum[i - 1] + mu[i];
            phisum[i] = phisum[i - 1] + phi[i];
        }
    }

    ll sum_mu(ll n) {
        if (n <= N) return musum[n];
        if (memoMu.count(n)) return memoMu[n];
        ll res = 1;                        // Σ mu 的 h = ε
        for (ll l = 2, r; l <= n; l = r + 1) {
            r = n / (n / l);
            res -= (r - l + 1) * sum_mu(n / l);
        }
        return memoMu[n] = res;
    }

    ll sum_phi(ll n) {
        if (n <= N) return phisum[n];
        if (memoPhi.count(n)) return memoPhi[n];
        ll res = n * (n + 1) / 2;          // Σ φ 的 h = id
        for (ll l = 2, r; l <= n; l = r + 1) {
            r = n / (n / l);
            res -= (r - l + 1) * sum_phi(n / l);
        }
        return memoPhi[n] = res;
    }
};
```

**复杂度**：`O(n^(2/3))`（预处理到 `n^(2/3)` 更优）。

## 6. 二次剩余（Cipolla）

**用途**：解 `x² ≡ a (mod p)`，`p` 为奇素数。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using i128 = __int128_t;

ll qpow(ll a, ll b, ll m) {
    ll r = 1 % m; a %= m;
    while (b) { if (b & 1) r = (i128)r * a % m; a = (i128)a * a % m; b >>= 1; }
    return r;
}

// 求 x² ≡ a (mod p) 的一个解；无解返回 -1
ll mod_sqrt(ll a, ll p) {
    if (a == 0) return 0;
    if (qpow(a, (p - 1) / 2, p) != 1) return -1;      // 非二次剩余
    if (p % 4 == 3) return qpow(a, (p + 1) / 4, p);   // 简单情形
    // Tonelli-Shanks（p % 4 == 1）
    ll q = p - 1, s = 0;
    while ((q & 1) == 0) { q >>= 1; ++s; }
    ll z = 2;
    while (qpow(z, (p - 1) / 2, p) != p - 1) ++z;
    ll m = s, c = qpow(z, q, p), t = qpow(a, q, p), r = qpow(a, (q + 1) / 2, p);
    while (t != 1) {
        ll i = 0, tt = t;
        while (tt != 1) { tt = (i128)tt * tt % p; ++i; }
        ll b = qpow(c, 1LL << (m - i - 1), p);
        m = i;
        c = (i128)b * b % p;
        t = (i128)t * c % p;
        r = (i128)r * b % p;
    }
    return r;
}
```

## 7. 类欧几里得（floor sum）

**用途**：快速计算 `Σ_{i=0}^{n-1} ⌊(a*i+b)/m⌋`，`O(log max(a,m))`。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using i128 = __int128_t;

// Σ_{i=0}^{n-1} floor((a*i+b)/m)
ll floor_sum(ll n, ll m, ll a, ll b) {
    ll ans = 0;
    while (true) {
        if (a >= m) { ans += (n - 1) * n / 2 * (a / m); a %= m; }
        if (b >= m) { ans += n * (b / m); b %= m; }
        ll yMax = a * n + b;
        if (yMax < m) break;
        n = yMax / m;
        b = yMax % m;
        swap(m, a);
    }
    return ans;
}
```

**实例**：P5170【模板】类欧几里得算法（同时求三个和式）。

| 需求 | 算法 | 复杂度 |
|---|---|---|

| 小范围素数 | 线性筛 | `O(n)` |
| 区间素数 | 埃氏筛 | `O(n log log n)` |
```
