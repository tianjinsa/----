# 20 计算几何

> 几何题的核心是**精度**和**退化情况**。模板尽量用整数（叉积、点积比大小），只在最后一步用 `long double`。
> 本文件包含本场比赛 B 题用到的：**凸多边形最远点（决策单调分治）**、**半圆上包络 + 圆弧积分**。

## 1. 基础：点、向量、叉积

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ld = long double;

struct Point {
    ld x, y;
    Point operator+(const Point &o) const { return {x + o.x, y + o.y}; }
    Point operator-(const Point &o) const { return {x - o.x, y - o.y}; }
    Point operator*(ld k) const { return {x * k, y * k}; }
    Point operator/(ld k) const { return {x / k, y / k}; }
    bool operator<(const Point &o) const { return x != o.x ? x < o.x : y < o.y; }
    bool operator==(const Point &o) const { return x == o.x && y == o.y; }
};

ld dot(const Point &a, const Point &b) { return a.x * b.x + a.y * b.y; }
ld cross(const Point &a, const Point &b) { return a.x * b.y - a.y * b.x; }
ld len2(const Point &a) { return dot(a, a); }
ld len(const Point &a) { return sqrtl(len2(a)); }
ld dist(const Point &a, const Point &b) { return len(b - a); }

// 角度 / 旋转
Point rotate(const Point &p, ld ang) {
    return {p.x * cosl(ang) - p.y * sinl(ang), p.x * sinl(ang) + p.y * cosl(ang)};
}
ld arg(const Point &p) { return atan2l(p.y, p.x); }
```

**叉积的符号**：`cross(b-a, c-a) > 0` 表示 `a→b→c` 逆时针（左转）。

## 2. 判断：共线 / 相交 / 点在多边形内

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ld = long double;

struct P { ll x, y; };
ll crs(const P &o, const P &a, const P &b) {          // (a-o) × (b-o)
    return (a.x - o.x) * (b.y - o.y) - (a.y - o.y) * (b.x - o.x);
}
ll dt(const P &o, const P &a, const P &b) {           // (a-o) · (b-o)
    return (a.x - o.x) * (b.x - o.x) + (a.y - o.y) * (b.y - o.y);
}

// 三点共线
bool collinear(const P &a, const P &b, const P &c) { return crs(a, b, c) == 0; }

// 点 c 在线段 ab 上（含端点）
bool on_segment(const P &a, const P &b, const P &c) {
    return crs(a, b, c) == 0 && dt(c, a, b) <= 0;
}

// 线段 ab 与 cd 是否相交（含端点接触）
bool seg_intersect(const P &a, const P &b, const P &c, const P &d) {
    ll d1 = crs(c, d, a), d2 = crs(c, d, b);
    ll d3 = crs(a, b, c), d4 = crs(a, b, d);
    if (((d1 > 0 && d2 < 0) || (d1 < 0 && d2 > 0)) &&
        ((d3 > 0 && d4 < 0) || (d3 < 0 && d4 > 0))) return true;
    if (d1 == 0 && on_segment(c, d, a)) return true;
    if (d2 == 0 && on_segment(c, d, b)) return true;
    if (d3 == 0 && on_segment(a, b, c)) return true;
    if (d4 == 0 && on_segment(a, b, d)) return true;
    return false;
}
```

**点在多边形内（射线法，支持凹多边形）**：

```cpp
#include <bits/stdc++.h>
using namespace std;
using ld = long double;

struct Pt { ld x, y; };

// 返回 1 在内、0 在外、-1 在边界上
int point_in_polygon(const vector<Pt> &poly, const Pt &p) {
    int n = (int)poly.size();
    int cnt = 0;
    for (int i = 0; i < n; ++i) {
        const Pt &a = poly[i], &b = poly[(i + 1) % n];
        // 在边上
        ld cr = (b.x - a.x) * (p.y - a.y) - (b.y - a.y) * (p.x - a.x);
        if (fabsl(cr) < 1e-12 && min(a.x, b.x) - 1e-12 <= p.x && p.x <= max(a.x, b.x) + 1e-12 &&
            min(a.y, b.y) - 1e-12 <= p.y && p.y <= max(a.y, b.y) + 1e-12) return -1;
        if ((a.y > p.y) != (b.y > p.y)) {
            ld xin = a.x + (p.y - a.y) / (b.y - a.y) * (b.x - a.x);
            if (xin > p.x) ++cnt;
        }
    }
    return cnt % 2;
}
```

## 3. 凸包（Andrew 单调链）

**思路**：按 `x` 排序，分别求上、下凸壳，拼起来。`O(n log n)`。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

struct Point {
    ll x, y;
    bool operator<(const Point &o) const { return x != o.x ? x < o.x : y < o.y; }
    bool operator==(const Point &o) const { return x == o.x && y == o.y; }
};

ll cross(const Point &o, const Point &a, const Point &b) {
    return (a.x - o.x) * (b.y - o.y) - (a.y - o.y) * (b.x - o.x);
}

// 返回凸包顶点，逆时针顺序，不含重复点
// 若要保留共线点，把 <= 0 改成 < 0
vector<Point> convex_hull(vector<Point> p) {
    sort(p.begin(), p.end());
    p.erase(unique(p.begin(), p.end()), p.end());
    int n = (int)p.size();
    if (n <= 2) return p;
    vector<Point> h(2 * n);
    int k = 0;
    for (int i = 0; i < n; ++i) {
        while (k >= 2 && cross(h[k - 2], h[k - 1], p[i]) <= 0) --k;
        h[k++] = p[i];
    }
    for (int i = n - 2, t = k + 1; i >= 0; --i) {
        while (k >= t && cross(h[k - 2], h[k - 1], p[i]) <= 0) --k;
        h[k++] = p[i];
    }
    h.resize(k - 1);
    return h;
}
```

**用途**：凸包、最大三角形（旋转卡壳）、点集直径。

## 4. 旋转卡壳（凸包直径 / 最远点对）

**思路**：对每条边，找到离它最远的点；随着边转动，最远点也单调移动。`O(n)`。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

struct Point { ll x, y; };
ll cross(const Point &o, const Point &a, const Point &b) {
    return (a.x - o.x) * (b.y - o.y) - (a.y - o.y) * (b.x - o.x);
}
ll dist2(const Point &a, const Point &b) {
    ll dx = a.x - b.x, dy = a.y - b.y;
    return dx * dx + dy * dy;
}

// hull: 逆时针凸包
ll convex_diameter(const vector<Point> &hull) {
    int n = (int)hull.size();
    if (n == 1) return 0;
    if (n == 2) return dist2(hull[0], hull[1]);
    ll ans = 0;
    int j = 1;
    for (int i = 0; i < n; ++i) {
        int ni = (i + 1) % n;
        while (true) {
            int nj = (j + 1) % n;
            // 面积比较：|cross| 越大越远
            ll cur = llabs(cross(hull[i], hull[ni], hull[j]));
            ll nxt = llabs(cross(hull[i], hull[ni], hull[nj]));
            if (nxt > cur) j = nj;
            else break;
        }
        ans = max({ans, dist2(hull[i], hull[j]), dist2(hull[ni], hull[j])});
    }
    return ans;
}
```

**用途**：凸包直径、凸包上最远点对、最大面积三角形。

## 5. 凸多边形上"每个点的最远点"——决策单调分治 ★

> **本场比赛 B 题的核心技术之一**。旋转卡壳的单调性对**凸多边形上所有点**不一定成立（因为最远点可能跨越"最小下标"），所以要把问题拆成"前缀最远点"和"后缀最远点"，各自用**决策单调性分治**求。

**为什么会决策单调**：凸四边形 `a,b,c,d`（逆时针）满足 `|ac| + |bd| > |ad| + |bc|`，因此最远点下标不会逆序。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

// 凸包点 p[0..n-1] 逆时针。
// 求每个 i 的 max_{j 在候选区间内} dist2(p[i], p[j])
struct MonotoneFarthest {
    vector<array<ll,2>> p;                 // 点
    vector<ll> best;                       // 答案（距离平方）

    static ll dist2(const array<ll,2> &a, const array<ll,2> &b) {
        ll dx = a[0] - b[0], dy = a[1] - b[1];
        return dx * dx + dy * dy;
    }

    // [l, r] 是待求解的点下标范围；[optL, optR] 是候选最远点的下标范围
    // suffix = true: 只允许 j > i；false: 只允许 j < i
    void solve(int l, int r, int optL, int optR, bool suffix) {
        if (l > r) return;
        int mid = (l + r) >> 1;
        int begin = max(optL, suffix ? mid + 1 : 0);
        int end   = min(optR, suffix ? (int)p.size() - 1 : mid - 1);
        int bestIdx = begin;
        ll bestVal = -1;
        for (int j = begin; j <= end; ++j) {
            ll d = dist2(p[mid], p[j]);
            if (d > bestVal) { bestVal = d; bestIdx = j; }
        }
        best[mid] = max(best[mid], bestVal);
        solve(l, mid - 1, optL, bestIdx, suffix);       // 左半候选上界缩小
        solve(mid + 1, r, bestIdx, optR, suffix);       // 右半候选下界缩小
    }

    // 调用：best 初始化为 -1；先 solve(0, n-1, 0, n-1, true)，再 solve(0, n-1, 0, n-1, false)
    // 只有在最优下标单调的前提下才正确（凸多边形成立），否则会错。
};
```

> **注意**：这个分治要求 `bestIdx` 随 `mid` **单调不减**。凸多边形满足，一般点集不满足。**如果题目不是凸多边形，请直接 `O(n^2)` 或改用 KD 树**。

## 6. 半平面交 / 直线包络（上包络）★

> B 题把"半圆上包络"转化成"直线包络"：圆的方程 `y² = r² - (x-c)² = -x² + 2cx + (r²-c²)`，
> 所有圆共享 `-x²` 项，所以**比较圆高 = 比较直线 `2cx + (r²-c²)`**。
> 圆心 `c` 递增 ⇒ 斜率 `2c` 递增 ⇒ 可以用**单调栈**维护上包络。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ld = long double;

// 直线 y = k*x + b 的上包络（斜率严格递增地加入）
struct UpperEnvelope {
    struct Line { ld k, b; };
    vector<Line> stk;
    vector<ld> start;                     // 每条直线生效的左端点

    // 两条直线交点 x
    static ld inter(const Line &a, const Line &b) {
        return (b.b - a.b) / (a.k - b.k);
    }

    void add(ld k, ld b) {
        Line cur{k, b};
        ld s = -numeric_limits<ld>::infinity();
        while (!stk.empty()) {
            s = inter(stk.back(), cur);
            if (s <= start.back()) { stk.pop_back(); start.pop_back(); }
            else break;
        }
        if (stk.empty()) s = -numeric_limits<ld>::infinity();
        stk.push_back(cur);
        start.push_back(s);
    }

    // 查询 x 处的最大值
    ld query(ld x) const {
        int idx = int(upper_bound(start.begin(), start.end(), x) - start.begin()) - 1;
        if (idx < 0) idx = 0;
        return stk[idx].k * x + stk[idx].b;
    }
};
```

**B 题的用法**（圆心 `c` 递增，半径 `r`）：

```cpp
// 加入直线：k = 2c，b = r² - c²
// 每条直线生效区间 [start[i], start[i+1])，与 [0, P] 求交
// 面积 = Σ ∫_{L}^{R} sqrt(r² - (x-c)²) dx
```

## 7. 圆弧下面积积分（B 题用）

```cpp
#include <bits/stdc++.h>
using namespace std;
using ld = long double;

// ∫ sqrt(r² - x²) dx 的原函数（x 是相对圆心的横坐标）
// F(x) = (x*sqrt(r²-x²) + r²*asin(x/r)) / 2
ld primitive(ld x, ld r) {
    ld t = max((ld)-1, min((ld)1, x / r));       // 截断，防 asin 定义域误差
    ld under = r * r - x * x;
    if (under < 0) under = 0;                    // 截断，防 sqrt 负数
    return (x * sqrtl(under) + r * r * asinl(t)) / 2;
}

// 圆弧 y = sqrt(r² - (x-c)²) 在 [L, R] 下的面积
// 前提：[L, R] ⊆ [c-r, c+r]
ld arc_area(ld c, ld r, ld L, ld R) {
    return primitive(R - c, r) - primitive(L - c, r);
}
```

**精度提示**：
- 全程 `long double`，输出用 `printf("%.15Lf\n", ans)` 或 `cout << fixed << setprecision(15)`。
- 距离比较用**整数距离平方**（B 题就是这么做）。
- `sqrt` / `asin` 的参数一定要**截断到定义域**。

## 8. 圆的交面积 / 三角形面积

```cpp
#include <bits/stdc++.h>
using namespace std;
using ld = long double;

constexpr ld PI = acosl(-1.0L);

// 两圆交面积
ld circle_intersection_area(ld x1, ld y1, ld r1, ld x2, ld y2, ld r2) {
    ld d = hypotl(x1 - x2, y1 - y2);
    if (d >= r1 + r2) return 0;
    if (d <= fabsl(r1 - r2)) return PI * min(r1, r2) * min(r1, r2);
    ld a1 = acosl((d * d + r1 * r1 - r2 * r2) / (2 * d * r1));
    ld a2 = acosl((d * d + r2 * r2 - r1 * r1) / (2 * d * r2));
    return r1 * r1 * (a1 - sinl(2 * a1) / 2) + r2 * r2 * (a2 - sinl(2 * a2) / 2);
}

// 三角形面积（有向，整数）
long long tri_area2(long long ax, long long ay,
                    long long bx, long long by,
                    long long cx, long long cy) {
    return llabs((bx - ax) * (cy - ay) - (by - ay) * (cx - ax));
}
```

## 9. 最小圆覆盖（随机增量）

**思路**：随机打乱点，逐个加入；若点在当前圆外，以该点为圆心重来。期望 `O(n)`。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ld = long double;

struct P { ld x, y; };
struct Circle { P c; ld r; };

ld dist2(const P &a, const P &b) {
    return (a.x - b.x) * (a.x - b.x) + (a.y - b.y) * (a.y - b.y);
}

// 三点定圆
Circle circumcircle(const P &a, const P &b, const P &c) {
    ld bx = b.x - a.x, by = b.y - a.y;
    ld cx = c.x - a.x, cy = c.y - a.y;
    ld d = 2 * (bx * cy - by * cx);
    ld ux = (cy * (bx * bx + by * by) - by * (cx * cx + cy * cy)) / d;
    ld uy = (bx * (cx * cx + cy * cy) - cx * (bx * bx + by * by)) / d;
    P o{a.x + ux, a.y + uy};
    return {o, sqrtl(dist2(o, a))};
}

// 返回最小覆盖圆
Circle min_enclosing_circle(vector<P> p) {
    unsigned seed = (unsigned)chrono::steady_clock::now().time_since_epoch().count();
    shuffle(p.begin(), p.end(), mt19937(seed));
    Circle c{{0, 0}, 0};
    for (int i = 0; i < (int)p.size(); ++i) {
        if (dist2(p[i], c.c) <= c.r * c.r + 1e-12) continue;
        c = {p[i], 0};
        for (int j = 0; j < i; ++j) {
            if (dist2(p[j], c.c) <= c.r * c.r + 1e-12) continue;
            c = {{(p[i].x + p[j].x) / 2, (p[i].y + p[j].y) / 2}, sqrtl(dist2(p[i], p[j])) / 2};
            for (int k = 0; k < j; ++k) {
                if (dist2(p[k], c.c) <= c.r * c.r + 1e-12) continue;
                c = circumcircle(p[i], p[j], p[k]);
            }
        }
    }
    return c;
}
```

| 交互/多测 | 每次清空全局数组 |

**有向面积判断多边形方向**：

```cpp
// 返回正：逆时针；负：顺时针
long long signed_area2(const vector<pair<long long,long long>> &p) {
    long long s = 0;
    int n = (int)p.size();
    for (int i = 0; i < n; ++i) {
        auto [x1, y1] = p[i];
        auto [x2, y2] = p[(i + 1) % n];
        s += x1 * y2 - x2 * y1;
    }
    return s;
}
```
