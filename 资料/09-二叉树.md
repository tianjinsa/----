## 1. 顺序存储（完全二叉树 / 堆式编号）

**编号规则**（1-based）：节点 `p` 的左儿子 `2p`，右儿子 `2p+1`，父亲 `p/2`。
适用于**完全二叉树**或**堆**；非完全二叉树会浪费空间（最坏 `2^h`）。

```cpp
#include <bits/stdc++.h>
using namespace std;

// 完全二叉树顺序存储：1..n 编号，父子关系由下标决定
struct SeqBinaryTree {
    int n;
    vector<int> a;                 // a[1..n]

    explicit SeqBinaryTree(int n_) : n(n_), a(n_ + 1, 0) {}

    int left(int p) const  { return 2 * p; }
    int right(int p) const { return 2 * p + 1; }
    int parent(int p) const { return p / 2; }

    // 中序遍历（递归）：完全二叉树按位序 a[1..n] 存，中序即 "左根右"
    void inorder(int p, vector<int> &out) const {
        if (p > n) return;
        inorder(left(p), out);
        out.push_back(a[p]);
        inorder(right(p), out);
    }
    // 层序 = 下标顺序
};
```

**判完全二叉树**：给一棵 `n` 个节点的树，从根 BFS，若在遇到"空儿子"之后还遇到非空节点，则不是完全二叉树。

```cpp
#include <bits/stdc++.h>
using namespace std;

// 链式存储的树，判断是否完全二叉树
struct CompleteCheck {
    int n;
    vector<int> L, R;              // L[i], R[i] 为左右儿子，0 表示空
    CompleteCheck(int n_) : n(n_), L(n_ + 1, 0), R(n_ + 1, 0) {}

    bool is_complete(int root) const {
        queue<int> q; q.push(root);
        bool seen_null = false;
        while (!q.empty()) {
            int u = q.front(); q.pop();
            if (!u) { seen_null = true; continue; }
            if (seen_null) return false;       // 空节点后又出现非空
            q.push(L[u]); q.push(R[u]);
        }
        return true;
    }
};
```

## 2. 链式存储 + 三种遍历（递归 / 非递归）

```cpp
#include <bits/stdc++.h>
using namespace std;

struct BTree {
    vector<int> lc, rc, val;
    int root, tot;

    BTree(int maxn) : lc(maxn, 0), rc(maxn, 0), val(maxn, 0), root(0), tot(0) {}

    int newNode(int v) { ++tot; val[tot] = v; lc[tot] = rc[tot] = 0; return tot; }

    // ---- 递归遍历 ----
    void preorder(int u, vector<int> &out) const {
        if (!u) return;
        out.push_back(val[u]);
        preorder(lc[u], out);
        preorder(rc[u], out);
    }
    void inorder(int u, vector<int> &out) const {
        if (!u) return;
        inorder(lc[u], out);
        out.push_back(val[u]);
        inorder(rc[u], out);
    }
    void postorder(int u, vector<int> &out) const {
        if (!u) return;
        postorder(lc[u], out);
        postorder(rc[u], out);
        out.push_back(val[u]);
    }

    // ---- 非递归前序 ----
    vector<int> preorder_iter(int u) const {
        vector<int> out;
        vector<int> stk;
        if (!u) return out;
        stk.push_back(u);
        while (!stk.empty()) {
            int x = stk.back(); stk.pop_back();
            out.push_back(val[x]);
            if (rc[x]) stk.push_back(rc[x]);   // 先压右，后压左
            if (lc[x]) stk.push_back(lc[x]);
        }
        return out;
    }

    // ---- 非递归中序 ----
    vector<int> inorder_iter(int u) const {
        vector<int> out, stk;
        int cur = u;
        while (cur || !stk.empty()) {
            while (cur) { stk.push_back(cur); cur = lc[cur]; }
            cur = stk.back(); stk.pop_back();
            out.push_back(val[cur]);
            cur = rc[cur];
        }
        return out;
    }

    // ---- 非递归后序（双栈法） ----
    vector<int> postorder_iter(int u) const {
        vector<int> out;
        if (!u) return out;
        vector<int> stk{u};
        while (!stk.empty()) {
            int x = stk.back(); stk.pop_back();
            out.push_back(val[x]);
            if (lc[x]) stk.push_back(lc[x]);   // 先压左
            if (rc[x]) stk.push_back(rc[x]);
        }
        reverse(out.begin(), out.end());
        return out;
    }

    // ---- 层序 ----
    vector<int> level_order(int u) const {
        vector<int> out;
        queue<int> q;
        if (u) q.push(u);
        while (!q.empty()) {
            int x = q.front(); q.pop();
            out.push_back(val[x]);
            if (lc[x]) q.push(lc[x]);
            if (rc[x]) q.push(rc[x]);
        }
        return out;
    }
};
```

> **后序非递归（单栈 + last 指针）** 更省内存，但双栈法最不容易写错，比赛推荐双栈。

## 3. 由遍历序列重建二叉树 ★

**结论**：
- 前序 + 中序 → 唯一
- 后序 + 中序 → 唯一
- 前序 + 后序 → **不唯一**（但可确定计数）
- 层序 + 中序 → 唯一

```cpp
#include <bits/stdc++.h>
using namespace std;

// 前序 + 中序 建树，返回根节点下标（1-based，0 表示空）
// pre[l1..r1], in[l2..r2]
struct BuildTree {
    vector<int> lc, rc, val, pos;    // pos[值] = 中序下标
    int tot = 0;

    explicit BuildTree(int n) : lc(n + 1), rc(n + 1), val(n + 1), pos(n + 1) {}

    int build(const vector<int> &pre, int l1, int r1,
              const vector<int> &in, int l2, int r2) {
        if (l1 > r1) return 0;
        int v = pre[l1];
        int p = pos[v];                           // v 在中序中的位置
        int leftSize = p - l2;
        int u = ++tot; val[u] = v;
        lc[u] = build(pre, l1 + 1, l1 + leftSize, in, l2, p - 1);
        rc[u] = build(pre, l1 + leftSize + 1, r1, in, p + 1, r2);
        return u;
    }

    int build_from_pre_in(const vector<int> &pre, const vector<int> &in) {
        int n = (int)pre.size();
        pos.assign(n + 1, 0);
        for (int i = 0; i < n; ++i) pos[in[i]] = i;
        return build(pre, 0, n - 1, in, 0, n - 1);
    }
};
```

**后序 + 中序**：后序的最后一个元素是根。

```cpp skip
// 依赖上面的 BuildTree 结构（pos/tot/lc/rc/val），单独编译会报错
int build_post_in(const vector<int> &post, int l1, int r1,
                  const vector<int> &in, int l2, int r2) {
    if (l1 > r1) return 0;
    int v = post[r1];
    int p = pos[v];
    int rightSize = r2 - p;
    int u = ++tot; val[u] = v;
    lc[u] = build_post_in(post, l1, r1 - rightSize - 1, in, l2, p - 1);
    rc[u] = build_post_in(post, r1 - rightSize, r1 - 1, in, p + 1, r2);
    return u;
}
```

**由"前序 + 后序"数形态数**：根固定，左右子树划分不唯一。设 `solve(l1,r1,l2,r2)`：
若 `pre[l1] == post[r2]`，对每个可能的分割点 `k`（满足 `pre[l1+1] == post[k]`），
`solve(l1+1, l1+1+cnt, l2, k)` × `solve(...)` 求和。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

// 前序 pre[0..n-1] 与 后序 post[0..n-1]，求不同二叉树个数
struct CountForms {
    vector<int> pre, post;
    map<pair<int,int>, ll> memo;              // 记忆化 (l1-l2 偏移, 长度)

    ll solve(int l1, int r1, int l2, int r2) {
        if (l1 > r1) return 1;
        auto key = make_pair(l1 - l2, r1 - l1 + 1);
        auto it = memo.find(key);
        if (it != memo.end()) return it->second;
        ll res = 0;
        // 枚举左子树大小 s = 0..r1-l1
        for (int s = 0; s <= r1 - l1; ++s) {
            int mid = l2 + s - 1;             // 后序左子树结尾
            if (s == 0 || pre[l1 + 1] == post[mid]) {
                res += solve(l1 + 1, l1 + s, l2, l2 + s - 1) *
                       solve(l1 + s + 1, r1, l2 + s, r2 - 1);
            }
        }
        return memo[key] = res;
    }
};
```

## 4. 二叉搜索树 BST

**性质**：左子树全部 `<` 根，右子树全部 `>` 根；中序遍历 = 升序。

**比赛里**：BST 很少手写（会被卡成链），一般用 `set/map`（红黑树）或 **FHQ Treap / Splay**。
但**"给定 BST 的插入序列，求树形态"**、**"判断序列是否是同一棵 BST"** 这类题要用到 BST 性质。

```cpp
#include <bits/stdc++.h>
using namespace std;

// 由插入序列构建 BST（O(n^2) 最坏；n<=5000 可用）
struct BSTFromInsert {
    vector<int> lc, rc, val;
    int root = 0, tot = 0;
    explicit BSTFromInsert(int n) : lc(n + 1), rc(n + 1), val(n + 1) {}

    void insert(int &u, int v) {
        if (!u) { u = ++tot; val[u] = v; return; }
        if (v < val[u]) insert(lc[u], v);
        else if (v > val[u]) insert(rc[u], v);
        // 相等则忽略（按题目要求）
    }
    void build(const vector<int> &a) {
        for (int x : a) insert(root, x);
    }
};
```

**判断两序列对应同一棵 BST**：分别建树后比较**前序/后序序列**是否相同（中序都是排序结果，无法区分）。

**BST 找后继/前驱**：用 `set` 的 `lower_bound` / `upper_bound`。

## 5. 堆（优先队列）

### 5.1 `priority_queue`

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    // 大根堆（默认）
    priority_queue<int> maxq;
    // 小根堆
    priority_queue<int, vector<int>, greater<int>> minq;
    // 存 pair 默认按 first 再 second
    // 自定义：注意比较器返回 true 表示 a 的优先级低于 b（a 排在 b 后面）
    auto cmp = [](const pair<int,int> &a, const pair<int,int> &b) {
        return a.second > b.second;              // 以 second 小者为先
    };
    priority_queue<pair<int,int>, vector<pair<int,int>>, decltype(cmp)> pq(cmp);

    maxq.push(3); maxq.push(1); maxq.push(2);
    while (!maxq.empty()) { int x = maxq.top(); maxq.pop(); }
    return 0;
}
```

### 5.2 手写堆（可删任意元素 / 可修改）

```cpp
#include <bits/stdc++.h>
using namespace std;

// 大根堆，支持 push/pop/删除堆中已知位置的元素
struct Heap {
    vector<int> h;                              // 下标从 1 开始
    Heap() { h.push_back(0); }

    void up(int i) {
        while (i > 1 && h[i] > h[i >> 1]) { swap(h[i], h[i >> 1]); i >>= 1; }
    }
    void down(int i) {
        int n = (int)h.size() - 1;
        while (true) {
            int l = i << 1, r = l | 1, mx = i;
            if (l <= n && h[l] > h[mx]) mx = l;
            if (r <= n && h[r] > h[mx]) mx = r;
            if (mx == i) break;
            swap(h[i], h[mx]);
            i = mx;
        }
    }
    void push(int v) { h.push_back(v); up((int)h.size() - 1); }
    int top() const { return h[1]; }
    void pop() {
        h[1] = h.back(); h.pop_back();
        if (h.size() > 1) down(1);
    }
    void erase_at(int pos) {                    // 删除下标 pos（需自己维护）
        if (pos == (int)h.size() - 1) { h.pop_back(); return; }
        h[pos] = h.back(); h.pop_back();
        up(pos); down(pos);
    }
    int size() const { return (int)h.size() - 1; }
    bool empty() const { return h.size() == 1; }
};

// 堆排序：O(n log n)，原地
void heap_sort(vector<int> &a) {                // a 下标 0..n-1
    auto down = [&](int i, int n, auto &&self) -> void {
        while (true) {
            int l = 2 * i + 1, r = l + 1, mx = i;
            if (l < n && a[l] > a[mx]) mx = l;
            if (r < n && a[r] > a[mx]) mx = r;
            if (mx == i) return;
            swap(a[i], a[mx]);
            i = mx;
        }
    };
    int n = (int)a.size();
    for (int i = n / 2 - 1; i >= 0; --i) down(i, n, down);
    for (int i = n - 1; i > 0; --i) { swap(a[0], a[i]); down(0, i, down); }
}
```

### 5.3 对顶堆：动态中位数

```cpp
#include <bits/stdc++.h>
using namespace std;

struct MedianFinder {
    priority_queue<int> lo;                               // 大根堆，存较小一半
    priority_queue<int, vector<int>, greater<int>> hi;    // 小根堆，存较大一半

    void add(int x) {
        if (lo.empty() || x <= lo.top()) lo.push(x); else hi.push(x);
        if (lo.size() > hi.size() + 1) { hi.push(lo.top()); lo.pop(); }
        else if (hi.size() > lo.size()) { lo.push(hi.top()); hi.pop(); }
    }
    double median() const {
        if (lo.size() == hi.size()) return (lo.top() + hi.top()) / 2.0;
        return lo.top();
    }
};
```

## 6. Huffman 树（合并果子 / 编码）

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

// 合并果子最小代价：每次取最小的两个合并
ll huffman_merge_cost(const vector<ll> &w) {
    priority_queue<ll, vector<ll>, greater<ll>> pq(w.begin(), w.end());
    ll cost = 0;
    while (pq.size() > 1) {
        ll a = pq.top(); pq.pop();
        ll b = pq.top(); pq.pop();
        cost += a + b;
        pq.push(a + b);
    }
    return cost;
}
```

## 7. 二叉树在比赛中的"真面目"

| 题目描述 | 实际算法 |
|---|---|
| "完全二叉树按层编号，求 LCA" | 用 `p/2` 上跳，直到相等 |
| "给前序中序求后序" | 重建树 |
| "求二叉树最大深度/节点数" | DFS |
| "二叉搜索树第 k 小" | 中序遍历 / 树状数组 |
| "带权路径长度最小" | Huffman |
| "堆的操作" | `priority_queue` |
| "二叉树计数" | Catalan 数 |
| "树形 DP" | 见 [18-DP 专题](18-动态规划专题.md) |

**完全二叉树编号求 LCA（P1305 类型）**：

```cpp
#include <bits/stdc++.h>
using namespace std;
int lca_complete(int a, int b) {
    while (a != b) {
        if (a > b) a >>= 1; else b >>= 1;
    }
    return a;
}
```

**Catalan 计数**（`n` 个节点的二叉树形态数）：`C(2n,n)/(n+1)`，见 [02-数论](02-数论与组合数学.md)。

## 8. 平衡二叉树 / 线索二叉树（了解即可）

- **AVL**：左右子树高度差 ≤ 1，旋转维护（比赛不用，写 FHQ 更快）。
- **红黑树**：`std::map/set` 的底层，比赛不用手写。
- **Treap / Splay / FHQ Treap**：比赛实用平衡树，见 [23-平衡树](23-平衡树-FHQ-Treap与Splay.md)。
- **线索二叉树**：把空的左右指针指向前驱/后继，用于 O(1) 空间遍历（比赛极少考，知道概念即可）。

**结论**：比赛里"平衡树"一律用 **FHQ Treap** 或 **Splay**，不要手写 AVL/红黑树。
