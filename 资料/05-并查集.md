## 1. 普通并查集（路径压缩 + 按大小合并）

```cpp
#include <bits/stdc++.h>
using namespace std;

struct DSU {
    vector<int> fa, sz;
    int comps;                       // 连通块个数

    explicit DSU(int n) : fa(n + 1), sz(n + 1, 1), comps(n) {
        iota(fa.begin(), fa.end(), 0);
    }

    int find(int x) {                // 路径压缩（迭代，避免递归深）
        int r = x;
        while (fa[r] != r) r = fa[r];
        while (fa[x] != r) { int nxt = fa[x]; fa[x] = r; x = nxt; }
        return r;
    }

    // 合并；返回是否真的合并了（false 表示原本就同块）
    bool unite(int a, int b) {
        a = find(a); b = find(b);
        if (a == b) return false;
        if (sz[a] < sz[b]) swap(a, b);
        fa[b] = a; sz[a] += sz[b];
        --comps;
        return true;
    }

    bool same(int a, int b) { return find(a) == find(b); }
    int size(int x) { return sz[find(x)]; }
};
```

**复杂度**：路径压缩 + 按秩合并，均摊 `O(α(n))`，近似 `O(1)`。

## 2. 带权并查集（维护到根的偏移）

维护 `d[x]` = `x` 到父节点的"权值"，路径压缩时累加。适用于：**食物链、奇偶性、区间和方程、模意义下的差值约束**。

```cpp
#include <bits/stdc++.h>
using namespace std;

// 带权并查集：维护 d[x] = x 与 fa[x] 的"差"
// 常用于：x 与 y 差为 w、x 与 y 同/异类
struct WeightedDSU {
    vector<int> fa, rk;
    vector<long long> d;             // d[x] = val[x] - val[fa[x]]

    explicit WeightedDSU(int n) : fa(n + 1), rk(n + 1, 1), d(n + 1, 0) {
        iota(fa.begin(), fa.end(), 0);
    }

    int find(int x) {
        if (fa[x] == x) return x;
        int f = fa[x];
        fa[x] = find(fa[x]);
        d[x] += d[f];                // 累加到根的偏移
        return fa[x];
    }

    // 条件：val[x] - val[y] == w；矛盾返回 false
    bool unite(int x, int y, long long w) {
        int rx = find(x), ry = find(y);
        if (rx == ry) return d[x] - d[y] == w;
        if (rk[rx] < rk[ry]) { swap(rx, ry); swap(x, y); w = -w; }
        fa[ry] = rx;
        d[ry] = d[x] - d[y] - w;     // 推导：val[x]-val[y]=w，令 val[ry] 相对 val[rx]
        if (rk[rx] == rk[ry]) ++rk[rx];
        return true;
    }

    // 查询 val[x] - val[y]（需先 find 压缩）
    long long diff(int x, int y) {
        find(x); find(y);
        return d[x] - d[y];
    }
};
```

**推导备忘**：设 `val[x] - val[y] = w`。已知 `d[x] = val[x] - val[rx]`，`d[y] = val[y] - val[ry]`。
合并时 `fa[ry] = rx`，需 `d[ry] = val[ry] - val[rx]`。
由 `val[x] = val[rx] + d[x]`、`val[y] = val[ry] + d[y]`，代入 `val[x] - val[y] = w`：
`d[x] - d[y] - (val[ry] - val[rx]) = w` ⇒ `d[ry] = val[ry] - val[rx] = d[x] - d[y] - w`。✔

**经典题：食物链（P2024）** —— 把"同类/吃/被吃"编码成模 3 的差值，用带权并查集。

```cpp
#include <bits/stdc++.h>
using namespace std;

// 种类并查集 / 带权并查集 解决"食物链"
struct FoodChain {
    vector<int> fa, d;               // d[x] = (x 到根的关系)，0 同类，1 吃根，2 被根吃
    explicit FoodChain(int n) : fa(n + 1), d(n + 1, 0) {
        iota(fa.begin(), fa.end(), 0);
    }
    int find(int x) {
        if (fa[x] == x) return x;
        int f = fa[x];
        fa[x] = find(fa[x]);
        d[x] = (d[x] + d[f]) % 3;
        return fa[x];
    }
    // 返回本次语句是否是真话；type=1 同类，type=2 x 吃 y
    bool say(int type, int x, int y, int n) {
        if (x > n || y > n) return false;
        if (type == 2 && x == y) return false;
        int rx = find(x), ry = find(y);
        if (rx == ry) {
            if (type == 1) return d[x] == d[y];
            return (d[x] - d[y] + 3) % 3 == 1;    // x 吃 y
        }
        // 合并：令 fa[rx] = ry，需要 d[rx]
        // 同类：d[x] == d[y] => d[rx] = d[y] - d[x]
        // x 吃 y：d[x] - d[y] == 1 => d[rx] = d[y] - d[x] + 1
        int need = (type == 1) ? 0 : 1;
        fa[rx] = ry;
        d[rx] = ((d[y] - d[x] + need) % 3 + 3) % 3;
        return true;
    }
};
```

**经典题：P1525 关押罪犯**（二分 + 二分图 / 种类并查集）、**P1892 团伙**（敌人的敌人是朋友 → 拆点并查集）。

## 3. 种类并查集（拆点法）

思路：把每个元素拆成 `k` 个副本，`x*3 + t` 表示"x 属于第 t 类"。条件 `x, y 不同类` → 合并 `x` 与 `y` 的"非同类"副本。比带权并查集更好写、更好调。

```cpp
#include <bits/stdc++.h>
using namespace std;

// 拆点并查集：n 个元素，每个拆成 2 份（0: 本身，1: 反面）
// 条件：x 与 y 相反  =>  unite(x, y+n); unite(x+n, y)
// 矛盾判定：find(x) == find(x+n)
struct SplitDSU {
    vector<int> fa;
    int n;
    explicit SplitDSU(int n_) : n(n_), fa(2 * n_ + 2) {
        iota(fa.begin(), fa.end(), 0);
    }
    int find(int x) {
        while (fa[x] != x) { fa[x] = fa[fa[x]]; x = fa[x]; }
        return x;
    }
    void unite(int a, int b) { fa[find(a)] = find(b); }
    bool opposite_ok(int x, int y) {
        unite(x, y + n); unite(x + n, y);
        return find(x) != find(x + n) && find(y) != find(y + n);
    }
    bool contradiction(int x) { return find(x) == find(x + n); }
};
```

## 4. 可撤销并查集（配合线段树分治/莫队/回滚）

**关键限制**：**不能用路径压缩**（会破坏历史），只能用**按秩/按大小合并**，深度 `O(log n)`。

```cpp
#include <bits/stdc++.h>
using namespace std;

struct RollbackDSU {
    struct Snapshot { int a, b, szb; };
    vector<int> fa, sz;
    vector<Snapshot> stk;
    int comps;

    explicit RollbackDSU(int n) : fa(n + 1), sz(n + 1, 1), comps(n) {
        iota(fa.begin(), fa.end(), 0);
    }

    int find(int x) const {          // 不压缩，O(log n)
        while (fa[x] != x) x = fa[x];
        return x;
    }

    bool unite(int a, int b) {       // 返回是否合并成功
        a = find(a); b = find(b);
        if (a == b) { stk.push_back({0, 0, 0}); return false; }
        if (sz[a] < sz[b]) swap(a, b);
        stk.push_back({a, b, sz[b]});
        fa[b] = a; sz[a] += sz[b]; --comps;
        return true;
    }

    int snapshot() const { return (int)stk.size(); }

    void rollback(int snap) {
        while ((int)stk.size() > snap) {
            auto [a, b, szb] = stk.back();
            stk.pop_back();
            if (!a) continue;
            fa[b] = b; sz[a] -= szb; ++comps;
        }
    }
};
```

**典型用法**：线段树分治（每个操作影响一个时间区间，挂到线段树节点上，DFS 时合并、回溯时撤销）。模板见 [25-分治进阶](25-分治进阶-整体二分点分治dsu-on-tree.md)。

## 5. 二维并查集 / 网格连通

网格题把 `(i,j)` 编号成 `i * m + j`，四联通就 `unite` 相邻格子。注意**边界判断**。

```cpp
#include <bits/stdc++.h>
using namespace std;
int main() {
    int n = 3, m = 4;
    auto id = [&](int i, int j) { return i * m + j; };
    vector<int> fa(n * m);
    iota(fa.begin(), fa.end(), 0);
    function<int(int)> find = [&](int x) { return fa[x] == x ? x : fa[x] = find(fa[x]); };
    auto unite = [&](int a, int b) { fa[find(a)] = find(b); };
    for (int i = 0; i < n; ++i)
        for (int j = 0; j < m; ++j) {
            if (i + 1 < n) unite(id(i, j), id(i + 1, j));
            if (j + 1 < m) unite(id(i, j), id(i, j + 1));
        }
    return 0;
}
```

## 6. 常见套路

| 场景 | 做法 |
|---|---|
| 判断无向图成环 | 加边时 `unite` 返回 `false` 即有环 |
| Kruskal 最小生成树 | 按边权排序，`unite` 成功就选这条边 |
| 连通块个数 | 维护 `comps`，合并时自减 |
| 动态加边、查询连通 | 并查集；**要删边** → 线段树分治 + 可撤销并查集 / 离线倒序 |
| `x 与 y 同类/异类` | 种类并查集（拆点）或带权并查集 |
| 区间加、问 `[l,r]` 是否全同 | 并查集 + 每个点指向右侧第一个不同点（"跳跃指针"） |
| 离线区间合并去重 | 并查集维护"下一个未处理位置" |
| 树上并查集（Tarjan 离线 LCA） | `find` 沿父链，配合 DFS 回溯 |
| 字典序最小 / 贪心合并 | 按秩合并时保留根信息或维护 `vector` |

**"下一个未处理位置"模板**（经典技巧，O(n α)）：

```cpp
#include <bits/stdc++.h>
using namespace std;
// nxt[i] 表示 >= i 的第一个未被处理的位置，初始 nxt[i]=i
// 处理完 i 后 nxt[i] = find(i+1)
struct NextDSU {
    vector<int> nxt;
    explicit NextDSU(int n) : nxt(n + 2) { iota(nxt.begin(), nxt.end(), 0); }
    int find(int x) { return nxt[x] == x ? x : nxt[x] = find(nxt[x]); }
    void erase(int x) { nxt[x] = find(x + 1); }
    // 遍历未处理位置：for (int i = find(l); i <= r; i = find(i + 1)) ...
};
```

## 7. 判断题：什么时候不能用并查集

- **需要删除边**：普通并查集不支持。用线段树分治 / 倒序处理 / LCT。
- **需要知道连通块内具体元素**：用 `vector` 合并（小并大，`O(n log n)`）或 `set` 启发式合并。
- **需要图的强连通分量**：用 Tarjan，不是并查集。
- **并查集 + 模意义下的和/积**：用带权并查集，注意**除法要转逆元**。
