# 03 中国剩余定理与模线性方程

> 你点名的第一个重点。这里给全：**CRT（模数两两互质）**、**exCRT（模数任意）**、**exLucas（模数非质数组合数）**、**任意模数 NTT 思路**。
> 三者关系：CRT 是"已知各分量求整体"；exCRT 允许模数不互质；exLucas 是"模数分解成素数幂 → 分别 Lucas → CRT 合并"。

---

## 1. 中国剩余定理 CRT（模数两两互质）

求最小的 `x >= 0` 满足

```
x ≡ a1 (mod m1)
x ≡ a2 (mod m2)
...
x ≡ an (mod mn)
```

**公式**：`M = Π mi`，`Mi = M / mi`，`ti = Mi^(-1) mod mi`，`x = Σ ai * Mi * ti (mod M)`

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using i128 = __int128_t;

ll exgcd(ll a, ll b, ll &x, ll &y) {
    if (!b) { x = 1; y = 0; return a; }
    ll g = exgcd(b, a % b, y, x);
    y -= a / b * x;
    return g;
}
ll inv_mod(ll a, ll m) {           // gcd(a,m)=1 时
    ll x, y; exgcd(a, m, x, y);
    return ((x % m) + m) % m;
}

// CRT：要求所有 m[i] 两两互质，且答案 < Πm（保证不超 long long 时使用 i128 内部计算）
// 返回最小非负解；无解返回 -1
ll crt(const vector<ll> &a, const vector<ll> &m) {
    int n = a.size();
    ll M = 1;
    for (int i = 0; i < n; ++i) {
        M *= m[i];
        if (M > (ll)4e18) return -1;         // 溢出保护，按需调整
    }
    ll ans = 0;
    for (int i = 0; i < n; ++i) {
        ll Mi = M / m[i];
        ll ti = inv_mod(Mi % m[i], m[i]);
        ans = (ans + (i128)(a[i] % m[i]) * Mi % M * ti) % M;
    }
    return ((ans % M) + M) % M;
}
```

**模板题**：洛谷 P1495「曹冲养猪」（`x ≡ a_i (mod m_i)`，模数互质）。
**注意**：`M` 可能超过 `long long`（`Πm_i > 9e18`）。此时改用 `i128` 或 exCRT 内部取模（下面 exCRT 就是这个思路）。

## 2. 扩展中国剩余定理 exCRT（模数无需互质）★

**核心思想**：两两合并。

已有解 `x ≡ r1 (mod m1)`，新方程 `x ≡ r2 (mod m2)`：
令 `x = r1 + m1 * k`，代入得 `m1 * k ≡ r2 - r1 (mod m2)`，用 exgcd 解 `k`，得到新的 `r, m = lcm(m1, m2)`。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using i128 = __int128_t;

// 返回 (g, x, y) 使 a*x + b*y = g
ll exgcd(ll a, ll b, ll &x, ll &y) {
    if (!b) { x = 1; y = 0; return a; }
    ll g = exgcd(b, a % b, y, x);
    y -= a / b * x;
    return g;
}

// 单个同余方程 a*x ≡ b (mod m)，返回最小非负解 x，无解返回 -1
// 通解 x0 + k*(m/g)
ll solve_congruence(ll a, ll b, ll m) {
    ll x, y;
    ll g = exgcd(a, m, x, y);
    if (b % g) return -1;
    ll mg = m / g;
    x = (i128)(x % mg) * ((b / g) % mg) % mg;
    x = (x % mg + mg) % mg;
    return x;
}

// exCRT：x ≡ a[i] (mod m[i])，模数任意，返回最小非负解，无解返回 -1
// 若答案可能超过 64 位请把 ll 换成 i128（见下方 i128 版本）
ll excrt(const vector<ll> &a, const vector<ll> &m) {
    ll r = 0, M = 1;                     // 当前 x ≡ r (mod M)
    for (size_t i = 0; i < a.size(); ++i) {
        ll ai = a[i] % m[i], mi = m[i];
        // 解 M*k ≡ ai - r (mod mi)
        ll x, y;
        ll g = exgcd(M, mi, x, y);
        ll diff = ((ai - r) % mi + mi) % mi;
        if (diff % g) return -1;         // 无解
        ll mg = mi / g;
        x = (i128)(x % mg) * ((diff / g) % mg) % mg;
        x = (x % mg + mg) % mg;
        // 新 r = r + M*x，新 M = lcm(M, mi)
        i128 nr = (i128)r + (i128)M * x;
        i128 nM = (i128)M / g * mi;
        r = (ll)(nr % nM);
        M = (ll)nM;
        if (r < 0) r += M;
    }
    return r % M;
}
```

**⚠️ 上面用 `i128` 临时算 `nr`，但如果 `lcm` 本身超过 `1e18`（比如两个 `1e18` 左右的模数）仍然会溢出。** 真正任意大的版本用 `i128` 全程：

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using i128 = __int128_t;

i128 exgcd128(i128 a, i128 b, i128 &x, i128 &y) {
    if (!b) { x = 1; y = 0; return a < 0 ? -a : a; }
    i128 g = exgcd128(b, (a % b + b) % b, y, x);
    y -= (a / b) * x;
    return g;
}

// 返回 {r, M}，表示 x ≡ r (mod M)；无解返回 {0, 0}
pair<i128, i128> excrt128(vector<i128> a, vector<i128> m) {
    i128 r = 0, M = 1;
    for (size_t i = 0; i < a.size(); ++i) {
        i128 ai = ((a[i] % m[i]) + m[i]) % m[i];
        i128 x, y;
        i128 g = exgcd128(M, m[i], x, y);
        i128 diff = ((ai - r) % m[i] + m[i]) % m[i];
        if (diff % g) return {0, 0};
        i128 mg = m[i] / g;
        x = (x % mg + mg) % mg;
        x = (i128)((x * ((diff / g) % mg)) % mg);
        i128 nM = M / g * m[i];
        r = (r + (i128)M * x) % nM;
        M = nM;
    }
    return {r % M, M};
}
```

**模板题**：洛谷 P4777「【模板】扩展中国剩余定理（EXCRT）」、P1495。

**常见题型**
- 直接给同余方程组。
- 给递推/周期问题，问"第几个时刻同时满足" → 每个条件化成 mod 方程再 exCRT。
- 模数不互质且需要判无解（`diff % g != 0`）。

## 3. exLucas（任意模数组合数）

求 `C(n, m) mod p`，`p` 不一定是质数。做法：

1. 分解 `p = Π pi^ki`
2. 对每个 `pi^ki` 求 `C(n,m) mod pi^ki`（递归计算去掉 p 因子的阶乘）
3. 用 CRT 合并

**预处理**：`fac[i] = i!` 去掉所有 `p` 因子后 `mod pk`（`i` 从 `0` 到 `pk`），时间复杂度 `O(pk)`，所以要求 `pk <= 1e6` 左右。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using i128 = __int128_t;

ll qpw(ll a, ll b, ll m) {
    ll r = 1 % m; a %= m;
    while (b) { if (b & 1) r = (i128)r * a % m; a = (i128)a * a % m; b >>= 1; }
    return r;
}
ll exgcd(ll a, ll b, ll &x, ll &y) {
    if (!b) { x = 1; y = 0; return a; }
    ll g = exgcd(b, a % b, y, x);
    y -= a / b * x;
    return g;
}

// ---------- exLucas: C(n,m) mod p, p 任意 ----------
namespace exLucas {
    ll p, pk, fac[1000005];

    // 计算 n! 中，去掉所有 p 因子后 mod pk 的结果，同时把 p 的指数加到 cnt
    ll calc(ll n, ll &cnt) {
        if (!n) return 1;
        cnt += n / p;
        return calc(n / p, cnt) * qpw(fac[pk], n / pk, pk) % pk * fac[n % pk] % pk;
    }

    // 在模 pk 下求 C(n,m)
    ll C(ll n, ll m) {
        if (m < 0 || m > n) return 0;
        ll c1 = 0, c2 = 0, c3 = 0;
        ll a = calc(n, c1), b = calc(m, c2), c = calc(n - m, c3);
        ll cnt = c1 - c2 - c3;
        ll phi = pk / p * (p - 1);
        ll res = a * qpw(b, phi - 1, pk) % pk * qpw(c, phi - 1, pk) % pk;
        return res * qpw(p, cnt, pk) % pk;
    }
}

// 主函数：C(n,m) mod P，P 任意，P <= 1e9 左右（分解后 pk 不超过 1e6 更稳妥）
// 返回值用 CRT 合并（这里只处理两个分量时可用，多分量建议把下面改成数组）
ll exlucas_C(ll n, ll m, ll P) {
    if (m < 0 || m > n) return 0;
    // 分解 P
    vector<pair<ll, int>> fs;
    ll x = P;
    for (ll d = 2; d * d <= x; ++d)
        if (x % d == 0) {
            int c = 0;
            while (x % d == 0) { x /= d; ++c; }
            fs.push_back({d, c});
        }
    if (x > 1) fs.push_back({x, 1});

    // 对每个素因子幂求值，然后 CRT 合并
    vector<ll> rem, mod;
    for (auto [prime, cnt] : fs) {
        ll pk = 1;
        for (int i = 0; i < cnt; ++i) pk *= prime;
        exLucas::p = prime; exLucas::pk = pk;
        exLucas::fac[0] = 1;
        for (ll i = 1; i <= pk; ++i)
            exLucas::fac[i] = (i % prime == 0) ? exLucas::fac[i - 1]
                                              : exLucas::fac[i - 1] * i % pk;
        rem.push_back(exLucas::C(n, m));
        mod.push_back(pk);
    }
    // CRT 合并
    ll r = 0, M = 1;
    for (size_t i = 0; i < rem.size(); ++i) {
        ll x2, y2;
        ll g = exgcd(M, mod[i], x2, y2);
        ll mg = mod[i] / g;
        ll diff = ((rem[i] - r) % mod[i] + mod[i]) % mod[i];
        ll k = (i128)(x2 % mg + mg) % mg * ((diff / g) % mg) % mg;
        r = r + M * k;
        M = M / g * mod[i];
        r = (r % M + M) % M;
    }
    return r;
}
```

**⚠️ 这是 `exLucas` 中最容易写错的两处：**
1. `calc` 里 `fac[n % pk]` 和 `qpw(fac[pk], n/pk, pk)` 的顺序/含义，`fac` 要预计算到 `pk` 且**跳过 p 的倍数**。
2. 最后乘 `p^cnt` 时，若 `cnt >= 1` 结果自动为 0 的边界。`pw(p, cnt, pk)` 已正确处理。

**模板题**：洛谷 P4720「【模板】扩展卢卡斯」。

## 4. 任意模数 NTT（MTT）

当需要做 `NTT` 但模数不是 `998244353` 时，三种做法：

| 做法 | 原理 | 适用 |
|---|---|---|
| 三模 NTT + CRT | 用 3 个 NTT 模数分别算，再 CRT | 系数 < 1e9 常用 |
| 拆位 FFT | 系数拆成高低 15 位 | 任意模数 |
| 直接 `__int128` FFT | long double FFT 精度不够时 | 小数据 |

下面是**三模数 NTT 的 CRT 合并**核心（完整 NTT 见 04 多项式文件）：

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using i128 = __int128_t;

constexpr ll M1 = 998244353, M2 = 1004535809, M3 = 469762049;
constexpr ll inv_M1_mod_M2 = 0; // 需程序计算见下

// 三模 CRT：已知 x ≡ a1 (mod M1), x ≡ a2 (mod M2), x ≡ a3 (mod M3)
// 求 x mod MOD（MOD 任意，x < M1*M2*M3）
// 公式：t1 = a1
//       t2 = (a2 - t1) * inv(M1) mod M2
//       t3 = (a3 - t1 - t2*M1) * inv(M1*M2) mod M3
//       x  = t1 + t2*M1 + t3*M1*M2
ll qpow(ll a, ll b, ll m) {
    ll r = 1; a %= m;
    while (b) { if (b & 1) r = r * a % m; a = a * a % m; b >>= 1; }
    return r;
}

ll crt3(ll a1, ll a2, ll a3, ll MOD) {
    static const ll inv1 = qpow(M1 % M2, M2 - 2, M2);
    static const ll inv12 = qpow((i128)M1 % M3 * (M2 % M3) % M3, M3 - 2, M3);
    ll t1 = a1 % M1;
    ll t2 = (i128)((a2 - t1 % M2 + M2) % M2) * inv1 % M2;
    ll part = (a3 - t1 % M3 - (i128)(t2 % M3) * (M1 % M3) % M3) % M3;
    part = (part + M3) % M3;
    ll t3 = (i128)part * inv12 % M3;
    // 全部用 i128 计算再取模，避免溢出
    i128 x = (i128)t1 + (i128)t2 * M1 + (i128)t3 * M1 % (i128)MOD * M2 % (i128)MOD;
    x %= MOD;
    return (ll)(x % MOD + MOD) % MOD;
}
```

> 实际比赛更常用的是**拆系数 FFT（MTT）**或直接 `#define` 改 NTT 模数。若题目模数是 `1e9+7` 且 `n <= 2e5`，三模 CRT 够用；若需要"任意模数"且精度敏感，用拆位法（见 27 多项式全家桶）。

## 5. 常见考法总结

| 题型 | 使用 |
|---|---|
| 给若干同余方程求解 | CRT（互质）/ exCRT（不互质） |
| 大数组合数模合数 | exLucas |
| 周期/循环节 + 剩余 | exCRT 合并多个 mod 条件 |
| 多个「x 在模某数下等于某值」 | exCRT，注意判无解 |
| 模数过大（> 1e18） | `i128` 版 exCRT |
| FFT 任意模 | 三模 CRT / 拆位 MTT |

**CRT 的"最小正整数解"陷阱**：`(ans % M + M) % M`，若 `ans == 0` 且题目要求正整数，输出 `M`。


### 怎么用（照抄这段 `main`，只改【输入】【输出】两处）

上面是模板本体。实际写题时：**把上面的 struct 整段粘过来 → 再粘这段 `main` → 只改标了【输入】【输出】的行**。

```cpp skip
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

// ===== 粘贴点：把上面的 exgcd / inv_mod / crt / excrt / exlucas_C 全部粘到这里 =====

int main() {
    // ===================== 【输入】改这里 =====================
    int n;
    scanf("%d", &n);                     // 方程个数
    vector<ll> a(n), m(n);
    for (int i = 0; i < n; ++i)
        scanf("%lld %lld", &a[i], &m[i]); // 第 i 行：余数 模数
    // ========================================================

    // ===================== 【核心】算法流程 =====================
    // 1) 单条件：x ≡ a[i] (mod m[i])
    // 2) 两两合并：设已有 x ≡ r (mod M)，新条件 x ≡ a (mod m)
    //    代入 x = r + M*k  →  M*k ≡ a-r (mod m)
    //    用 exgcd 解 k；无解 ⇒ 整个方程组无解，返回 -1
    // 3) 合并结果：r ← r + M*k，M ← lcm(M, m)，继续下一个条件
    ll x = excrt(a, m);                  // 返回最小非负解；无解返回 -1
    // ========================================================

    // ===================== 【输出】改这里 =====================
    if (x < 0) puts("NO");               // 方程组矛盾
    else       printf("%lld\n", x);
    // ========================================================
    return 0;
}
```

**组合数模合数**只要换一行核心调用：

```cpp skip
ll ans = exlucas_C(n, m, P);   // C(n,m) mod P，P 可为合数；n、m、P 自己读入
```

## 6. 思路与例题索引

| 算法 | 思路一句话 | 用途 | 例题 |
|---|---|---|---|
| 普通 CRT | 每项构造"除本位外全为 0"的贡献，再求和 | 模数**两两互质**时合并同余式 | P1495 曹冲养猪 |
| exCRT | 两两合并：`x = r1 + m1·k` 代入新方程解 `k` | 模数**不互质**、需要判无解 | P4777【模板】EXCRT |
| exLucas | 模数分解成素数幂 → 分别 Lucas → CRT 合并 | 组合数模**合数** | P4720【模板】扩展卢卡斯 |
| 三模 NTT | 三个 NTT 模数分别算，再 CRT | 任意模数卷积 | P4245【模板】任意模数多项式乘法 |
| 模逆元 | 费马（质数）/ exgcd（一般） | 除法取模 | P3811【模板】乘法逆元 |

**怎么套**：先看模数是否互质 → 互质用 CRT，否则 exCRT；组合数再看模数是否质数 → 不是就用 exLucas。
