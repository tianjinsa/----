## 1. Dinic 最大流 ★（必背）

**核心**：BFS 分层 + DFS 多路增广（当前弧优化）。`O(n^2 m)`，实际远快于上界。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
constexpr ll LINF = 0x3f3f3f3f3f3f3f3fLL;

struct Dinic {
    struct Edge { int to, rev; ll cap; };
    int n;
    vector<vector<Edge>> g;
    vector<int> level, it;

    explicit Dinic(int n_) : n(n_), g(n_ + 1), level(n_ + 1), it(n_ + 1) {}
    void addEdge(int u, int v, ll cap) {
        g[u].push_back({v, (int)g[v].size(), cap});
        g[v].push_back({u, (int)g[u].size() - 1, 0});
    }
    // 无向图加边：两条有向边容量都是 cap
    void addBiEdge(int u, int v, ll cap) {
        g[u].push_back({v, (int)g[v].size(), cap});
        g[v].push_back({u, (int)g[u].size() - 1, cap});
    }

    bool bfs(int s, int t) {
        fill(level.begin(), level.end(), -1);
        queue<int> q; level[s] = 0; q.push(s);
        while (!q.empty()) {
            int u = q.front(); q.pop();
            for (auto &e : g[u]) if (e.cap > 0 && level[e.to] < 0) {
                level[e.to] = level[u] + 1;
                q.push(e.to);
            }
        }
        return level[t] >= 0;
    }
    ll dfs(int u, int t, ll f) {
        if (u == t) return f;
        for (int &i = it[u]; i < (int)g[u].size(); ++i) {
            Edge &e = g[u][i];
            if (e.cap > 0 && level[e.to] == level[u] + 1) {
                ll d = dfs(e.to, t, min(f, e.cap));
                if (d > 0) {
                    e.cap -= d;
                    g[e.to][e.rev].cap += d;
                    return d;
                }
            }
        }
        return 0;
    }
    ll maxFlow(int s, int t) {
        ll flow = 0;
        while (bfs(s, t)) {
            fill(it.begin(), it.end(), 0);
            ll f;
            while ((f = dfs(s, t, LINF)) > 0) flow += f;
        }
        return flow;
    }
};
```

**注意**：
- `cap` 用 `ll`，`node` 用 `int`。
- `dfs` 里 `if (d > 0) return d;` 是**单路**写法（正确但稍慢）；**多路**写法不 `return`，把 `f` 减掉继续：

```cpp skip
// 依赖 Dinic 结构的成员，片段，单独编译不通过
    ll dfs(int u, int t, ll f) {          // 多路增广版，更快
        if (u == t) return f;
        ll used = 0;
        for (int &i = it[u]; i < (int)g[u].size(); ++i) {
            Edge &e = g[u][i];
            if (e.cap > 0 && level[e.to] == level[u] + 1) {
                ll d = dfs(e.to, t, min(f - used, e.cap));
                if (d > 0) {
                    e.cap -= d;
                    g[e.to][e.rev].cap += d;
                    used += d;
                    if (used == f) return used;
                }
            }
        }
        return used;
    }
```

## 2. 二分图最大匹配

### 2.1 Kuhn（匈牙利）/ 匈牙利算法 `O(nm)`

```cpp
#include <bits/stdc++.h>
using namespace std;

struct Kuhn {
    int n, m;                            // 左部 1..n，右部 1..m
    vector<vector<int>> g;
    vector<int> match;                   // match[v] = 右部 v 匹配的左部点
    vector<int> vis;
    int timer = 0;

    explicit Kuhn(int n_, int m_) : n(n_), m(m_), g(n_ + 1), match(m_ + 1, 0), vis(m_ + 1, 0) {}
    void addEdge(int u, int v) { g[u].push_back(v); }      // u 左, v 右

    bool dfs(int u) {
        for (int v : g[u]) {
            if (vis[v] == timer) continue;
            vis[v] = timer;
            if (!match[v] || dfs(match[v])) { match[v] = u; return true; }
        }
        return false;
    }
    int run() {
        int res = 0;
        for (int u = 1; u <= n; ++u) {
            ++timer;
            if (dfs(u)) ++res;
        }
        return res;
    }
};
```

### 2.2 Hopcroft-Karp `O(m√n)`

```cpp
#include <bits/stdc++.h>
using namespace std;

struct HopcroftKarp {
    int n, m;
    vector<vector<int>> g;
    vector<int> ml, mr, dist;
    int INF = 1e9;

    explicit HopcroftKarp(int n_, int m_) : n(n_), m(m_), g(n_ + 1),
        ml(n_ + 1, 0), mr(m_ + 1, 0), dist(n_ + 1) {}
    void addEdge(int u, int v) { g[u].push_back(v); }

    bool bfs() {
        queue<int> q;
        for (int u = 1; u <= n; ++u) {
            if (!ml[u]) { dist[u] = 0; q.push(u); }
            else dist[u] = INF;
        }
        bool found = false;
        while (!q.empty()) {
            int u = q.front(); q.pop();
            for (int v : g[u]) {
                int w = mr[v];
                if (!w) found = true;
                else if (dist[w] == INF) { dist[w] = dist[u] + 1; q.push(w); }
            }
        }
        return found;
    }
    bool dfs(int u) {
        for (int v : g[u]) {
            int w = mr[v];
            if (!w || (dist[w] == dist[u] + 1 && dfs(w))) {
                ml[u] = v; mr[v] = u;
                return true;
            }
        }
        dist[u] = INF;
        return false;
    }
    int run() {
        int res = 0;
        while (bfs()) for (int u = 1; u <= n; ++u) if (!ml[u] && dfs(u)) ++res;
        return res;
    }
};
```

### 2.3 KM 最大权完美匹配 `O(n^3)`（左部点数 = 右部点数）

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

// 最大权完美匹配（要求 n <= m，且存在完美匹配）
// w[i][j] 为左 i 到右 j 的权值；求最大权
struct KM {
    int n;
    vector<vector<ll>> w;
    vector<ll> lx, ly, slack;
    vector<int> matchY, pre;
    vector<bool> vx, vy;

    explicit KM(int n_) : n(n_), w(n_ + 1, vector<ll>(n_ + 1, 0)),
        lx(n_ + 1), ly(n_ + 1), slack(n_ + 1), matchY(n_ + 1, 0),
        pre(n_ + 1, 0), vx(n_ + 1), vy(n_ + 1) {}

    void setWeight(int i, int j, ll val) { w[i][j] = val; }

    void augment(int u) {
        fill(vx.begin(), vx.end(), false);
        fill(vy.begin(), vy.end(), false);
        fill(slack.begin(), slack.end(), LLONG_MAX / 4);
        int y = 0;
        matchY[0] = u;
        do {
            vx[y] = true;
            int x = matchY[y], nxt = 0;
            ll delta = LLONG_MAX / 4;
            for (int j = 1; j <= n; ++j) if (!vy[j]) {
                ll cur = lx[x] + ly[j] - w[x][j];
                if (cur < slack[j]) { slack[j] = cur; pre[j] = y; }
                if (slack[j] < delta) { delta = slack[j]; nxt = j; }
            }
            for (int j = 0; j <= n; ++j) {
                if (vy[j]) { lx[matchY[j]] -= delta; ly[j] += delta; }
                else slack[j] -= delta;
            }
            y = nxt;
        } while (matchY[y]);
        while (y) { int py = pre[y]; matchY[y] = matchY[py]; y = py; }
    }
    ll run() {                            // 返回最大权值
        for (int i = 1; i <= n; ++i) {
            lx[i] = *max_element(w[i].begin() + 1, w[i].end());
            ly[i] = 0;
        }
        for (int i = 1; i <= n; ++i) augment(i);
        ll res = 0;
        for (int j = 1; j <= n; ++j) if (matchY[j]) res += w[matchY[j]][j];
        return res;
    }
};
```

> **最小权匹配**：把权值取反（或 `C - w`）转成最大权。

## 3. 最小费用最大流（MCMF）

### 3.1 SPFA 版（能处理负权，但慢）

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
constexpr ll LINF = 0x3f3f3f3f3f3f3f3fLL;

struct MCMF {
    struct Edge { int to, rev; ll cap, cost; };
    int n;
    vector<vector<Edge>> g;
    vector<ll> dis;
    vector<int> preV, preE, inq;

    explicit MCMF(int n_) : n(n_), g(n_ + 1), dis(n_ + 1), preV(n_ + 1),
        preE(n_ + 1), inq(n_ + 1) {}

    void addEdge(int u, int v, ll cap, ll cost) {
        g[u].push_back({v, (int)g[v].size(), cap, cost});
        g[v].push_back({u, (int)g[u].size() - 1, 0, -cost});
    }
    void addBiEdge(int u, int v, ll cap, ll cost) {   // 无向边
        g[u].push_back({v, (int)g[v].size(), cap, cost});
        g[v].push_back({u, (int)g[u].size() - 1, cap, cost});
    }

    // 返回 {最大流, 最小费用}
    pair<ll,ll> run(int s, int t) {
        ll flow = 0, cost = 0;
        while (true) {
            fill(dis.begin(), dis.end(), LINF);
            fill(inq.begin(), inq.end(), 0);
            dis[s] = 0;
            queue<int> q; q.push(s); inq[s] = 1;
            while (!q.empty()) {
                int u = q.front(); q.pop(); inq[u] = 0;
                for (int i = 0; i < (int)g[u].size(); ++i) {
                    auto &e = g[u][i];
                    if (e.cap > 0 && dis[e.to] > dis[u] + e.cost) {
                        dis[e.to] = dis[u] + e.cost;
                        preV[e.to] = u; preE[e.to] = i;
                        if (!inq[e.to]) { q.push(e.to); inq[e.to] = 1; }
                    }
                }
            }
            if (dis[t] == LINF) break;
            ll f = LINF;
            for (int v = t; v != s; v = preV[v]) f = min(f, g[preV[v]][preE[v]].cap);
            for (int v = t; v != s; v = preV[v]) {
                g[preV[v]][preE[v]].cap -= f;
                g[v][g[preV[v]][preE[v]].rev].cap += f;
            }
            flow += f; cost += f * dis[t];
        }
        return {flow, cost};
    }
};
```

### 3.2 势能 Dijkstra 版（无负权时更快）

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
constexpr ll LINF = 0x3f3f3f3f3f3f3f3fLL;

struct MCMF_Dij {
    struct Edge { int to, rev; ll cap, cost; };
    int n;
    vector<vector<Edge>> g;
    vector<ll> h, dis;                    // h: 势能
    vector<int> preV, preE;

    explicit MCMF_Dij(int n_) : n(n_), g(n_ + 1), h(n_ + 1, 0), dis(n_ + 1),
        preV(n_ + 1), preE(n_ + 1) {}

    void addEdge(int u, int v, ll cap, ll cost) {
        g[u].push_back({v, (int)g[v].size(), cap, cost});
        g[v].push_back({u, (int)g[u].size() - 1, 0, -cost});
    }

    pair<ll,ll> run(int s, int t) {
        ll flow = 0, cost = 0;
        while (true) {
            fill(dis.begin(), dis.end(), LINF);
            dis[s] = 0;
            priority_queue<pair<ll,int>, vector<pair<ll,int>>, greater<>> pq;
            pq.push({0, s});
            while (!pq.empty()) {
                auto [d, u] = pq.top(); pq.pop();
                if (d > dis[u]) continue;
                for (int i = 0; i < (int)g[u].size(); ++i) {
                    auto &e = g[u][i];
                    if (e.cap <= 0) continue;
                    ll nd = d + e.cost + h[u] - h[e.to];
                    if (nd < dis[e.to]) {
                        dis[e.to] = nd;
                        preV[e.to] = u; preE[e.to] = i;
                        pq.push({nd, e.to});
                    }
                }
            }
            if (dis[t] == LINF) break;
            for (int i = 1; i <= n; ++i) if (dis[i] < LINF) h[i] += dis[i];
            ll f = LINF;
            for (int v = t; v != s; v = preV[v]) f = min(f, g[preV[v]][preE[v]].cap);
            for (int v = t; v != s; v = preV[v]) {
                g[preV[v]][preE[v]].cap -= f;
                g[v][g[preV[v]][preE[v]].rev].cap += f;
            }
            flow += f; cost += f * (h[t] - h[s]);
        }
        return {flow, cost};
    }
};
```

> **要求初始无负权边**（否则先用 SPFA 求初始势能）。`h` 的更新是 `h[i] += dis[i]`。

## 4. 上下界网络流（了解）

| 类型 | 建图 |
|---|---|
| 无源汇上下界可行流 | 每条边 `(u,v,[lo,hi])` 拆成 `u->v` 容量 `hi-lo`，记录 `d[u]-=lo, d[v]+=lo`；对 `d>0` 连 `SS`，`d<0` 连 `TT`，跑最大流看是否满流 |
| 有源汇上下界可行流 | 加边 `t->s` 容量 `INF`，转成无源汇 |
| 有源汇上下界最大流 | 先可行流，再删 `t->s` 边跑一次 `s->t` 最大流 |
| 有源汇上下界最小流 | 先可行流，再跑 `t->s` 最大流（反向增广） |

## 6. 一般图最大匹配（带花树）

**用途**：一般图（非二分图）的最大匹配。**只有题目明确是"一般图"或"奇环"时才需要**，否则用匈牙利/HK 即可。

```cpp
#include <bits/stdc++.h>
using namespace std;

// 一般图最大匹配（带花树），1-based，n <= 500 左右
struct Blossom {
    int n, tim = 0;
    vector<vector<int>> g;
    vector<int> match, fa, pre, vis, dfn;
    queue<int> q;

    explicit Blossom(int n_) : n(n_), g(n_ + 1), match(n_ + 1, 0), fa(n_ + 1),
        pre(n_ + 1, 0), vis(n_ + 1, 0), dfn(n_ + 1, 0) {}

    void addEdge(int u, int v) { g[u].push_back(v); g[v].push_back(u); }

    int find(int x) { return fa[x] == x ? x : fa[x] = find(fa[x]); }
    int lca(int u, int v) {
        ++tim;
        while (true) {
            u = find(u);
            if (dfn[u] == tim) return u;
            dfn[u] = tim;
            if (!match[u]) break;
            u = pre[match[u]];
        }
        while (true) {
            v = find(v);
            if (dfn[v] == tim) return v;
            dfn[v] = tim;
            if (!match[v]) break;
            v = pre[match[v]];
        }
        return 0;
    }
    void shrink(int u, int v, int rt) {
        while (find(u) != rt) {
            pre[u] = v;
            v = match[u];
            if (vis[v] == 2) { vis[v] = 1; q.push(v); }
            if (find(u) == u) fa[u] = rt;
            if (find(v) == v) fa[v] = rt;
            u = pre[v];
        }
    }
    int bfs(int s) {
        for (int i = 1; i <= n; ++i) { fa[i] = i; vis[i] = 0; pre[i] = 0; }
        while (!q.empty()) q.pop();
        q.push(s); vis[s] = 1;
        while (!q.empty()) {
            int u = q.front(); q.pop();
            for (int v : g[u]) {
                if (find(v) == find(u)) continue;
                if (vis[v] == 2) continue;
                if (vis[v] == 1) {
                    int rt = lca(u, v);
                    if (rt) shrink(u, v, rt), shrink(v, u, rt);
                } else {
                    vis[v] = 2;
                    pre[v] = u;
                    if (!match[v]) {              // 找到增广路
                        int x = v;
                        while (x) {
                            int y = pre[x], z = match[y];
                            match[x] = y; match[y] = x;
                            x = z;
                        }
                        return 1;
                    }
                    vis[match[v]] = 1;
                    q.push(match[v]);
                }
            }
        }
        return 0;
    }
    int run() {
        int res = 0;
        for (int i = 1; i <= n; ++i) if (!match[i]) res += bfs(i);
        return res;
    }
};
```

> 这份是标准写法（`vis` 1=黑/外，2=白/内）。`bfs(i)` 返回是否找到增广路。总复杂度 `O(n^3)`。

### 怎么用（照抄这段 `main`，只改【输入】【输出】两处）

上面是模板本体。实际写题时：**把上面的 struct 整段粘过来 → 再粘这段 `main` → 只改标了【输入】【输出】的行**。

```cpp skip
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

// ===== 粘贴点：把上面的 Dinic 粘到这里 =====

int main() {
    // ===================== 【输入】改这里 =====================
    int n, m, S, T;
    scanf("%d %d %d %d", &n, &m, &S, &T);   // 点数 边数 源点 汇点
    // ========================================================

    // ===================== 【核心】算法流程 =====================
    // Dinic 最大流：
    //   1) BFS 给残量网络分层，层数 = 从源点走几条边
    //   2) DFS 只沿"层数 +1"的边走，一次找出多条增广路（多路增广）
    //   3) 当前弧优化：走过的边不再回头看
    //   4) 重复 1~3 直到汇点不可达。O(n^2 m)，实际远快
    Dinic din(n);
    for (int i = 0; i < m; ++i) {
        int u, v; ll cap;
        scanf("%d %d %lld", &u, &v, &cap);
        din.addEdge(u, v, cap);          // 无向图改用 addBiEdge
    }
    ll flow = din.maxFlow(S, T);
    // ========================================================

    // ===================== 【输出】改这里 =====================
    printf("%lld\n", flow);
    // ========================================================
    return 0;
}
```
