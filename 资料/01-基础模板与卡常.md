**`#define int long long` 的坑**：会连锁改变 `pair<int,int>`、`map<int,...>`、`min/max` 的类型，容易在模板里炸开，而且 `-O2` 下 long long 运算比 int 慢约 1.5~2 倍。
**推荐做法**：不要 define，只在真正需要的地方把变量声明成 `ll`；写完程序后用 `g++ -m32`？不，用「把所有 int 换成 ll 再交一遍」的老办法检查溢出。

## 2. 快读快写（含 EOF 版本）

```cpp
#include <bits/stdc++.h>
using namespace std;

namespace FastIO {
    static const int BUFSZ = 1 << 26;
    char ibuf[BUFSZ], obuf[BUFSZ];
    int ipos = 0, opos = 0;
    inline int gc() {
        if (ipos == BUFSZ) { ipos = 0; fread(ibuf, 1, BUFSZ, stdin); }
        return ipos < BUFSZ ? ibuf[ipos++] : EOF;
    }
    inline bool readChar(char &c) {
        int ch = gc();
        while (ch != EOF && (ch == ' ' || ch == '\n' || ch == '\r' || ch == '\t')) ch = gc();
        if (ch == EOF) return false;
        c = (char)ch;
        return true;
    }
    template <class T> inline bool readInt(T &out) {
        int ch = gc(), sign = 0;
        T num = 0;
        while (ch != EOF && ch != '-' && (ch < '0' || ch > '9')) ch = gc();
        if (ch == EOF) return false;
        if (ch == '-') { sign = 1; ch = gc(); }
        for (; ch >= '0' && ch <= '9'; ch = gc()) num = num * 10 + (ch - '0');
        out = sign ? -num : num;
        return true;
    }
    inline void pushChar(char c) {
        if (opos == BUFSZ) { fwrite(obuf, 1, BUFSZ, stdout); opos = 0; }
        obuf[opos++] = c;
    }
    inline void writeInt(long long x) {
        if (x < 0) { pushChar('-'); x = -x; }
        char s[24]; int n = 0;
        do { s[n++] = char('0' + x % 10); x /= 10; } while (x);
        while (n) pushChar(s[--n]);
    }
    inline void writeStr(const char *s) { while (*s) pushChar(*s++); }
    struct Flusher {
        ~Flusher() { if (opos) fwrite(obuf, 1, opos, stdout), opos = 0; }
    } flusher_;
}
using FastIO::readInt;
using FastIO::writeInt;
```

用法（注意读入返回 bool，便于 `while (readInt(n))` 处理多组）：

```cpp
#include <bits/stdc++.h>
using namespace std;
namespace io {
    static char buf[1 << 26];
    static int pos = 0, len = 0;
    inline int gc() {
        if (pos == len) {
            len = (int)fread(buf, 1, 1 << 26, stdin);
            pos = 0;
        }
        return pos == len ? -1 : buf[pos++];
    }
    template <class T> inline bool rd(T &x) {
        int c = gc(), f = 0; x = 0;
        while (c != -1 && c != '-' && (c < '0' || c > '9')) c = gc();
        if (c == -1) return false;
        if (c == '-') { f = 1; c = gc(); }
        for (; c >= '0' && c <= '9'; c = gc()) x = x * 10 + (c - '0');
        if (f) x = -x;
        return true;
    }
    template <class T> inline void wt(T x) {
        if (x < 0) { putchar('-'); x = -x; }
        if (x > 9) wt(x / 10);
        putchar('0' + (int)(x % 10));
    }
}
int main() {
    int n;
    while (io::rd(n)) { io::wt(n); putchar('\n'); }
    return 0;
}
```

> 第二种写法里 `putchar_unlocked ? 0 : 0;` 只是提醒：Linux 可用 `putchar_unlocked`，但为了跨平台别用。真正要极致卡常就用第一种 `fwrite` 缓冲。

## 3. `__int128` 的输入输出与乘法防溢出

```cpp
#include <bits/stdc++.h>
using namespace std;
using i128 = __int128_t;
using u128 = __uint128_t;

istream &operator>>(istream &is, i128 &x) {
    string s; is >> s;
    int i = 0, f = 1; x = 0;
    if (s[0] == '-') { f = -1; i = 1; }
    for (; i < (int)s.size(); ++i) x = x * 10 + (s[i] - '0');
    x *= f;
    return is;
}
ostream &operator<<(ostream &os, i128 x) {
    if (x == 0) return os << '0';
    if (x < 0) { os << '-'; x = -x; }
    string s;
    while (x) { s += char('0' + (int)(x % 10)); x /= 10; }
    reverse(s.begin(), s.end());
    return os << s;
}

// 防溢出乘法：求 a*b % m，m 可到 1e18
inline long long mul_mod(long long a, long long b, long long m) {
    return (long long)((i128)a * b % m);
}
```

## 5. 卡常技巧清单（按收益排序）

| 技巧 | 说明 | 典型收益 |
|---|---|---|
| `#pragma GCC optimize("O2")` | 本地调试用 O2；**交题时评测机自带 O2，不必写**，写了有时反而变慢 | — |
| `#pragma GCC target("avx2")` | 会炸非 AVX2 机器，**别交** | 危险 |
| 快读快写 | 输入 1e7 个数时省 0.5s 以上 | 大 |
| 循环内避免 `%` | 用 `if (x >= MOD) x -= MOD` 代替取模 | 大 |
| 减少 `vector` 动态分配 | 用 `static` 数组 / `reserve` / 全局数组 | 大 |
| `struct` 传引用 | `f(const vector<int>& v)` 而非 `f(vector<int> v)` | 中 |
| 位运算代替判断 | `x & 1`、`x >> 1`、用 `mask` 存状态 | 中 |
| 把常量提到循环外 | `const int N = v.size();` | 中 |
| `inline` 小函数 | 递归函数别乱加 | 小 |
| 能用 `int` 就别用 `long long` | 见上 | 大 |
| 二维数组按行访问 | 缓存友好；`a[i][j]` 内层是 j | 大 |
| 避免 `endl` | 用 `'\n'` | 大（输出多时） |
| `ios::sync_with_stdio(false); cin.tie(nullptr);` | 用 cin/cout 时必写 | 大 |
| 手写栈替代递归 | 树剖 DFS 卡栈时 | 中 |
| `__builtin` 系列 | popcount / clz / ctz | 小 |
| 循环展开 / 去掉分支 | 只在最后 TLE 时做 | 小 |

**卡常顺序建议**：先检查算法复杂度 → 再看常数（有没有 log 套 log、有没有 map/set 大量操作、有没有 string 拼接）→ 再用快读 → 最后才是 pragma。

## 6. 调试宏（比赛也能留，不影响正确性）

```cpp
#include <bits/stdc++.h>
using namespace std;

#ifdef LOCAL
#define dbg(x) cerr << #x << " = " << (x) << endl
#define dbgv(v) do { \
        cerr << #v << " = ["; \
        for (size_t i = 0; i < (v).size(); ++i) \
            cerr << (i ? ", " : "") << (v)[i]; \
        cerr << "]\n"; \
    } while (0)
#else
#define dbg(x) ((void)0)
#define dbgv(v) ((void)0)
#endif

int main() {
    vector<int> a = {1, 2, 3};
    dbg(a.size());
    dbgv(a);
    return 0;
}
```

> 交题时**不要**定义 `LOCAL`，这样宏展开成空，安全。若你嫌麻烦，交题前删掉整个 `#ifdef` 块即可。

## 7. 常用 STL 速查（比赛里最容易想不起来的）

```cpp
#include <bits/stdc++.h>
using namespace std;
int main() {
    // 优先队列小根堆
    priority_queue<int, vector<int>, greater<int>> pq;
    pq.push(3); pq.push(1);
    int top = pq.top();          // 1
    // 自定义比较（注意是"小于"语义的反向：返回 true 表示 a 排在 b 后面）
    auto cmp = [](const pair<int,int> &a, const pair<int,int> &b) {
        return a.second > b.second;
    };
    priority_queue<pair<int,int>, vector<pair<int,int>>, decltype(cmp)> pq2(cmp);

    // set 找前驱后继
    set<int> s = {1, 5, 9};
    auto it = s.lower_bound(5);            // >= 5
    int pre = (it == s.begin()) ? -1 : *prev(it);
    int nxt = (it == s.end()) ? -1 : *it;

    // map 遍历 + 删除
    map<string, int> mp{{"a", 1}, {"b", 2}};
    for (auto &[k, v] : mp) v++;
    for (auto it2 = mp.begin(); it2 != mp.end(); ) {
        if (it2->second > 1) it2 = mp.erase(it2);
        else it2 = next(it2);
    }

    // 二分
    vector<int> v = {1, 2, 2, 3};
    int l = int(lower_bound(v.begin(), v.end(), 2) - v.begin());  // 1
    int r = int(upper_bound(v.begin(), v.end(), 2) - v.begin());  // 3
    bool has = binary_search(v.begin(), v.end(), 3);                // true

    // 全排列 / 下一排列
    vector<int> p = {1, 2, 3};
    do { } while (next_permutation(p.begin(), p.end()));

    // 累积 / 局部最小最大
    long long sum = accumulate(v.begin(), v.end(), 0LL);
    auto [mn, mx] = minmax_element(v.begin(), v.end());

    // nth_element 求第 k 小（0-indexed）
    nth_element(v.begin(), v.begin() + 1, v.end());

    // 按值删除
    v.erase(remove(v.begin(), v.end(), 2), v.end());
    return 0;
}
```

**C++17/20/23 里能用的新东西（很少会 Ban）**

- C++17：结构化绑定 `auto [a,b] = ...`、`if constexpr`、`std::optional`、`std::string_view`、`std::gcd/lcm`、折叠表达式。
- C++20：`<=>` 三路比较、`std::ranges`（`sort(v)` 直接传容器）、`std::midpoint`、`std::lerp`、`std::bit_width`。
- C++23：`std::ranges::to`、`std::expected`、多维 `operator[]`（`mdspan`）。
- **稳妥起见**：交题用 C++17，评测机一般就是 17；上面这些新特性只在确定支持时用。
