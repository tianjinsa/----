# 22 高级数据结构（主席树 / 李超树 / 左偏树 / 线段树合并 / ODT）

> 这些结构"码量大但含金量高"，都是**难替代**的：主席树做可持久化、李超树做最值函数、左偏树做可并堆、线段树合并做子树信息、ODT 做区间推平。
> **不收录**：红黑树、跳表（`map/set` 能替代，且几乎不考）。

---

## 1. 可持久化线段树 / 主席树

### 1.1 用途

- **静态区间第 k 小**（P3834）：`O((n+m) log n)`。
- 可持久化数组（P3919）：单点修改保留历史版本。
- 区间内不同数的个数、区间 mex、树上路径第 k 小。

### 1.2 原理

每个版本只新建 `O(log n)` 个节点，其余节点**共享**上一版本的。

```
root[i] 指向"前 i 个数"的权值线段树
区间 [l, r] 的第 k 小 = 在 root[r] 与 root[l-1] 的差值树上二分
```

```cpp
#include <bits/stdc++.h>
using namespace std;

// 静态区间第 k 小
struct ChairmanTree {
    struct Node { int ls, rs, cnt; };
    vector<Node> tr;
    vector<int> root;
    int n, m, tot = 0;

    explicit ChairmanTree(int n_, int estimatedNodes) : n(n_), tr() {
        tr.reserve(estimatedNodes);
        tr.push_back({0, 0, 0});                 // 0 号空节点
        root.assign(n_ + 1, 0);
    }

    int update(int prev, int l, int r, int pos) {
        int cur = (int)tr.size();
        tr.push_back(tr[prev]);                  // 复制一份
        ++tr[cur].cnt;
        if (l == r) return cur;
        int mid = (l + r) >> 1;
        if (pos <= mid) tr[cur].ls = update(tr[prev].ls, l, mid, pos);
        else tr[cur].rs = update(tr[prev].rs, mid + 1, r, pos);
        return cur;
    }

    // 查询 [ql, qr] 中第 k 小（1-based 排名）
    int query(int u, int v, int l, int r, int k) const {
        if (l == r) return l;
        int mid = (l + r) >> 1;
        int leftCnt = tr[tr[v].ls].cnt - tr[tr[u].ls].cnt;
        if (k <= leftCnt) return query(tr[u].ls, tr[v].ls, l, mid, k);
        return query(tr[u].rs, tr[v].rs, mid + 1, r, k - leftCnt);
    }

    // 建树：a[1..n] 已离散化成 [1, m]
    void build(const vector<int> &a, int m_) {
        m = m_;
        for (int i = 1; i <= n; ++i) root[i] = update(root[i - 1], 1, m, a[i]);
    }

    // 区间 [l, r] 第 k 小（返回离散化后的值）
    int kth(int l, int r, int k) const { return query(root[l - 1], root[r], 1, m, k); }
};
```

> **节点数**：`n * (log2(n) + 1) + 1`，`n = 2e5` 时约 `4e6`，用 `vector` 存要注意内存（`3 * int` 每节点 = 12 字节 → 约 48MB）。

### 1.3 可持久化数组（P3919）

```cpp
#include <bits/stdc++.h>
using namespace std;

// 每次修改生成新版本，旧版本不变
struct PersistentArray {
    struct Node { int ls, rs, val; };
    vector<Node> tr;
    vector<int> root;
    int n;

    PersistentArray(int n_) : n(n_) {
        tr.push_back({0, 0, 0});
        root.push_back(0);
    }
    int build(int l, int r, const vector<int> &a) {
        int cur = (int)tr.size();
        tr.push_back({0, 0, 0});
        if (l == r) { tr[cur].val = a[l]; return cur; }
        int mid = (l + r) >> 1;
        tr[cur].ls = build(l, mid, a);
        tr[cur].rs = build(mid + 1, r, a);
        return cur;
    }
    int modify(int prev, int l, int r, int pos, int v) {
        int cur = (int)tr.size();
        tr.push_back(tr[prev]);
        if (l == r) { tr[cur].val = v; return cur; }
        int mid = (l + r) >> 1;
        if (pos <= mid) tr[cur].ls = modify(tr[prev].ls, l, mid, pos, v);
        else tr[cur].rs = modify(tr[prev].rs, mid + 1, r, pos, v);
        return cur;
    }
    int query(int p, int l, int r, int pos) const {
        if (l == r) return tr[p].val;
        int mid = (l + r) >> 1;
        if (pos <= mid) return query(tr[p].ls, l, mid, pos);
        return query(tr[p].rs, mid + 1, r, pos);
    }
};
```

### 1.4 树上路径第 k 小（P2633）

`root[u]` = `root[fa[u]]` 加上 `val[u]`；查询 `u-v` 路径时用
`cnt = tr[a].cnt + tr[b].cnt - tr[lca].cnt - tr[fa[lca]].cnt`。

## 2. 李超树（Li Chao Tree）

### 2.1 用途

- **插入直线/线段，查询某个 `x` 处的最大值/最小值**，`O(log^2 n)` 或 `O(log n)`。
- 典型题：P4097【模板】李超线段树、P4254 [JSOI2008] Blue Mary 开公司、P4069 [SDOI2016] 游戏。

### 2.2 原理

每个节点存一条"在区间中点处最优"的直线。插入新直线时，与节点直线在中点处比较，保留更好的，把另一条递归到对应半边。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ld = long double;

// 求最大值版本
struct LiChaoMax {
    struct Line {
        ld k = -1e30, b = -1e30;             // y = kx + b
        ld get(ld x) const { return k * x + b; }
    };
    struct Node { Line ln; };

    int n;                                    // 离散化后的 x 坐标个数
    vector<ld> xs;                            // 所有查询的 x 坐标（升序）
    vector<Node> tr;

    explicit LiChaoMax(const vector<ld> &coords) : xs(coords) {
        sort(xs.begin(), xs.end());
        xs.erase(unique(xs.begin(), xs.end()), xs.end());
        n = (int)xs.size();
        tr.assign(4 * n + 4, Node{});
    }

    void insert(int p, int l, int r, Line cur) {
        int mid = (l + r) >> 1;
        bool lef = cur.get(xs[l]) > tr[p].ln.get(xs[l]);
        bool mdf = cur.get(xs[mid]) > tr[p].ln.get(xs[mid]);
        if (mdf) swap(tr[p].ln, cur);          // 保留中点更优的
        if (l == r) return;
        if (lef != mdf) insert(p << 1, l, mid, cur);
        else insert(p << 1 | 1, mid + 1, r, cur);
    }

    ld query(int p, int l, int r, int idx) const {
        ld res = tr[p].ln.get(xs[idx]);
        if (l == r) return res;
        int mid = (l + r) >> 1;
        if (idx <= mid) return max(res, query(p << 1, l, mid, idx));
        return max(res, query(p << 1 | 1, mid + 1, r, idx));
    }

    // 对外：加入直线 y = kx + b
    void add_line(ld k, ld b) { insert(1, 0, n - 1, Line{k, b}); }
    // 对外：查询 x = xs[idx] 的最大值
    ld query(int idx) const { return query(1, 0, n - 1, idx); }
};
```

**要点**：
- **必须离散化 `x`**（李超树在离散坐标上建）。
- 求最小值：把 `>` 改成 `<`，初值改成 `1e30`。
- 插入**线段**（限定 `x` 范围）：先定位到 `O(log n)` 个区间再插入。

**实例（P4254 开公司）**：每条收益是 `(p, q)` → 直线 `y = p*x + q`，查询某时刻最大收益。直接李超树。

## 3. 左偏树 / 可并堆（本场 F 题用到）

### 3.1 用途

- **合并两个堆**、**取堆顶**、**弹出堆顶**，均摊 `O(log n)`。
- 典型题：P3377 左偏树模板、P2713 罗马游戏、**本场 F 题（折叠游戏）**。

### 3.2 原理

每个节点维护 `dist`（到最近空节点的距离），保证**左儿子的 `dist` >= 右儿子的**（左偏）。
合并时只沿右链递归，深度 `O(log n)`。

```cpp
#include <bits/stdc++.h>
using namespace std;

// 大根左偏树（按 key 比较），下标 1..n
struct LeftistHeap {
    vector<int> lc, rc, dist, val, fa;
    int n;

    explicit LeftistHeap(int n_) : lc(n_ + 1, 0), rc(n_ + 1, 0), dist(n_ + 1, 0),
        val(n_ + 1, 0), fa(n_ + 1, 0) { n = n_; }

    void set_value(int u, int v) { val[u] = v; }
    bool greater_key(int a, int b) const { return val[a] > val[b]; }   // 大根

    // 合并 a、b 两棵堆，返回新根
    int merge(int a, int b) {
        if (!a || !b) return a + b;
        if (greater_key(b, a)) swap(a, b);           // 让 a 是较大值
        rc[a] = merge(rc[a], b);
        fa[rc[a]] = a;
        if (dist[lc[a]] < dist[rc[a]]) swap(lc[a], rc[a]);
        dist[a] = dist[rc[a]] + 1;
        return a;
    }
    // 弹出根，返回新根
    int pop(int a) {
        fa[lc[a]] = fa[rc[a]] = 0;
        int r = merge(lc[a], rc[a]);
        lc[a] = rc[a] = 0;
        return r;
    }
    int top(int a) const { return val[a]; }
};
```

> `fa[]` 只用于"找某点所在堆的根"（配合路径压缩），如果不需要可以删。

### 3.3 F 题（折叠游戏）怎么用它

> **题目核心**：每次折叠把直径上对称的两点合并，需要快速"找出当前树从固定端点的最长路径"。
> **做法**：
> 1. 固定一个直径端点 `root`，则从 `root` 出发的最长路就是当前直径。
> 2. 每个点 `u` 维护"孩子高度"的**大根堆**（左偏树），堆顶就是最该走的下一步。
> 3. 每次折叠：沿堆顶不断 `pop` 出整条直径 → 把对称点的堆合并 → 只重建保留的一半。
> 4. 复杂度 `O(n log n)`。

```cpp
// 关键片段（伪代码级，展示数据结构的角色）
// head[u] : u 的孩子高度堆（左偏树根）
// height[u] : u 的子树高度
// 每轮：
//   1) 从 root 反复 pop 出直径 path
//   2) 对 i<=D/2: head[path[i]] = merge(head[path[i]], head[path[D-i]])
//   3) 从中间往根重建 height，并把 path[i] 加回 head[path[i-1]]
```

## 4. 线段树合并

### 4.1 用途

- 树上每个点有一个"值域线段树"，求**子树内信息的合并**。
- 典型题：P4556 [Vani 有约会] 雨天的尾巴、P3224 永无乡、CF 600E Lomsat gelral。

### 4.2 原理

合并两棵动态开点线段树：若有一棵为空，直接返回另一棵；否则递归合并左右儿子。
总复杂度 `O(总节点数)`（势能分析）。

```cpp
#include <bits/stdc++.h>
using namespace std;

// 动态开点线段树 + 合并
struct SegMerge {
    struct Node { int ls = 0, rs = 0, mx = 0, pos = 0; };
    vector<Node> tr;
    int L, R;

    SegMerge(int l, int r, int maxNodes) : L(l), R(r) {
        tr.reserve(maxNodes);
        tr.push_back(Node{});
    }
    int newNode() { tr.push_back(Node{}); return (int)tr.size() - 1; }

    void add(int &p, int l, int r, int pos, int v) {
        if (!p) p = newNode();
        if (l == r) { tr[p].mx += v; tr[p].pos = l; return; }
        int m = (l + r) >> 1;
        if (pos <= m) add(tr[p].ls, l, m, pos, v);
        else add(tr[p].rs, m + 1, r, pos, v);
        push_up(p);
    }
    void push_up(int p) {
        int a = tr[p].ls, b = tr[p].rs;
        if (tr[a].mx >= tr[b].mx) { tr[p].mx = tr[a].mx; tr[p].pos = tr[a].pos; }
        else { tr[p].mx = tr[b].mx; tr[p].pos = tr[b].pos; }
    }
    // 合并 q 到 p，返回合并后的根
    int merge(int p, int q, int l, int r) {
        if (!p || !q) return p + q;
        if (l == r) { tr[p].mx += tr[q].mx; return p; }
        int m = (l + r) >> 1;
        tr[p].ls = merge(tr[p].ls, tr[q].ls, l, m);
        tr[p].rs = merge(tr[p].rs, tr[q].rs, m + 1, r);
        push_up(p);
        return p;
    }
};
```

**实例（P4556 雨天的尾巴）**：树上差分 + 线段树合并，求每个点子树内出现最多的颜色。

## 5. ODT（Chtholly Tree / 区间推平）

### 5.1 用途

- **区间赋值**操作多、查询随机的题目（数据随机时极快）。
- 典型题：CF 896C Willem, Chtholly and Seniorious、P2572 [SCOI2010] 序列操作。
- **注意**：只有"推平操作多"才能用，否则会退化成 `O(n^2)`。

### 5.2 实现

用 `set` 维护若干段 `[l, r]` 及其值；每次操作把区间 `split` 成边界对齐的段再处理。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

struct ODT {
    struct Node {
        int l, r;
        mutable ll v;                          // mutable 允许在 set 中修改
        bool operator<(const Node &o) const { return l < o.l; }
    };
    set<Node> s;

    explicit ODT(int n, ll init = 0) { s.insert({1, n, init}); }

    // 把 pos 切成两段，返回以 pos 开头的迭代器
    set<Node>::iterator split(int pos) {
        auto it = s.lower_bound({pos, 0, 0});
        if (it != s.end() && it->l == pos) return it;
        --it;
        if (it->r < pos) return s.end();
        int l = it->l, r = it->r;
        ll v = it->v;
        s.erase(it);
        s.insert({l, pos - 1, v});
        return s.insert({pos, r, v}).first;
    }

    // 区间赋值
    void assign(int l, int r, ll v) {
        auto itr = split(r + 1), itl = split(l);
        s.erase(itl, itr);
        s.insert({l, r, v});
    }

    // 区间加
    void add(int l, int r, ll v) {
        auto itr = split(r + 1), itl = split(l);
        for (auto it = itl; it != itr; ++it) it->v += v;
    }

    // 区间第 k 小
    ll kth(int l, int r, int k) {
        auto itr = split(r + 1), itl = split(l);
        vector<pair<ll,int>> v;
        for (auto it = itl; it != itr; ++it) v.push_back({it->v, it->r - it->l + 1});
        sort(v.begin(), v.end());
        for (auto [val, cnt] : v) {
            if (k <= cnt) return val;
            k -= cnt;
        }
        return -1;
    }
};
```

**注意**：`split(r + 1)` 要**先算**，因为 `split(l)` 可能让 `itr` 失效。

## 6. 树套树（BIT 套线段树 / 主席树套树）

**用途**：动态区间第 k 小、带修改的二维数点。
**实现**：树状数组的每个节点挂一棵动态开点权值线段树。完整代码较长，通常用**整体二分**或 **CDQ** 代替（见 25、14 号文件）。

**选择建议**：

| 需求 | 推荐 |
|---|---|
| 静态区间第 k 小 | 主席树 |
| 动态区间第 k 小 | 整体二分 / 树套树 |
| 子树信息合并 | 线段树合并 |
| 可并堆 | 左偏树 |
| 最值直线查询 | 李超树 |
| 区间推平 | ODT（数据随机） |

## 7. 内存与常数提醒

| 结构 | 节点数估计 | 备注 |
|---|---|---|
| 主席树 | `n log n` | `2e5` → `4e6` 节点 |
| 线段树合并 | `总插入次数 * log` | 每个操作 `O(log)` |
| 李超树 | `4 * 离散点数` | 静态数组即可 |
| 左偏树 | `n` | 每点一个节点 |
| ODT | 段数 | 推平后段数变少 |

**通用技巧**：用 `vector<Node>` + `reserve` 代替 `new`，避免指针慢。


### 怎么用（照抄这段 `main`，只改【输入】【输出】两处）

上面是模板本体。实际写题时：**把上面的 struct 整段粘过来 → 再粘这段 `main` → 只改标了【输入】【输出】的行**。

```cpp skip
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

// ===== 粘贴点：把上面的 ChairmanTree 粘到这里 =====

int main() {
    // ===================== 【输入】改这里 =====================
    int n, q;
    scanf("%d %d", &n, &q);               // n 个数，q 次询问
    vector<int> a(n + 1);
    for (int i = 1; i <= n; ++i) scanf("%d", &a[i]);
    // ========================================================

    // ===================== 【核心】算法流程 =====================
    // 主席树 = 可持久化权值线段树：
    //   1) 离散化，值域压到 [1, m]
    //   2) root[i] = 把 a[i] 插入 root[i-1] 后得到的新根（只新建 O(log n) 个点）
    //   3) 询问 [l,r] 第 k 小：在 root[r] 和 root[l-1] 上同时二分，
    //      左子树个数 = cnt[r.left] - cnt[l-1.left]，够 k 就往左，否则 k 减掉往右
    vector<int> v = a;
    sort(v.begin() + 1, v.end());
    v.erase(unique(v.begin() + 1, v.end()), v.end());
    int m = (int)v.size() - 1;
    for (int i = 1; i <= n; ++i)
        a[i] = int(lower_bound(v.begin() + 1, v.end(), a[i]) - v.begin());

    ChairmanTree ct(n, 4 * n * 20);       // 节点数 ≈ n log n
    ct.build(a, m);

    while (q--) {
        int l, r, k;
        scanf("%d %d %d", &l, &r, &k);
        // ===================== 【输出】改这里 =====================
        printf("%d\n", v[ct.kth(l, r, k)]);   // kth 返回离散化下标，要映射回原值
        // ========================================================
    }
    return 0;
}
```

## 8. 思路与例题索引

| 算法 | 思路一句话 | 用途 | 例题 |
|---|---|---|---|
| 主席树 | 每个版本只新建 `O(log n)` 节点 | 静态区间第 k 小 | P3834【模板】可持久化线段树 2 |
| 可持久化数组 | 修改复制路径 | 历史版本查询 | P3919【模板】可持久化数组 |
| 树上主席树 | `root[u]` 由父亲扩展 | 树上路径第 k 小 | P2633 Count on a tree |
| 李超树 | 每个节点存"中点最优"的直线 | 插入直线查最值 | P4097【模板】李超线段树 |
| 左偏树 | `dist` 保证左偏，只沿右链合并 | 可并堆 | P3377【模板】左偏树 |
| 线段树合并 | 一方为空直接返回 | 子树信息合并 | P4556 雨天的尾巴 |
| ODT | `set` 存区间，`split` 对齐 | 区间推平 | CF 896C、P2572 |
| 树套树 | BIT 每个节点挂权值树 | 动态区间第 k 小 | P2617 Dynamic Rankings |

**选择**：静态 → 主席树；动态 → 整体二分/树套树；子树合并 → 线段树合并。
