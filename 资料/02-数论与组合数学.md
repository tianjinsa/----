## 1. 快速幂 / 龟速乘 / 逆元

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using i128 = __int128_t;

// a^b % mod，b >= 0
ll qpow(ll a, ll b, ll mod) {
    a %= mod;
    ll r = 1;
    while (b) {
        if (b & 1) r = (i128)r * a % mod;
        a = (i128)a * a % mod;
        b >>= 1;
    }
    return r;
}

// a*b % mod，mod 可达 1e18（龟速乘，O(log b)）
ll mul_mod(ll a, ll b, ll mod) {
    ll r = 0;
    a %= mod; b %= mod;
    while (b) {
        if (b & 1) { r += a; if (r >= mod) r -= mod; }
        a <<= 1; if (a >= mod) a -= mod;
        b >>= 1;
    }
    return r;
}

// 费马小定理求逆元（mod 必须是质数且 a % mod != 0）
ll inv_fermat(ll a, ll mod) { return qpow(a, mod - 2, mod); }
```

**注意**：`qpow` 里 `mod` 若为 1 结果恒 0，是正常的。

## 2. 扩展欧几里得 & 线性同余方程

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using i128 = __int128_t;

// 求 a*x + b*y = gcd(a,b)，返回 gcd；x,y 为特解
ll exgcd(ll a, ll b, ll &x, ll &y) {
    if (!b) { x = 1; y = 0; return a < 0 ? -a : a; }
    ll g = exgcd(b, a % b, y, x);
    y -= a / b * x;
    return g;
}

// 求 a*x ≡ b (mod m) 的最小非负解，无解返回 -1
// 等价于求 a*x + m*y = b
ll solve_linear_congruence(ll a, ll b, ll m) {
    ll x, y;
    ll g = exgcd(a, m, x, y);
    if (b % g) return -1;                 // 无解
    ll mod = m / g;
    x = (i128)(x % mod) * ((b / g) % mod) % mod;
    x = ((x % mod) + mod) % mod;
    return x;                              // 通解：x + k*mod
}

// 求 a 在模 m 下的逆元（m 不要求质数，但要求 gcd(a,m)=1），无解返回 -1
ll inv_exgcd(ll a, ll m) {
    return solve_linear_congruence(a, 1, m);
}
```

## 3. 线性筛：素数 / φ / μ / d / σ / 最小质因子

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
constexpr int MAXN = 1000000 + 5;

int primes[MAXN], pcnt;
bool isComp[MAXN];
int phi[MAXN];        // 欧拉函数
int mu[MAXN];         // 莫比乌斯函数
int d[MAXN];          // 约数个数
int dmin[MAXN];       // 最小质因子的指数 (用于算 d)
int sigma[MAXN];      // 约数之和
int smin[MAXN];       // 最小质因子的等比和 1+p+...+p^k
int minp[MAXN];       // 最小质因子

void sieve(int n) {
    phi[1] = mu[1] = d[1] = sigma[1] = 1;
    dmin[1] = smin[1] = minp[1] = 0;
    for (int i = 2; i <= n; ++i) {
        if (!isComp[i]) {
            primes[pcnt++] = i;
            minp[i] = i;
            phi[i] = i - 1;
            mu[i] = -1;
            d[i] = 2; dmin[i] = 1;
            sigma[i] = i + 1; smin[i] = i + 1;
        }
        for (int j = 0; j < pcnt && (ll)i * primes[j] <= n; ++j) {
            int p = primes[j], x = i * p;
            isComp[x] = true;
            minp[x] = p;
            if (i % p == 0) {
                phi[x] = phi[i] * p;
                mu[x] = 0;
                dmin[x] = dmin[i] + 1;
                d[x] = d[i] / (dmin[i] + 1) * (dmin[x] + 1);
                smin[x] = smin[i] * p + 1;
                sigma[x] = sigma[i] / smin[i] * smin[x];
                break;
            } else {
                phi[x] = phi[i] * (p - 1);
                mu[x] = -mu[i];
                dmin[x] = 1;
                d[x] = d[i] * 2;
                smin[x] = p + 1;
                sigma[x] = sigma[i] * (p + 1);
            }
        }
    }
}
```

**单个数求 φ（试除法）**

## 4. 组合数（预处理阶乘逆元）

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using i128 = __int128_t;

template <int MOD>
struct Comb {
    vector<int> fac, ifac;
    explicit Comb(int n) : fac(n + 1), ifac(n + 1) {
        fac[0] = 1;
        for (int i = 1; i <= n; ++i) fac[i] = (ll)fac[i - 1] * i % MOD;
        ifac[n] = qpow(fac[n], MOD - 2);
        for (int i = n; i >= 1; --i) ifac[i - 1] = (ll)ifac[i] * i % MOD;
    }
    static int qpow(ll a, ll b) {
        ll r = 1; a %= MOD;
        while (b) { if (b & 1) r = r * a % MOD; a = a * a % MOD; b >>= 1; }
        return (int)r;
    }
    int C(int n, int m) const {
        if (m < 0 || m > n) return 0;
        return (ll)fac[n] * ifac[m] % MOD * ifac[n - m] % MOD;
    }
    int A(int n, int m) const {          // 排列数
        if (m < 0 || m > n) return 0;
        return (ll)fac[n] * ifac[n - m] % MOD;
    }
    int Catalan(int n) const {           // 第 n 个卡特兰数
        if (n <= 0) return 1;
        return (ll)C(2 * n, n) * (MOD + 1) / (n + 1) % MOD;
    }
};

// 用法：Comb<1000000007> comb(200000);
```

> **陷阱**：`Comb` 里 `(MOD + 1) / (n + 1)` 只在能整除时成立。正确写法是 `C(2n,n) * inv(n+1)`：
> `int Catalan(int n) const { return (ll)C(2*n,n) * qpow(n+1, MOD-2) % MOD; }`

## 5. 卢卡斯定理 Lucas（模数小质数）

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

// C(n,m) mod p，p 为质数，p 可以很小（如 1e5）
struct Lucas {
    int p;
    vector<ll> fac, ifac;
    explicit Lucas(int prime, int maxn = 0) : p(prime) {
        int n = maxn ? maxn : p - 1;
        fac.resize(n + 1); ifac.resize(n + 1);
        fac[0] = 1;
        for (int i = 1; i <= n; ++i) fac[i] = fac[i - 1] * i % p;
        ifac[n] = pw(fac[n], p - 2);
        for (int i = n; i >= 1; --i) ifac[i - 1] = ifac[i] * i % p;
    }
    ll pw(ll a, ll b) const {
        ll r = 1; a %= p;
        while (b) { if (b & 1) r = r * a % p; a = a * a % p; b >>= 1; }
        return r;
    }
    ll C_small(ll n, ll m) const {
        if (m < 0 || m > n) return 0;
        return fac[n] * ifac[m] % p * ifac[n - m] % p;
    }
    ll C(ll n, ll m) const {              // Lucas
        if (m < 0 || m > n) return 0;
        ll r = 1;
        while (n || m) {
            int ni = (int)(n % p), mi = (int)(m % p);
            if (mi > ni) return 0;
            r = r * C_small(ni, mi) % p;
            n /= p; m /= p;
        }
        return r;
    }
};
```

## 6. 数论分块（整除分块）

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

// 求 sum_{i=1}^{n} floor(n/i)
ll sum_floor_div(ll n) {
    ll ans = 0;
    for (ll l = 1, r; l <= n; l = r + 1) {
        r = n / (n / l);
        ans += (n / l) * (r - l + 1);
    }
    return ans;
}
```

## 7. 莫比乌斯反演常用结论

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

// 预处理 mu 前缀和 muPre 后：
// sum_{i=1}^{n} sum_{j=1}^{m} [gcd(i,j)==1]
//  = sum_{d=1}^{min(n,m)} mu(d) * floor(n/d) * floor(m/d)
// 用数论分块 O(sqrt) 计算。

// 常用恒等式：
//  [gcd(i,j)==1] = sum_{d | gcd(i,j)} mu(d)
//  sum_{d|n} phi(d) = n
//  sum_{d|n} mu(d) = [n==1]
//  f(n)=sum_{d|n} g(d)  =>  g(n)=sum_{d|n} mu(n/d) f(d)
```

**模板：`Σ Σ [gcd=1]`**

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

struct Mobius {
    vector<int> primes, mu, pre;
    vector<bool> comp;
    explicit Mobius(int n) : mu(n + 1), pre(n + 1), comp(n + 1) {
        mu[1] = 1;
        for (int i = 2; i <= n; ++i) {
            if (!comp[i]) { primes.push_back(i); mu[i] = -1; }
            for (int p : primes) {
                if ((ll)i * p > n) break;
                comp[i * p] = true;
                if (i % p == 0) { mu[i * p] = 0; break; }
                mu[i * p] = -mu[i];
            }
        }
        for (int i = 1; i <= n; ++i) pre[i] = pre[i - 1] + mu[i];
    }
    ll count_coprime(ll n, ll m) {   // sum_{i<=n,j<=m} [gcd==1]
        ll ans = 0, N = min(n, m);
        for (ll l = 1, r; l <= N; l = r + 1) {
            r = min(n / (n / l), m / (m / l));
            ans += (ll)(pre[r] - pre[l - 1]) * (n / l) * (m / l);
        }
        return ans;
    }
};
```

## 8. 扩展欧拉定理（降幂）

> 当 `b >= φ(m)` 时：`a^b ≡ a^(b mod φ(m) + φ(m)) (mod m)`

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

ll phi(ll n) {
    ll r = n;
    for (ll p = 2; p * p <= n; ++p)
        if (n % p == 0) { r = r / p * (p - 1); while (n % p == 0) n /= p; }
    if (n > 1) r = r / n * (n - 1);
    return r;
}

ll qpow(ll a, ll b, ll m) {
    ll r = 1 % m; a %= m;
    while (b) { if (b & 1) r = r * a % m; a = a * a % m; b >>= 1; }
    return r;
}

// a^b mod m，b 以十进制字符串给出（可极大）
ll pow_big(ll a, const string &b, ll m) {
    ll p = phi(m), e = 0;
    bool big = false;
    for (char c : b) {
        e = e * 10 + (c - '0');
        if (e >= p) { big = true; e %= p; }
    }
    if (big) e += p;
    return qpow(a, e, m);
}
```

> 注意：若 `b < φ(m)`，直接算；上面的写法在 `b` 很大时才加 `φ(m)`，已处理。

## 10. 矩阵快速幂（斐波那契 / 线性递推）

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

template <int N, int MOD>
struct Mat {
    ll a[N][N];
    Mat(bool ident = false) {
        memset(a, 0, sizeof(a));
        if (ident) for (int i = 0; i < N; ++i) a[i][i] = 1;
    }
    Mat operator*(const Mat &o) const {
        Mat r;
        for (int i = 0; i < N; ++i)
            for (int k = 0; k < N; ++k) {
                if (!a[i][k]) continue;
                for (int j = 0; j < N; ++j)
                    r.a[i][j] = (r.a[i][j] + a[i][k] * o.a[k][j]) % MOD;
            }
        return r;
    }
    Mat pow(ll e) const {
        Mat base = *this, r(true);
        while (e) { if (e & 1) r = r * base; base = base * base; e >>= 1; }
        return r;
    }
};

// 斐波那契：F(1)=1,F(2)=1 => F(n) = mat^(n-2)
ll fib(ll n) {
    if (n <= 2) return 1;
    Mat<2, 1000000007> m;
    m.a[0][0] = 1; m.a[0][1] = 1; m.a[1][0] = 1; m.a[1][1] = 0;
    m = m.pow(n - 2);
    return (m.a[0][0] + m.a[0][1]) % 1000000007;
}
```

## 11. 快速倍增法求斐波那契（Fast Doubling）★最优

> **比矩阵快速幂快得多**：每次只做 3 次乘法 + 2 次加法（矩阵版是 8 次乘法），常数极小。
> **用途**：求 `F(n) mod M`（`n` 可达 `1e18`）、求前 `n` 项和、求"广义斐波那契"的幂次。

### 11.1 核心公式

已知 `a = F_k`，`b = F_{k-1}`，则下标翻倍：

```
F_{2k}   = a * (a + 2b)
F_{2k-1} = a² + b²
F_{2k+1} = F_{2k} + F_{2k-1}
```

**推导**：由矩阵形式 `[[F_{k+1}, F_k], [F_k, F_{k-1}]] = [[1,1],[1,0]]^k`，两边平方得
`[[F_{2k+1}, F_{2k}], [F_{2k}, F_{2k-1}]]`，展开左下角与右下角即得上式。

### 11.2 递归版（最好背）

设 `solve(n)` 返回 `{F_n, F_{n-1}}`，`k = n/2`：

- `n` 偶：返回 `{F_{2k}, F_{2k-1}}`
- `n` 奇：返回 `{F_{2k+1}, F_{2k}} = {c + d, c}`（`c = F_{2k}`, `d = F_{2k-1}`）
- 边界：`n = 0` 返回 `{0, 1}`（利用广义定义 `F_{-1} = 1`）

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

// 返回 {F(n), F(n-1)} mod M
pair<ll, ll> fib_pair(ll n, ll M) {
    if (n == 0) return {0, 1 % M};          // F(0)=0, F(-1)=1
    auto [a, b] = fib_pair(n >> 1, M);      // a = F(k), b = F(k-1), k = n/2
    ll two_b = (2 * b) % M;
    ll c = a * ((a + two_b) % M) % M;       // F(2k)   = a(a+2b)
    ll d = (a * a % M + b * b % M) % M;     // F(2k-1) = a² + b²
    if (n & 1) return {(c + d) % M, c};     // n 奇：{F(2k+1), F(2k)}
    return {c, d};                          // n 偶：{F(2k), F(2k-1)}
}

ll fib(ll n, ll M = 1000000007) { return fib_pair(n, M).first; }
```

### 11.3 迭代版（无递归栈，O(1) 空间）

按二进制位从左到右"翻倍 + 加一"：

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

// 迭代快速倍增：从高位到低位处理 n 的二进制
ll fib_iter(ll n, ll M) {
    ll a = 0, b = 1;                        // a = F(0), b = F(1)
    // 处理 n 的每一位（从最高位到最低位）
    for (int bit = 62; bit >= 0; --bit) {
        // 翻倍：{a,b} = {F(k), F(k+1)} -> {F(2k), F(2k+1)}
        ll c = a * ((2 * b % M - a + M) % M) % M;   // F(2k)   = a(2b - a)
        ll d = (a * a % M + b * b % M) % M;         // F(2k+1) = a² + b²
        if ((n >> bit) & 1) { a = d; b = (c + d) % M; }   // +1: {F(2k+1), F(2k+2)}
        else { a = c; b = d; }
    }
    return a;                               // a = F(n)
}
```

> **迭代版用的等价形式**：若维护 `{F_k, F_{k+1}}`，则
> `F_{2k} = F_k (2F_{k+1} - F_k)`，`F_{2k+1} = F_k² + F_{k+1}²`。
> 这与递归版完全等价（把 `b = F_{k+1}` 换元即可），但**少一次乘法**。

### 11.4 前 n 项和

```
Σ_{i=0}^{n} F_i = F_{n+2} - 1
```

证明（裂项相消）：`F_i = F_{i+2} - F_{i+1}`，求和后中间项全部抵消，只剩 `F_{n+2} - F_1 = F_{n+2} - 1`。

### 11.5 实例

| 题 | 用法 |
|---|---|
| P1962 斐波那契数列 | 直接输出 `fib(n, 1e9+7)`，`n <= 1e18` |
| P2626 斐波那契数列（升级版） | `fib(n, 2^31)` + 分解质因数 |
| P1255 数楼梯 | `fib(n+1)`（高精度版需手写大整数） |
| P3986 斐波那契数列 | 枚举 + 快速倍增判存在 |
| 广义斐波那契 `G(n) = pG(n-1) + qG(n-2)` | 用矩阵快速幂；**若 `q = 1` 可用快速倍增的推广** |

**广义斐波那契的倍增推广**（`F_{n} = pF_{n-1} + qF_{n-2}`）：
由矩阵 `[[p, q], [1, 0]]^k` 的平方可得，但公式较繁；`q = 1` 时有
`F_{2k} = F_k (2F_{k+1} - p F_k)`，`F_{2k+1} = F_k² · q^k + F_{k+1}²`（需要额外维护 `q^k`）。
**比赛建议**：广义情形直接用矩阵快速幂，别硬套。

### 11.6 对比

| 方法 | 每次迭代乘法数 | 常数 | 代码量 |
|---|---|---|---|
| 矩阵快速幂 | 8 | 大 | 中 |
| **快速倍增（递归）** | **3** | **小** | **极小** |
| 快速倍增（迭代） | 3 | 极小 | 小 |

**结论**：求斐波那契一律用快速倍增，`O(log n)`、常数最小、最好背。

### 11.7 常见坑

1. `F_{-1} = 1` 是递归边界能成立的关键，别写成 0。
2. 取模时 `2b` 可能溢出 `int`，用 `ll`。
3. 迭代版 `2 * b % M - a` 可能为负，要 `+ M`。
4. `n = 0` 时返回 0；题目若从 `F_1 = 1` 开始，注意下标偏移。
5. 求 `F(n) mod m` 且 `m` 不是质数时**依然成立**（不需要逆元）。

### 怎么用（照抄这段 `main`，只改【输入】【输出】两处）

上面是模板本体。实际写题时：**把 `qpow` / `fib_pair` / `fib` 整段粘过来 → 再粘这段 `main` → 只改标了【输入】【输出】的行**。

```cpp skip
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

// ===== 粘贴点：把上面的 qpow / fib_pair / fib 粘到这里 =====

int main() {
    // ===================== 【输入】改这里 =====================
    ll a, b, m;
    scanf("%lld %lld %lld", &a, &b, &m);  // 求 a^b mod m
    ll n;
    scanf("%lld", &n);                    // 求 F(n) mod 1e9+7
    // ========================================================

    // ===================== 【核心】算法流程 =====================
    // 快速幂：把指数 b 看成二进制，不断"平方底数、右移指数"，
    //         遇到 1 就把当前底数乘进答案。O(log b)
    // 快速倍增：已知 F(k)、F(k-1)，直接算出 F(2k)、F(2k-1)
    //   F(2k)   = F(k) * (F(k) + 2*F(k-1))
    //   F(2k-1) = F(k)^2 + F(k-1)^2
    //   然后按 n 的奇偶挑一组返回，递归 n/2。O(log n)
    ll p = qpow(a, b, m);
    ll f = fib(n);                        // 默认模 1000000007
    // ========================================================

    // ===================== 【输出】改这里 =====================
    printf("%lld\n%lld\n", p, f);
    // ========================================================
    return 0;
}
```
