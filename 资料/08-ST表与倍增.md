# 08 ST 表与倍增

> 你点名的内容。ST 表：`O(n log n)` 预处理、`O(1)` 查询，**只支持可重复贡献**的运算（max/min/gcd/or/and）。倍增：一切"跳 2^k 步"的套路。

---

## 1. ST 表：区间最值（RMQ）★

```cpp
#include <bits/stdc++.h>
using namespace std;

struct SparseTable {
    int n, LOG;
    vector<vector<int>> st;

    explicit SparseTable(const vector<int> &a) {          // a 下标 1..n
        n = (int)a.size() - 1;
        LOG = n ? 32 - __builtin_clz((unsigned)n) : 0;
        st.assign(LOG + 1, vector<int>(n + 1));
        for (int i = 1; i <= n; ++i) st[0][i] = a[i];
        for (int k = 1; k <= LOG; ++k)
            for (int i = 1; i + (1 << k) - 1 <= n; ++i)
                st[k][i] = min(st[k - 1][i], st[k - 1][i + (1 << (k - 1))]);
    }

    // [l, r] 最小值，O(1)
    int query(int l, int r) const {
        int k = 31 - __builtin_clz((unsigned)(r - l + 1));
        return min(st[k][l], st[k][r - (1 << k) + 1]);
    }
};
```

**为什么能 `O(1)`**：`min` 可重复贡献（overlap 不影响结果）。`max/gcd/or/and` 同理。
**不能用于**：区间和、区间积（重叠会重复计算）→ 用前缀和或线段树。

## 2. ST 表：任意可重复贡献运算（模板化）

```cpp
#include <bits/stdc++.h>
using namespace std;

template <class T, class Op>
struct GenericST {
    int n, LOG;
    vector<vector<T>> st;
    Op op;

    GenericST(const vector<T> &a, Op op_) : op(op_) {
        n = (int)a.size() - 1;
        LOG = n ? std::__lg(n) + 1 : 0;
        st.assign(LOG + 1, vector<T>(n + 1));
        for (int i = 1; i <= n; ++i) st[0][i] = a[i];
        for (int k = 1; k <= LOG; ++k)
            for (int i = 1; i + (1 << k) - 1 <= n; ++i)
                st[k][i] = op(st[k - 1][i], st[k - 1][i + (1 << (k - 1))]);
    }
    T query(int l, int r) const {
        int k = std::__lg(r - l + 1);
        return op(st[k][l], st[k][r - (1 << k) + 1]);
    }
};
// 用法：
// GenericST<int, function<int(int,int)>> st(a, [](int x, int y){ return max(x, y); });
// 或直接传 std::gcd<int> 风格函数对象
```

## 3. ST 表：二维 RMQ（矩阵最值）

`st[k][i][j]` 表示左上角 `(i,j)`、边长 `2^k` 的正方形最值。查询矩形用 4 个块。

```cpp
#include <bits/stdc++.h>
using namespace std;

struct ST2D {
    int n, m, LOG;
    vector<vector<vector<int>>> st;

    ST2D(const vector<vector<int>> &a) {          // a 下标 [1..n][1..m]
        n = (int)a.size() - 1;
        m = (int)a[0].size() - 1;
        LOG = std::__lg(max(n, m)) + 1;
        st.assign(LOG + 1, vector<vector<int>>(n + 1, vector<int>(m + 1, 0)));
        for (int i = 1; i <= n; ++i)
            for (int j = 1; j <= m; ++j) st[0][i][j] = a[i][j];
        for (int k = 1; k <= LOG; ++k) {
            int d = 1 << (k - 1);
            for (int i = 1; i + (1 << k) - 1 <= n; ++i)
                for (int j = 1; j + (1 << k) - 1 <= m; ++j)
                    st[k][i][j] = min({st[k - 1][i][j],
                                       st[k - 1][i + d][j],
                                       st[k - 1][i][j + d],
                                       st[k - 1][i + d][j + d]});
        }
    }
    int query(int x1, int y1, int x2, int y2) const {
        int k = std::__lg(min(x2 - x1 + 1, y2 - y1 + 1));
        int d = 1 << k;
        return min({st[k][x1][y1], st[k][x2 - d + 1][y1],
                    st[k][x1][y2 - d + 1], st[k][x2 - d + 1][y2 - d + 1]});
    }
};
```

## 4. 倍增：最近公共祖先 LCA

```cpp
#include <bits/stdc++.h>
using namespace std;

struct LCA {
    int n, LOG;
    vector<vector<int>> up, g;
    vector<int> dep;

    LCA(int n_, int root = 1) : n(n_), LOG(std::__lg(max(1, n_)) + 1),
        up(LOG + 1, vector<int>(n_ + 1, 0)), g(n_ + 1), dep(n_ + 1, 0) {
        // 之后 addEdge + build(root)
        (void)root;
    }
    void addEdge(int u, int v) { g[u].push_back(v); g[v].push_back(u); }

    void dfs(int u, int fa) {
        up[0][u] = fa; dep[u] = dep[fa] + 1;
        for (int k = 1; k <= LOG; ++k) up[k][u] = up[k - 1][up[k - 1][u]];
        for (int v : g[u]) if (v != fa) dfs(v, u);
    }
    void build(int root = 1) { dep[root] = 0; up[0][root] = root; dfs(root, root); }

    int lca(int u, int v) const {
        if (dep[u] < dep[v]) swap(u, v);
        int d = dep[u] - dep[v];
        for (int k = 0; k <= LOG; ++k) if (d >> k & 1) u = up[k][u];
        if (u == v) return u;
        for (int k = LOG; k >= 0; --k)
            if (up[k][u] != up[k][v]) { u = up[k][u]; v = up[k][v]; }
        return up[0][u];
    }
    int dist(int u, int v) const { return dep[u] + dep[v] - 2 * dep[lca(u, v)]; }
    int kth(int u, int k) const {                 // u 向上走 k 步
        for (int j = 0; j <= LOG; ++j) if (k >> j & 1) u = up[j][u];
        return u;
    }
};
```

## 5. 倍增：跳表 / 快速幂式跳跃

```cpp
#include <bits/stdc++.h>
using namespace std;

// 例：nxt[i] 表示 i 的下一步（函数图），求从 s 出发走 k 步到达的位置
// f[j][i] = 从 i 走 2^j 步到达的点
struct BinaryLifting {
    int n, LOG;
    vector<vector<int>> f;
    BinaryLifting(const vector<int> &nxt, int LOG_) : n((int)nxt.size()), LOG(LOG_),
        f(LOG_ + 1, vector<int>(nxt.size())) {
        f[0] = nxt;
        for (int j = 1; j <= LOG; ++j)
            for (int i = 1; i < n; ++i) f[j][i] = f[j - 1][f[j - 1][i]];
    }
    int jump(int s, long long k) const {
        for (int j = 0; j <= LOG; ++j) if (k >> j & 1) s = f[j][s];
        return s;
    }
};
```

**应用**：函数图找环、树上 k 级祖先、区间跳跃（"跳石头"）、倍增求 LIS 的 `nxt` 指针、AC 自动机的 fail 树倍增。

## 6. 倍增思想的其他应用

| 场景 | 倍增维护什么 |
|---|---|
| LCA | 祖先 `up[k][u]` |
| 树上路径第 k 个点 | 祖先表 |
| 树上路径最值/和 | `mx[k][u]`、`sum[k][u]` |
| 城市跳跃（P1613 跑路） | 图上的 `2^k` 步可达性 + Floyd |
| 区间覆盖/跳跃游戏 | `nxt[k][i]` = 跳 `2^k` 次的位置 |
| 求 LIS 的贪心 | 维护 `f[i]`，二分替换（不是倍增，但同类） |
| 树上倍增求"第一个满足条件的祖先" | `up` 表 + 从高到低试 |

**树上倍增 + 阈值**（求第一个 `val >= x` 的祖先）：

```cpp
#include <bits/stdc++.h>
using namespace std;
// 核心：从高到低试跳，若 up[k][u] 不满足条件就跳上去，最后答案 = up[0][u]
int first_ancestor_ge(int u, int x, const vector<vector<int>> &up,
                      const vector<vector<int>> &mx, int LOG) {
    if (mx[0][u] >= x) return -1;              // 父亲都不满足
    int cur = u;
    for (int k = LOG; k >= 1; --k)             // 注意从 1 开始，最后一步单独走
        if (mx[k][cur] < x) cur = up[k][cur];
    return up[0][cur];                          // 此时父亲必满足
}
```

## 7. ST 表 + 单调栈：直方图最大矩形（细节）

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

ll max_histogram_rect(const vector<ll> &h) {        // h 下标 1..n
    int n = (int)h.size() - 1;
    vector<int> L(n + 1), R(n + 1), stk;
    for (int i = 1; i <= n; ++i) {
        while (!stk.empty() && h[stk.back()] >= h[i]) stk.pop_back();
        L[i] = stk.empty() ? 1 : stk.back() + 1;
        stk.push_back(i);
    }
    stk.clear();
    for (int i = n; i >= 1; --i) {
        while (!stk.empty() && h[stk.back()] >= h[i]) stk.pop_back();
        R[i] = stk.empty() ? n : stk.back() - 1;
        stk.push_back(i);
    }
    ll ans = 0;
    for (int i = 1; i <= n; ++i) ans = max(ans, h[i] * (R[i] - L[i] + 1));
    return ans;
}
```

## 8. 小结

- **可重复贡献运算** → ST 表（max/min/gcd/or/and）。
- **需要合并两个不重叠区间** → 线段树 / 前缀和。
- **`O(1)` 查询、不改** → ST 表。
- **带修改** → 线段树。
- **倍增**：一切"跳 `2^k`"的题；建表 `O(n log n)`，单次跳 `O(log n)`。

## 9. 思路与例题索引

| 算法 | 思路一句话 | 用途 | 例题 |
|---|---|---|---|
| ST 表 | `st[k][i]` = 从 i 起 2^k 个的最值，两块重叠合并 | 静态区间最值 `O(1)` | P3865【模板】ST 表 |
| 泛型 ST | 传入 `op`，要求可重复贡献 | 区间 gcd/or/and | P1890 gcd 区间 |
| 二维 ST | 四个角取最值 | 子矩阵最值 | P2216 理想的正方形 |
| 倍增 LCA | `up[k][u]` 跳 2^k 级祖先 | 树上 LCA / 距离 | P3379【模板】LCA |
| 倍增跳表 | `f[k][i]` = 走 2^k 步 | 函数图、区间跳跃 | P1613 跑路 |
| 倍增 + 阈值 | 从高到低试跳，找"第一个满足条件" | 路径上第一个 >= x | P1967 货车运输 |
| 单调栈 | 维护单调，弹出即确定边界 | 直方图最大矩形 | P4147 玉蟾宫 |

**判断**：`min/max/gcd/or/and` 可重复贡献 → ST 表；区间和/积 → 前缀和或线段树。
